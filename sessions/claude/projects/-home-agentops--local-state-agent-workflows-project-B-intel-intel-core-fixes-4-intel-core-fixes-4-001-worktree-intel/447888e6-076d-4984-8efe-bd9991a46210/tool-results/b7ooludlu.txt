    72	/** Recursively frozen, so a step that mutated its input would throw. */
    73	function deepFreeze<T>(value: T): T {
    74	  if (typeof value === "object" && value !== null && !Object.isFrozen(value)) {
    75	    Object.freeze(value);
    76	    for (const child of Object.values(value)) deepFreeze(child);
    77	  }
    78	  return value;
    79	}

    81	function viewOf(tick: number, world: World): IntelWorldView {
    82	  return { tick, players: Object.entries(world.players).map(([id, at]) => ({ id, ...at })), rubbings: world.rubbings };
    83	}

    85	/** Steps with frozen input and state, checking neither changes. */
    86	class Run {
    87	  state: IntelState;
    88	  constructor(
    89	    readonly config: IntelConfig,
    90	    seed = 1,
    91	  ) {
    92	    this.state = createIntelState(config, seed);
    93	  }
    94	
    95	  step(tick: number, world: World, requests: TargetedRumorRequest[] = []): IntelStepResult {
    96	    const view = deepFreeze(viewOf(tick, world));
    97	    const before = hashState(deepFreeze(this.state));
    98	    const stepped = stepIntel(this.config, this.state, view, deepFreeze(requests));
    99	    expect(hashState(this.state)).toBe(before);
   100	    this.state = stepped.state;
   101	    return stepped;
   102	  }
   103	
   104	  available(tick: number, world: World, playerId: string, type: RubbingType) {
   105	    return targetedRumorAvailability(this.config, this.state, viewOf(tick, world), playerId, type);
   106	  }
   107	}

   109	function to(result: IntelStepResult, playerId: string): IntelMessageBody[] {
   110	  return result.messages.filter((message) => message.to.kind === "player" && message.to.playerId === playerId).map((message) => message.body);
   111	}
   112	
   113	function everyone(result: IntelStepResult): IntelMessageBody[] {
   114	  return result.messages.filter((message) => message.to.kind === "everyone").map((message) => message.body);
   115	}

   117	function only<K extends IntelMessageBody["kind"]>(bodies: IntelMessageBody[], kind: K): Extract<IntelMessageBody, { kind: K }> {
   118	  const found = bodies.filter((body) => body.kind === kind);
   119	  expect(found).toHaveLength(1);
   120	  return found[0] as Extract<IntelMessageBody, { kind: K }>;
   121	}
