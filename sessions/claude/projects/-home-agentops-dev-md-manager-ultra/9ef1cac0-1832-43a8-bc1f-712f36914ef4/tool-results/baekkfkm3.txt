class FakeSessions:
    def __init__(self, directory, plan):
        self.directory, self.plan = directory, plan
        self.starts = []

    def run(self, node):
        self.starts.append(node)
        cwd = Path(self.plan["nodes"][node]["worktree"])
        (cwd / ("ui.txt" if node == "ui" else "backend.py")).write_text("after" if node == "ui" else "VALUE = 2\n")
        receipt = {"session_id": self.plan["nodes"][node]["session_id"], "status": "fake-worker-completed"}
        save_json(self.directory / f"{node}.interactive.json", receipt)
        save_json(self.directory / f"{node}.handoff.json", {"summary": "Synthetic implementation for offline test", "open_assumptions": []})
        return receipt


class OfflinePipeline(Pipeline):
    def stop_workers(self):
        self.event("freeze", "stopped", "Fake workers have no background processes")

            self.assertEqual(sorted(self.sessions.starts), ["adapter", "ui"])

            self.assertGreaterEqual(len(screenshots), 2)  # Worker + combined candidate.
            self.assertTrue(all(path.read_bytes().startswith(b"\x89PNG") for path in screenshots))
            # Lane temp directories must be short enough for Unix sockets (tsx IPC, Chromium)
            # and must not linger after the lane, while all evidence stays inside the run.

    def test_failed_check_reuses_worker_and_successful_sibling(self):
        self.fail_marker.touch()
        with SqliteSaver.from_conn_string(str(self.directory / "graph.sqlite")) as saver:
            graph = build_pipeline(saver, self.runtime)

    def test_configured_failure_drill_reopens_checkpoint_and_enforces_attempt_cap(self):

    def native_rows(self):
        return {node: {"id": f"id-{node}", "sessionId": f"session-{node}", "pid": 90000 + index}
                for index, node in enumerate(("ui", "adapter"))}

    def set_native(self, live):
        self.runtime.sessions = SimpleNamespace(executable="claude", inventory=lambda: list(live.values()),
                                               locate=lambda node, rows: next((row for row in rows if row["id"] == f"id-{node}"), None))

    def test_native_stop_success_and_partial_recovery(self):
        live = self.native_rows()
        self.set_native(live)
        def stop(argv, **_kwargs):
            node = argv[-1].removeprefix("id-")
            live.pop(node)
            return subprocess.CompletedProcess(argv, 0)
        with patch("workflow.pipeline.subprocess.run", side_effect=stop) as command, patch("workflow.pipeline.pid_alive", return_value=False):
            Pipeline.stop_workers(self.runtime)
            self.assertEqual(command.call_count, 2)
            Pipeline.stop_workers(self.runtime)
            self.assertEqual(command.call_count, 2)
        self.assertTrue(all(read_json(self.directory / f"{node}.stop.json")["stopped"] for node in ("ui", "adapter")))

    def test_completed_stop_intent_is_reconciled_without_another_stop(self):
        for node, row in self.native_rows().items():
            save_json(self.directory / f"{node}.stop.json", {"background_id": row["id"], "session_id": row["sessionId"], "pid": row["pid"], "stopped": False})
        self.set_native({})
        with patch("workflow.pipeline.subprocess.run") as command, patch("workflow.pipeline.pid_alive", return_value=False):
            Pipeline.stop_workers(self.runtime)
            command.assert_not_called()

    def test_failed_stop_and_lingering_pid_block_freeze(self):
        live = self.native_rows()
        self.set_native(live)
        with patch("workflow.pipeline.subprocess.run", return_value=subprocess.CompletedProcess([], 1)):
            with self.assertRaisesRegex(RuntimeError, "Stop failed"):
                Pipeline.stop_workers(self.runtime)
        self.assertFalse(read_json(self.directory / "ui.stop.json")["stopped"])
        with patch("workflow.pipeline.subprocess.run", return_value=subprocess.CompletedProcess([], 0)), patch("workflow.pipeline.pid_alive", return_value=True):
            with self.assertRaisesRegex(RuntimeError, "termination"):
                Pipeline.stop_workers(self.runtime)

    def test_missing_handoff_does_not_stop_workers(self):
        with patch.object(self.runtime, "stop_workers") as stop:
            with self.assertRaisesRegex(ValueError, "Missing ui handoff"):
                self.runtime.freeze()
            stop.assert_not_called()

    def test_ownership_violation_blocks_snapshot(self):
        self.sessions.run("ui"); self.sessions.run("adapter")
        (Path(self.plan["nodes"]["ui"]["worktree"]) / "backend.py").write_text("UNOWNED = True\n")
        with self.assertRaisesRegex(ValueError, "unowned"):
            self.runtime.freeze()

## Deliberate operator boundaries

- Select the feature, ownership and actual acceptance tests before preparing a run; the example policy is illustrative.
- Tests and setup commands are trusted code, not sandboxed workloads.
- Native terminal edits must stop before frozen verification.
- Independent review is supplied by Pi/a human as an explicit artifact; the graph never invents approval.
- Ambiguous native sessions, partial candidate allocations, source drift and exhausted usage stop for operator action, not silent relaunch/provider switching.
- `report.html` is a local snapshot viewer; controls remain authenticated-by-local-operator CLI actions, not a remote web service.
- Run/worktree cleanup and pushing remain separate, explicit actions.

# Reusable verification policy and evidence gates

Feature requirements are supplied per run. The gate implementation stays reusable.

## Defaults

| Worker | Required | Add when relevant |
| --- | --- | --- |
| Frontend | Build; browser tests for every named scenario; PNG screenshot per scenario | Typecheck, accessibility checks, mobile/desktop scenarios, visual regression assertions |
| Backend | Unit tests with nonzero passing test count | API/schema contract tests; isolated database/adapter integration tests; failure/recovery tests |
| Both | Ownership, exact run/attempt/revision identity, executed check logs, matching artifact hashes | Task-specific checks |

Independent review and integration approval remain mandatory. A screenshot alone is not a browser test, and a successful shell exit with zero executed tests is not sufficient test evidence. Screenshots should be examined during independent review; this gate checks their presence/hash/format, not visual correctness.

## Files

- `contracts/workflow/verification.schema.json`: authoritative JSON Schema for per-feature policies.
- `contracts/workflow/verificationEvidence.schema.json`: trusted verifier evidence sidecar; leaves existing v1 `WorkerResult` unchanged.
- `contracts/workflow/verification.example.json`: illustrative viewer policy. Its future browser test path is an example, not an implemented test suite.
- `workflow/verification.py`: policy validator and deterministic evidence evaluator.
- `workflow/test_verification.py`: positive/negative fixture tests.

These two supplementary schemas are authored directly, not generated by the existing Zod export script. The original v1 message schemas remain unchanged. Non-Python consumers must also implement the documented cross-field rules.

## Configure a feature

Each worker declares its role, node ID, exact owned path prefixes, and required checks. A check has a unique ID, kind, argv array, timeout and (for browser checks only) named scenarios. There are no ownership globs; `src/workflow` owns itself and descendants, not `src/workflow-other`. Worker ownership cannot overlap.

All listed checks are required. Frontend policies cannot omit build or browser checks; backend policies cannot omit unit tests. Browser scenarios cannot be an empty list. For responsive work, name desktop/mobile cases explicitly and implement the corresponding viewport assertions and screenshot capture in the browser test.

Policy commands are operator-approved configuration, never code invented by a worker at the join. An argv array does not make a command safe. The future runner must use `shell=False`, authorize the command, enforce timeouts and execute it in a separate verification worktree with isolated ports/caches/artifact paths. Do not execute arbitrary commands supplied by an untrusted UI or agent.

## Evidence binding

The sidecar binds the policy SHA-256, run ID, worker ID, attempt and output commit. Canonical policy JSON uses sorted keys, no whitespace separators and literal Unicode encoded as UTF-8. Each check ID points to one index in `WorkerResult.checks`; command text must equal Python `shlex.join(argv)`. The executed working directory must match the backend-owned verification worktree.

Test check receipts require parsed counts of passed/failed/skipped tests. Browser receipts additionally require exact scenario IDs, passed status, and a screenshot artifact for each. No-test runs, failed/skipped required scenarios, missing logs/screenshots, stale revisions, duplicate IDs and modified artifact bytes block the gate. Build/typecheck checks do not require test counts.

All artifact paths are resolved through a **backend-owned registry**, constrained to its root and hash-checked. Agent-provided URI strings are not fetched or treated as arbitrary paths. PNG header checks are format sanity only, not image decoding or proof that an image came from a browser.

## Boundary and remaining implementation

This slice evaluates evidence; it does **not** run commands, launch a browser, take screenshots, launch Claude, or advance the live graph. Its tests use clearly synthetic local fixtures.

Before wiring it into the live graph:

1. Freeze/stop interactive edits and independently derive the complete Git diff and immutable output commit. Worker-reported changed files alone cannot prove ownership.
2. Capture trusted command executions, parse actual test-runner results and collect browser screenshots in isolated verification worktrees. Never accept a worker's self-written sidecar as trusted execution proof.
3. Supply backend-owned expected run/node/attempt/base/output identity and verification cwd to `evaluate_worker`.
4. Preserve immutable logs/artifacts, then run independent review against those exact outputs.
5. Obtain explicit integration approval; do not infer it from a passing evidence gate or an idle terminal.

`evaluate_worker` always returns `integration_allowed: false`, with independent review and approval listed as pending gates—even if technical evidence passes. Unresolved worker assumptions are retained for that review. The live graph remains paused at its existing handoff until these remaining pieces are implemented.

Run local tests without any live agent launches:

```bash
python -m unittest workflow.test_verification workflow.test_interactive workflow.test_sessions workflow.test_graph -v
npm run test:contracts
```
