    23	DEFAULTS = {"finish": "verified-feature-branch", "permission_mode": "bypassPermissions",
    24	            "worker_timeout_seconds": 4 * 3600, "review_timeout_seconds": 1800, "reviewer_transport": "native"}
    25	TIMEOUT_KEYS = ("worker_timeout_seconds", "review_timeout_seconds")
    26	REVIEWER_TRANSPORTS = ("native", "print")
    27	# Plans pinned before the native reviewer existed lack reviewer_transport; they mean native.
    28	REQUIRED_KEYS = frozenset(DEFAULTS) - {"reviewer_transport"}

    31	def automatic_settings(worker_timeout_seconds: int | None = None, review_timeout_seconds: int | None = None,
    32	                       reviewer_transport: str | None = None) -> dict:
    33	    """Run-scoped automatic configuration; deadlines and transport are pinned into plan.json at prepare."""
    34	    settings = dict(DEFAULTS)
    35	    for key, value in (("worker_timeout_seconds", worker_timeout_seconds), ("review_timeout_seconds", review_timeout_seconds),
    36	                       ("reviewer_transport", reviewer_transport)):
    37	        if value is not None:
    38	            settings[key] = value
    39	    validate_automatic({"automatic": settings, "source_branch": "feature/validation-only"})
    40	    return settings
    41

    43	def validate_automatic(plan: dict) -> None:
    44	    settings = plan.get("automatic")
    45	    if not isinstance(settings, dict) or not REQUIRED_KEYS <= set(settings) <= set(DEFAULTS):
    46	        raise ValueError("Malformed automatic run configuration")
    47	    if settings["finish"] != DEFAULTS["finish"] or settings["permission_mode"] != "bypassPermissions":
    48	        raise ValueError("Unsupported automatic authority")
    49	    for key in TIMEOUT_KEYS:
    50	        if type(settings[key]) is not int or not 1 <= settings[key] <= 86400:
    51	            raise ValueError("Automatic timeouts must be bounded positive seconds (at most 86400)")
    52	    if "reviewer_transport" in settings and settings["reviewer_transport"] not in REVIEWER_TRANSPORTS:
    53	        raise ValueError("Unsupported reviewer transport; expected native or print")
    54	    if not plan.get("source_branch", "").startswith("feature/"):
    55	        raise ValueError("Automatic completion is restricted to a feature/ branch")

    58	def reviewer_transport(plan: dict) -> str:
    59	    """Effective transport: plans that predate the setting are native, though they never launch a new reviewer."""
    60	    return plan["automatic"].get("reviewer_transport", "native")

    63	def completion_prompt(directory: Path, plan: dict, node: str) -> str:
    64	    example = {"version": "1.0.0", "run_id": plan["run_id"], "node_id": node,
    65	               "launch_token": plan["nodes"][node]["session_id"], "status": "completed",
    66	               "summary": "Describe actual work and checks executed", "open_assumptions": []}
    67	    return ("\n\nAUTOMATIC MODE: permission checks are bypassed and Bash is available. "
    68	            "Do not wait for a human handoff. Stay within assigned ownership; do not commit, merge, push, "
    69	            "launch agents, change runtime evidence or switch billing/provider. "
    70	            "On completion write the following JSON shape atomically (temporary file then rename) to "
    71	            f"{directory / (node + '.completion.json')}. This one output file is allowed outside your worktree. "
    72	            "Use status blocked if you cannot finish; never manufacture checks. Write it as your last action, "
    73	            "then finish your turn and do not modify more files. Controller checks and independent review "
    74	            "still determine acceptance.\n" + json.dumps(example))

    77	def read_completion(runtime, node: str) -> dict:
    78	    path = runtime.directory / f"{node}.completion.json"
    79	    if path.is_symlink() or not path.is_file() or path.stat().st_size > 65536:
    80	        raise ValueError(f"Invalid completion file for {node}")
    81	    item = read_json(path)
    82	    expected = {"version", "run_id", "node_id", "launch_token", "status", "summary", "open_assumptions"}
    83	    if not isinstance(item, dict) or set(item) != expected:
    84	        raise ValueError("Malformed completion signal")
    85	    if (item["version"] != "1.0.0" or item["run_id"] != runtime.plan["run_id"] or item["node_id"] != node
    86	            or item["launch_token"] != runtime.plan["nodes"][node]["session_id"]):
    87	        raise ValueError("Stale or foreign worker completion signal")
    88	    if item["status"] not in {"completed", "blocked"} or not isinstance(item["summary"], str) or not item["summary"].strip():
    89	        raise ValueError("Invalid completion status/summary")
    90	    if not isinstance(item["open_assumptions"], list) or any(not isinstance(x, str) for x in item["open_assumptions"]):
    91	        raise ValueError("Invalid completion assumptions")
    92	    if item["status"] == "blocked":
    93	        raise RuntimeError(f"Worker {node} explicitly blocked: {item['summary']}")
    94	    return {"summary": item["summary"], "open_assumptions": item["open_assumptions"]}

    97	def wait_handoffs(runtime, *, clock=time.time, sleep=time.sleep) -> None:
    98	    """Idle alone never means completion. Deadlines survive controller restart."""
    99	    validate_automatic(runtime.plan)
   100	    if any((runtime.directory / f"{node}.stop.json").exists() for node in NODES):
   101	        # Recover a controller crash after durable handoffs/stop intent, before snapshot.
   102	        for node in NODES:
   103	            if read_completion(runtime, node) != read_json(runtime.directory / f"{node}.handoff.json"):
   104	                raise RuntimeError("Handoff changed after stop intent")
   105	        return
   106	    while True:
   107	        rows = runtime.sessions.inventory()
   108	        handoffs = {}
   109	        for node in NODES:
   110	            receipt = read_json(runtime.directory / f"{node}.interactive.json")
   111	            started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   112	            if clock() >= started + runtime.plan["automatic"]["worker_timeout_seconds"]:
   113	                raise RuntimeError(f"Worker {node} deadline exhausted; no automatic relaunch")
   114	            row = runtime.sessions.locate(node, rows)
   115	            if row is None:
   116	                raise RuntimeError("Native worker missing; reconciliation required")
   117	            if row["state"] == "blocked":
   118	                raise RuntimeError(f"Worker {node} blocked; inspect quota or native error. No billing/provider fallback.")
   119	            path = runtime.directory / f"{node}.completion.json"
   120	            if row["state"] in {"idle", "done"} and path.exists():
   121	                handoffs[node] = read_completion(runtime, node)
   122	        if set(handoffs) == set(NODES):
   123	            for node, value in handoffs.items():
   124	                save_json(runtime.directory / f"{node}.handoff.json", value)
   125	            return
   126	        sleep(2)

   129	REVIEW_SCHEMA = {
   130	    "type": "object", "additionalProperties": False, "required": ["verdict", "findings"],
   131	    "properties": {
   132	        "verdict": {"enum": ["approved", "blocked"]},
   133	        "findings": {"type": "array", "items": {
   134	            "type": "object", "additionalProperties": False,
   135	            "required": ["severity", "message", "disposition", "worker", "requirement"],
   136	            "properties": {"severity": {"enum": ["P0", "P1", "P2"]}, "message": {"type": "string", "minLength": 1},
   137	                           "disposition": {"enum": ["open", "resolved", "accepted"]},
   138	                           "worker": {"enum": ["ui", "adapter", "both", "none"]},
   139	                           "requirement": {"type": ["string", "null"], "minLength": 1}}}}}

   167	    return ("\n\nREVIEW COMPLETION PROTOCOL: you run as a native session. A human may type in this terminal; the transcript "
   168	            f"is the record, but your verdict is only the file {completion}. Write exactly this JSON shape there "
   169	            "(schema: contracts/workflow/reviewCompletion.schema.json in this checkout when present):\n" + json.dumps(example) + "\n"
   170	            "Keep version, run_id, node_id, launch_token, bundle_sha256 and candidate_commit exactly as shown; the controller "
   171	            "rejects any other binding without launching another reviewer. verdict is approved or blocked. Each finding "

   181	def read_review_completion(runtime) -> dict:
   182	    """Accept the reviewer's file only when it validates and is bound to this exact run, launch, bundle and candidate."""
   183	    from jsonschema.exceptions import ValidationError
   184	    from .verification import validate_schema
   185	    path = runtime.directory / "review.completion.json"
   186	    if path.is_symlink() or not path.is_file() or path.stat().st_size > REVIEW_COMPLETION_LIMIT:
   187	        raise RuntimeError("Invalid review completion file")
   188	    try:
   189	        item = read_json(path)
   190	    except ValueError as error:
   191	        raise RuntimeError(f"Malformed review completion signal: {error}") from None
   192	    try:
   193	        validate_schema("reviewCompletion", item)
   194	    except ValidationError as error:
   195	        raise RuntimeError(f"Review completion signal violates the schema: {error.message}") from None
   196	    receipt = read_json(runtime.directory / "automatic-review.json")
   197	    bundle, digest = runtime.validate_bundle()
   198	    if (item["run_id"] != runtime.plan["run_id"] or item["node_id"] != "review" or item["launch_token"] != receipt.get("launch_token")
   199	            or item["bundle_sha256"] != digest or item["candidate_commit"] != bundle["candidate_commit"]):
   200	        raise RuntimeError("Stale or foreign review completion signal")
   201	    return {"verdict": item["verdict"], "findings": item["findings"]}

   204	def wait_review(runtime, *, clock=None, sleep=None) -> dict:
   205	    """Idle alone never means a verdict. The deadline counts from the reviewer's launch and survives restarts."""
   206	    clock = clock or time.time
   207	    sleep = sleep or time.sleep
   208	    validate_automatic(runtime.plan)
   209	    receipt = read_json(runtime.directory / "review.interactive.json")
   210	    started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   211	    path = runtime.directory / "review.completion.json"
   212	    attention = False
   213	    while True:
   214	        if clock() >= started + runtime.plan["automatic"]["review_timeout_seconds"]:
   215	            raise RuntimeError("Reviewer deadline exhausted; no second reviewer is launched")
   216	        row = runtime.sessions.locate("review", runtime.sessions.inventory())
   217	        if row is None:
   218	            raise RuntimeError("Native reviewer missing; reconciliation required")
   219	        if row["state"] == "blocked" and not attention:
   220	            # A native session reports `blocked` when it needs a human: a question or a prompt
   221	            # it cannot answer itself. The operator may answer in the pane; the deadline bounds it.
   222	            attention = True
   223	            runtime.event("review", "interactive", "Reviewer needs attention in its pane (native state blocked); waiting until the deadline")
   224	        if row["state"] in {"idle", "done"} and path.exists():
   225	            return read_review_completion(runtime)
   226	        sleep(2)

   229	def review_candidate(runtime) -> dict:
   230	    """Called only by the LangGraph review node. Ambiguous invocations never replay."""
   231	    validate_automatic(runtime.plan)
   232	    bundle, digest = runtime.validate_bundle()
   233	    receipt_path = runtime.directory / "automatic-review.json"
   234	    if receipt_path.exists():
   235	        receipt = read_json(receipt_path)
   236	        if receipt.get("bundle_sha256") == digest and receipt.get("status") == "succeeded":
   237	            runtime.validate_review(receipt["review"])
   238	            return receipt["review"]
   239	        if receipt.get("bundle_sha256") == digest and receipt.get("status") == "running" and receipt.get("transport") == "native":
   240	            # The reviewer kept running while the controller was away; keep waiting for its file.
   241	            return _accept_native(runtime, bundle, digest, receipt_path, receipt)
   242	        raise RuntimeError("Prior reviewer invocation needs reconciliation; no automatic relaunch")
   243	    cwd = runtime.directory / "review-worktree"
   244	    if cwd.exists():
   245	        raise RuntimeError("Partial review worktree exists; reconcile rather than overwrite")
   246	    subprocess.run(["git", "-C", runtime.plan["repository"], "worktree", "add", "--detach", str(cwd), bundle["candidate_commit"]],
   247	                   check=True, capture_output=True)
   248	    patch = runtime.directory / "review.diff"
   249	    with patch.open("w") as handle:
   250	        subprocess.run(["git", "-C", str(cwd), "diff", "--binary", runtime.plan["base_commit"], bundle["candidate_commit"]], stdout=handle, check=True)
   251	    if reviewer_transport(runtime.plan) == "print":
   252	        return _review_print(runtime, bundle, digest, cwd, patch, receipt_path)
   253	    return _review_native(runtime, bundle, digest, patch, receipt_path)

   256	def _decide(runtime, bundle: dict, digest: str, reviewer: str, decision: dict, receipt: dict) -> dict:
   257	    """Persist review.json for any verdict; only an approved review without blocking findings passes."""
   258	    from .pipeline import blocking_findings
   259	    review = {"run_id": bundle["run_id"], "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   260	              "reviewer": reviewer, "independent": True, "verdict": decision["verdict"], "findings": decision["findings"]}
   261	    blocked = decision["verdict"] != "approved" or bool(blocking_findings(decision["findings"]))
   262	    if blocked:
   263	        review["verdict"] = "blocked"  # An approval with an unresolved P0/P1 is contradictory; the run is blocked.
   264	    save_json(runtime.directory / "review.json", review)
   265	    receipt["decision"] = decision
   266	    if blocked:
   267	        raise RuntimeError("Independent reviewer blocked the candidate")
   268	    runtime.validate_review(review)
   269	    receipt.update(status="succeeded", review=review, accepted_at=now())
   270	    return review

   273	def _review_native(runtime, bundle: dict, digest: str, patch: Path, receipt_path: Path) -> dict:
   274	    from .pipeline import digest_file
   275	    launch_token = str(uuid.uuid4())
   276	    receipt = {"transport": "native", "launch_token": launch_token, "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   277	               "patch_sha256": digest_file(patch), "status": "launching"}
   278	    save_json(receipt_path, receipt)
   279	    prompt = review_prompt(runtime, patch) + completion_protocol_prompt(runtime, launch_token, digest, bundle["candidate_commit"])
   280	    try:
   281	        launched = runtime.launch_reviewer(prompt, launch_token, bundle["candidate_commit"])
   282	    except BaseException as error:
   283	        # The launch may or may not have produced a session; only an operator can tell. Never launch again.
   284	        receipt.update(status="needs_reconciliation", error=str(error))
   285	        save_json(receipt_path, receipt)
   286	        raise
   287	    receipt.update(session_id=launched["session_id"], background_id=launched.get("background_id"), status="running")
   288	    save_json(receipt_path, receipt)
   289	    return _accept_native(runtime, bundle, digest, receipt_path, receipt)

   292	def _accept_native(runtime, bundle: dict, digest: str, receipt_path: Path, receipt: dict) -> dict:
   293	    from .pipeline import digest_file
   294	    cwd = runtime.directory / "review-worktree"
   295	    patch = runtime.directory / "review.diff"
   296	    try:
   297	        decision = wait_review(runtime)
   298	        row = runtime.sessions.locate("review", runtime.sessions.inventory())
   299	        worker_ids = {item["session_id"] for item in bundle["snapshots"].values()}
   300	        if row is None or row.get("sessionId") != receipt.get("session_id") or receipt.get("session_id") in worker_ids:
   301	            raise RuntimeError("Reviewer identity changed or is not independent; refusing the verdict")
   302	        if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   303	            raise RuntimeError("Reviewer worktree changed")
   304	        if runtime.validate_bundle()[1] != digest or digest_file(patch) != receipt["patch_sha256"]:
   305	            raise RuntimeError("Evidence changed during review")
   306	        review = _decide(runtime, bundle, digest, receipt["session_id"], decision, receipt)
   307	        save_json(receipt_path, receipt)
   308	    except KeyboardInterrupt:
   309	        # Operator/terminal interruption is not a reviewer failure: the native session keeps
   310	        # running and `automatic --live` resumes waiting for its completion file.
   311	        runtime.event("review", "interrupted", REVIEW_RESUME_NOTE.format(directory=runtime.directory))
   312	        raise
   313	    except BaseException as error:
   314	        # Deadline, blocked session, rejected file or blocked verdict: stop the one reviewer
   315	        # so it consumes no usage for a run that cannot continue. Nothing is relaunched.
   316	        receipt.update(status="blocked", error=str(error))
   317	        save_json(receipt_path, receipt)
   318	        try:
   319	            runtime.stop_reviewer()
   320	        except Exception as cleanup_error:
   321	            runtime.event("review", "blocked", f"Could not confirm reviewer stop: {cleanup_error}")
   322	        raise
   323	    runtime.stop_reviewer()
   324	    return review

   327	def _review_print(runtime, bundle: dict, digest: str, cwd: Path, patch: Path, receipt_path: Path) -> dict:
   328	    """Headless fallback (--reviewer-transport print): one `claude --print` job, no pane, no human input."""
   329	    from jsonschema import validate
   330	    from .pipeline import digest_file
   331	    session_id = str(uuid.uuid4())
   332	    receipt = {"transport": "print", "session_id": session_id, "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   333	               "status": "launching", "patch_sha256": digest_file(patch)}
   334	    save_json(receipt_path, receipt)
   335	    prompt = review_prompt(runtime, patch) + " Return the requested JSON schema."
   336	    command = [runtime.sessions.executable, "--print", "--output-format", "json", "--session-id", session_id,
   337	               "--safe-mode", "--strict-mcp-config", "--mcp-config", '{"mcpServers":{}}',
   338	               "--tools", "Read,Glob,Grep", "--permission-mode", "dontAsk", "--permission-prompts", "none",
   339	               "--add-dir", str(runtime.directory), "--json-schema", json.dumps(REVIEW_SCHEMA)]
   340	    env = {key: value for key, value in os.environ.items() if not key.startswith("HERDR_")}
   341	    process = None
   342	    try:
   343	        with (runtime.directory / "review.stdout.json").open("w") as output, (runtime.directory / "review.stderr.log").open("w") as errors:
   344	            process = subprocess.Popen(command, cwd=cwd, env=env, stdin=subprocess.PIPE, stdout=output, stderr=errors,
   345	                                       text=True, start_new_session=True)
   346	            receipt.update(status="running", pid=process.pid)
   347	            save_json(receipt_path, receipt)
   348	            process.communicate(prompt, timeout=runtime.plan["automatic"]["review_timeout_seconds"])
   349	        result = read_json(runtime.directory / "review.stdout.json")
   350	        if process.returncode != 0 or result.get("session_id") != session_id or result.get("is_error") is not False or result.get("subtype") != "success":
   351	            raise RuntimeError("Reviewer did not succeed; inspect retained output. No automatic retry/provider switch.")
   352	        decision = result.get("structured_output")
   353	        validate(decision, REVIEW_SCHEMA)
   354	        if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   355	            raise RuntimeError("Reviewer worktree changed")
   356	        if runtime.validate_bundle()[1] != digest or digest_file(patch) != receipt["patch_sha256"]:
   357	            raise RuntimeError("Evidence changed during review")
   358	        return _decide(runtime, bundle, digest, session_id, decision, receipt)
   359	    except BaseException as error:
   360	        if process is not None and process.poll() is None:
   361	            terminate(process)
   362	        receipt.update(status="blocked", error=str(error))
   363	        raise
   364	    finally:
   365	        save_json(receipt_path, receipt)

   368	def advance_failed_checks(runtime, state) -> bool:
   369	    """Retry only recorded failing verification packets, never launches or review."""
   370	    # A checkpoint can carry an error from an earlier attempt of a task that has since
   371	    # succeeded (its writes are applied and it is no longer pending). Only pending
   372	    # tasks with errors are failures to classify.
   373	    failures = [task.name for task in state.tasks if task.error and task.name in state.next]
   374	    if not failures or any(name not in {"verify_ui", "verify_adapter", "candidate"} for name in failures):
   375	        return False
   376	    targets = []
   377	    for name in failures:
   378	        phase = "candidate" if name == "candidate" else "worker"
   379	        stage_targets = []
   380	        for node in NODES if phase == "candidate" else (name.removeprefix("verify_"),):
   381	            attempt = runtime.attempt(phase, node)
   382	            path = runtime.directory / "verification" / phase / node / str(attempt) / "packet.json"
   383	            if path.exists() and read_json(path)["gate"]["status"] != "passed":
   384	                previous = runtime.directory / "verification" / phase / node / str(attempt - 1) / "packet.json"
   385	                if attempt > 1 and previous.exists() and read_json(previous)["gate"]["reasons"] == read_json(path)["gate"]["reasons"]:
   386	                    # Retries rerun immutable code; two identical failures mean the cause is
   387	                    # deterministic (code or environment), and more attempts only burn time.
   388	                    raise RuntimeError(f"{phase}/{node} failed identically on attempts {attempt - 1} and {attempt}; "
   389	                                       f"not transient, inspect {path}")
   390	                stage_targets.append((phase, node))
   391	        if not stage_targets:
   392	            return False
   393	        targets.extend(stage_targets)
   394	    if not targets:
   395	        return False
   396	    # Check all bounds before changing any counters.
   397	    if any(runtime.attempt(p, n) >= runtime.policy.get("max_verification_attempts", 3) for p, n in targets):
   398	        raise RuntimeError("Verification retry limit exhausted; work and evidence retained")
   399	    for phase, node in targets:
   400	        runtime.retry_check(phase, node)
   401	    return True

   404	def supervise(directory: Path) -> None:
   405	    """Each recovery uses a new controller process, not just an in-memory replay."""
   406	    validate_automatic(read_json(directory / "plan.json"))
   407	    with run_lock(directory, "automatic-supervisor.lock"):
   408	        for _ in range(45):
   409	            try:
   410	                result = subprocess.run([sys.executable, "-m", "workflow", "automatic-step", str(directory), "--live"],
   411	                                        cwd=Path(__file__).resolve().parents[1])
   412	            except KeyboardInterrupt:
   413	                raise RuntimeError(RESUME_NOTE.format(directory=directory)) from None
   414	            if result.returncode == 0:
   415	                return
   416	            if result.returncode != 75:
   417	                raise RuntimeError(f"Automatic controller blocked (exit {result.returncode}); inspect retained run")
   418	        raise RuntimeError("Automatic controller restart limit exhausted")
   419

   421	RESUME_NOTE = ("Supervisor interrupted. Native workers were NOT stopped and keep running; "
   422	               "resume with: python -m workflow automatic {directory} --live")

   425	def drive(runtime, *, single_step=False) -> str | None:
   426	    """Advance persisted graph state; CLI supervision restarts this process at joins."""
   427	    from .pipeline import build_pipeline, report
   428	    validate_automatic(runtime.plan)
   429	    if git(Path(runtime.plan["repository"]), "symbolic-ref", "--short", "HEAD") != runtime.plan["source_branch"]:
   430	        raise RuntimeError("Source feature branch changed; no automatic continuation")
   431	    runtime.event("controller", "running", f"Automatic checkpoint controller PID {os.getpid()}")
   432	    config = {"configurable": {"thread_id": runtime.plan["run_id"]}, "max_concurrency": 2}
   433	    while True:
   434	        with SqliteSaver.from_conn_string(str(runtime.directory / "pipeline.sqlite")) as saver:
   435	            graph = build_pipeline(saver, runtime)
   436	            state = graph.get_state(config)
   437	            if not state.values or any(name.startswith("launch_") for name in state.next):
   438	                raise RuntimeError("Automatic supervision requires a completed start; reconcile uncertain launches explicitly")
   439	            if not state.next:
   440	                commit = state.values.get("integrated_commit")
   441	                if (not commit or git(Path(runtime.plan["repository"]), "rev-parse", "HEAD") != commit
   442	                        or git(Path(runtime.plan["repository"]), "status", "--porcelain")):
   443	                    raise RuntimeError("No verified feature-branch completion")
   444	                runtime.validate_review(read_json(runtime.directory / "review.json"))
   445	                report(runtime, state)
   446	                return commit
   447	            pending = [item.value.get("kind") for task in state.tasks for item in task.interrupts]
   448	            value = None
   449	            if pending == ["worker_handoff"]:
   450	                try:
   451	                    wait_handoffs(runtime)
   452	                except KeyboardInterrupt:
   453	                    # Operator/terminal interruption is not a worker failure: leave the
   454	                    # native sessions running so `automatic --live` can resume polling.
   455	                    runtime.event("controller", "interrupted", RESUME_NOTE.format(directory=runtime.directory))
   456	                    report(runtime, state)
   457	                    raise
   458	                except BaseException as error:
   459	                    # Deadline, quota block, missing/blocked completion: stop the workers so
   460	                    # no session keeps consuming usage for a run that cannot continue.
   461	                    runtime.event("controller", "blocked", str(error))
   462	                    try:
   463	                        runtime.stop_workers()
   464	                    except Exception as cleanup_error:
   465	                        runtime.event("freeze", "blocked", f"Could not confirm worker stop: {cleanup_error}")
   466	                    report(runtime, state)
   467	                    raise
   468	                value = Command(resume={"freeze": True})
   469	            elif pending:
   470	                raise RuntimeError("Unexpected manual gate in automatic run; inspect state")
   471	            elif any(task.error for task in state.tasks) and not advance_failed_checks(runtime, state):
   472	                raise RuntimeError("Non-retryable graph failure; inspect retained evidence")
   473	            try:
   474	                graph.invoke(value, config)
   475	            except Exception:
   476	                failed = graph.get_state(config)
   477	                if not any(task.error for task in failed.tasks):
   478	                    raise
   479	                # Next loop reopens the checkpointer and classifies the exact failure.
   480	            finally:
   481	                report(runtime, graph.get_state(config))
   482	            if single_step:
   483	                return None  # Exit 75: supervisor reopens this checkpoint in a fresh process.
