    86	/** One run directory exactly as workflow/export_state.py and workflow/checks.py persist it. */
    87	async function writeRun(root: string, spec: RunSpec): Promise<string> {
    88	  const dir = join(root, spec.runId)
    89	  await mkdir(dir, { recursive: true })
    90	  const runId = spec.exportRunId ?? spec.runId
    91	  if (spec.diffFile !== undefined) await writeFile(join(dir, 'review.diff'), spec.diffFile)
    92	  const registrations: Registration[] = []
    93	  for (const packetSpec of spec.packets ?? []) {
    94	    const phase = packetSpec.phase ?? 'worker'
    95	    const attempt = packetSpec.attempt ?? 1
    96	    const packetDir = join(dir, 'verification', phase, packetSpec.node, String(attempt))
    97	    const artifactsDir = join(packetDir, 'artifacts')
    98	    await mkdir(artifactsDir, { recursive: true })
    99	    const artifacts = []
   100	    const paths: Record<string, string> = {}
   101	    for (const artifact of packetSpec.artifacts ?? []) {
   102	      const content = Buffer.isBuffer(artifact.content) ? artifact.content : Buffer.from(artifact.content)
   103	      const uri = artifact.uri ?? artifact.id
   104	      if (!artifact.skipWrite) await writeFile(join(artifactsDir, uri), content)
   105	      artifacts.push({ artifact_id: artifact.id, kind: artifact.kind, uri, sha256: artifact.registeredSha ?? sha256(content) })
   106	      paths[artifact.id] = join(artifactsDir, uri)
   107	    }
   108	    const logs = artifacts.filter(item => item.kind === 'log')
   109	    const checks = logs.map((log, index) => ({
   110	      command: `npm run check-${index}`, cwd: join(packetDir, 'worktree'), started_at: T0, finished_at: T1, exit_code: 0, log_artifact_id: log.artifact_id,
   111	    }))
   112	    const expected = { run_id: runId, node_id: packetSpec.node, attempt, base_commit: BASE, output_commit: OUTPUT, verification_cwd: join(packetDir, 'worktree') }
   113	    const packet: Record<string, unknown> = {
   114	      phase,
   115	      expected,
   116	      result: {
   117	        contract_version: '1.0.0', run_id: runId, node_id: packetSpec.node, attempt, session_id: packetSpec.session ?? `${packetSpec.node}-session-0001`,
   118	        status: 'succeeded', base_commit: BASE, output_commit: OUTPUT, changed_files: packetSpec.changed ?? [`src/${packetSpec.node}.ts`],
   119	        checks, open_assumptions: packetSpec.assumptions ?? [], artifacts,
   120	        summary: packetSpec.summary ?? `Trusted ${phase} check capture; not integration approval`, error: null,
   121	      },
   122	      evidence: { version: '1.0.0', policy_sha256: 'c'.repeat(64), run_id: runId, node_id: packetSpec.node, attempt, output_commit: OUTPUT, checks: [] },
   123	      artifact_root: artifactsDir, artifact_paths: paths, capture_errors: [], effective_commands: [],
   124	      gate: { ...(packetSpec.gate ?? { status: 'passed', reasons: [] }), pending_gates: ['independent_review', 'integration_approval'], integration_allowed: false },
   125	    }
   126	    packetSpec.mutate?.(packet)
   127	    const text = json(packet)
   128	    await writeFile(join(packetDir, 'packet.json'), text)
   129	    registrations.push({ phase, node_id: packetSpec.node, attempt, path: `verification/${phase}/${packetSpec.node}/${attempt}/packet.json`, sha256: sha256(text) })
   130	    if (packetSpec.tamper) await writeFile(join(packetDir, 'packet.json'), text + '\n')
   131	  }
   132	  const created = spec.created ?? T0
   133	  if (!spec.skipPlan) {
   134	    await writeFile(join(dir, 'plan.json'), json({
   135	      run_id: spec.planRunId ?? runId, repository: '/synthetic/repository', base_commit: BASE, allow_edits: true, mode: 'interactive',
   136	      nodes: { ui: { worktree: join(dir, 'worktree-ui') }, adapter: { worktree: join(dir, 'worktree-adapter') } },
   137	      policy_sha256: 'c'.repeat(64), created_at: created, source_branch: 'feature/synthetic',
   138	    }))
   139	  }
   140	  const events = spec.events ?? []
   141	  if (spec.eventsFile !== false) {
   142	    const text = typeof spec.eventsFile === 'string' ? spec.eventsFile : events.map(event => JSON.stringify(event) + '\n').join('')
   143	    if (text.length > 0) await writeFile(join(dir, 'events.jsonl'), text)
   144	  }
   145	  const sections = spec.review !== undefined || spec.inputs !== undefined
   146	  const state = {
   147	    version: spec.version ?? (sections ? '1.2.0' : '1.0.0'), run_id: runId, base_commit: BASE, created_at: created, definition: spec.definition ?? DEFINITION,
   148	    values: { run_id: runId, ...(spec.values ?? {}) }, next: spec.next ?? [], tasks: spec.tasks ?? [], events,
   149	    verification_packets: spec.registrations ? spec.registrations(registrations) : registrations, updated_at: spec.updated ?? T1,
   150	    ...(spec.review !== undefined ? { review: spec.review } : {}), ...(spec.inputs !== undefined ? { inputs: spec.inputs } : {}),
   151	  }
   152	  await writeFile(join(dir, 'run-state.json'), spec.stateText ?? json(state))
   153	  return dir
   154	}

   263	async function harness(run: (h: Harness) => Promise<void>, projects: Parameters<typeof registry>[1] = [{ id: 'alpha', name: 'Alpha', workflows: [{ id: 'main' }] }], options: Parameters<typeof createApp>[1] = {}) {
   264	  const root = await mkdtemp(join(tmpdir(), 'md-manager-projects-'))
   265	  try {
   266	    for (const project of projects) for (const workflow of project.workflows) {
   267	      const runsRoot = workflow.runsRoot ?? join(root, 'runs', project.id, workflow.id)
   268	      if (runsRoot.startsWith(root)) await mkdir(runsRoot, { recursive: true })
   269	    }
   270	    const config = await parseProjectsConfig(JSON.stringify(registry(root, projects)), 'test registry')
   271	    const app = createApp([], { ...options, projects: config })
   272	    try {
   273	      await run({ root, app, runsRoot: (project, workflow) => join(root, 'runs', project, workflow) })
   274	    } finally { await app.close() }
   275	  } finally { await rm(root, { recursive: true, force: true }) }

   287	function assertError(response: { status: number; body: string; json: () => unknown }, status: number, code: string, root?: string) {
   288	  assert.equal(response.status, status, response.body)
   289	  const body = response.json() as { error: { code: string; message: string } }
   290	  assert.deepEqual(Object.keys(body), ['error'])
   291	  assert.equal(body.error.code, code)
   292	  assert.ok(typeof body.error.message === 'string' && body.error.message.length > 0)
   293	  if (root) assert.ok(!response.body.includes(root), `absolute paths are never disclosed: ${response.body}`)
   294	  assert.ok(!/\/(?:home|tmp|var)\//.test(body.error.message), `message leaks a path: ${body.error.message}`)
   295	}

   421	async function snapshotTree(dir: string): Promise<[string, number, number][]> {
   422	  const entries: [string, number, number][] = []
   423	  for (const entry of await readdir(dir, { withFileTypes: true, recursive: true })) {
   424	    const path = join(entry.parentPath ?? dir, entry.name)
   425	    const info = await stat(path)
   426	    entries.push([path, info.size, info.mtimeMs])
   427	  }
   428	  return entries.sort()
   429	}
