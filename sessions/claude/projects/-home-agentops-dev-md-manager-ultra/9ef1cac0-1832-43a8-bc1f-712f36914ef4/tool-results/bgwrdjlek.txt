    78	/** One run directory exactly as workflow/export_state.py and workflow/checks.py persist it. */
    79	async function writeRun(root: string, spec: RunSpec): Promise<string> {
    80	  const dir = join(root, spec.runId)
    81	  await mkdir(dir, { recursive: true })
    82	  const runId = spec.exportRunId ?? spec.runId
    83	  const registrations: Registration[] = []
    84	  for (const packetSpec of spec.packets ?? []) {
    85	    const phase = packetSpec.phase ?? 'worker'
    86	    const attempt = packetSpec.attempt ?? 1
    87	    const packetDir = join(dir, 'verification', phase, packetSpec.node, String(attempt))
    88	    const artifactsDir = join(packetDir, 'artifacts')
    89	    await mkdir(artifactsDir, { recursive: true })
    90	    const artifacts = []
    91	    const paths: Record<string, string> = {}
    92	    for (const artifact of packetSpec.artifacts ?? []) {
    93	      const content = Buffer.isBuffer(artifact.content) ? artifact.content : Buffer.from(artifact.content)
    94	      const uri = artifact.uri ?? artifact.id
    95	      if (!artifact.skipWrite) await writeFile(join(artifactsDir, uri), content)
    96	      artifacts.push({ artifact_id: artifact.id, kind: artifact.kind, uri, sha256: artifact.registeredSha ?? sha256(content) })
    97	      paths[artifact.id] = join(artifactsDir, uri)
    98	    }
    99	    const logs = artifacts.filter(item => item.kind === 'log')
   100	    const checks = logs.map((log, index) => ({
   101	      command: `npm run check-${index}`, cwd: join(packetDir, 'worktree'), started_at: T0, finished_at: T1, exit_code: 0, log_artifact_id: log.artifact_id,
   102	    }))
   103	    const expected = { run_id: runId, node_id: packetSpec.node, attempt, base_commit: BASE, output_commit: OUTPUT, verification_cwd: join(packetDir, 'worktree') }
   104	    const packet: Record<string, unknown> = {
   105	      phase,
   106	      expected,
   107	      result: {
   108	        contract_version: '1.0.0', run_id: runId, node_id: packetSpec.node, attempt, session_id: packetSpec.session ?? `${packetSpec.node}-session-0001`,
   109	        status: 'succeeded', base_commit: BASE, output_commit: OUTPUT, changed_files: packetSpec.changed ?? [`src/${packetSpec.node}.ts`],
   110	        checks, open_assumptions: packetSpec.assumptions ?? [], artifacts,
   111	        summary: packetSpec.summary ?? `Trusted ${phase} check capture; not integration approval`, error: null,
   112	      },
   113	      evidence: { version: '1.0.0', policy_sha256: 'c'.repeat(64), run_id: runId, node_id: packetSpec.node, attempt, output_commit: OUTPUT, checks: [] },
   114	      artifact_root: artifactsDir, artifact_paths: paths, capture_errors: [], effective_commands: [],
   115	      gate: { ...(packetSpec.gate ?? { status: 'passed', reasons: [] }), pending_gates: ['independent_review', 'integration_approval'], integration_allowed: false },
   116	    }
   117	    packetSpec.mutate?.(packet)
   118	    const text = json(packet)
   119	    await writeFile(join(packetDir, 'packet.json'), text)
   120	    registrations.push({ phase, node_id: packetSpec.node, attempt, path: `verification/${phase}/${packetSpec.node}/${attempt}/packet.json`, sha256: sha256(text) })
   121	    if (packetSpec.tamper) await writeFile(join(packetDir, 'packet.json'), text + '\n')
   122	  }
   123	  const created = spec.created ?? T0
   124	  if (!spec.skipPlan) {
   125	    await writeFile(join(dir, 'plan.json'), json({
   126	      run_id: spec.planRunId ?? runId, repository: '/synthetic/repository', base_commit: BASE, allow_edits: true, mode: 'interactive',
   127	      nodes: { ui: { worktree: join(dir, 'worktree-ui') }, adapter: { worktree: join(dir, 'worktree-adapter') } },
   128	      policy_sha256: 'c'.repeat(64), created_at: created, source_branch: 'feature/synthetic',
   129	    }))
   130	  }
   131	  const events = spec.events ?? []
   132	  if (spec.eventsFile !== false) {
   133	    const text = typeof spec.eventsFile === 'string' ? spec.eventsFile : events.map(event => JSON.stringify(event) + '\n').join('')
   134	    if (text.length > 0) await writeFile(join(dir, 'events.jsonl'), text)
   135	  }
   136	  const state = {
   137	    version: '1.0.0', run_id: runId, base_commit: BASE, created_at: created, definition: spec.definition ?? DEFINITION,
   138	    values: { run_id: runId, ...(spec.values ?? {}) }, next: spec.next ?? [], tasks: spec.tasks ?? [], events,
   139	    verification_packets: spec.registrations ? spec.registrations(registrations) : registrations, updated_at: spec.updated ?? T1,
   140	  }
   141	  await writeFile(join(dir, 'run-state.json'), spec.stateText ?? json(state))
   142	  return dir
   143	}

   161	function registry(root: string, projects: { id: string; name?: string; workflows: { id: string; definition?: unknown; runsRoot?: string }[] }[]) {
   162	  return {
   163	    version: 1,
   164	    projects: projects.map(project => ({
   165	      project_id: project.id, name: project.name ?? project.id, repository: join(root, 'repo', project.id),
   166	      workflows: project.workflows.map(workflow => ({ workflow_id: workflow.id, runs_root: workflow.runsRoot ?? join(root, 'runs', project.id, workflow.id), definition: workflow.definition ?? DEFINITION })),
   167	    })),
   168	  }

   171	async function harness(run: (h: Harness) => Promise<void>, projects: Parameters<typeof registry>[1] = [{ id: 'alpha', name: 'Alpha', workflows: [{ id: 'main' }] }], options: Parameters<typeof createApp>[1] = {}) {
   172	  const root = await mkdtemp(join(tmpdir(), 'md-manager-projects-'))
   173	  try {
   174	    for (const project of projects) for (const workflow of project.workflows) {
   175	      const runsRoot = workflow.runsRoot ?? join(root, 'runs', project.id, workflow.id)
   176	      if (runsRoot.startsWith(root)) await mkdir(runsRoot, { recursive: true })
   177	    }
   178	    const config = await parseProjectsConfig(JSON.stringify(registry(root, projects)), 'test registry')
   179	    const app = createApp([], { ...options, projects: config })
   180	    try {
   181	      await run({ root, app, runsRoot: (project, workflow) => join(root, 'runs', project, workflow) })
   182	    } finally { await app.close() }
   183	  } finally { await rm(root, { recursive: true, force: true }) }
   184	}

   190	async function get(app: Harness['app'], path: string) {
   191	  const response = await app.inject({ method: 'GET', url: path })

   193	}

   195	function assertError(response: { status: number; body: string; json: () => unknown }, status: number, code: string, root?: string) {
   196	  assert.equal(response.status, status, response.body)
   197	  const body = response.json() as { error: { code: string; message: string } }
   198	  assert.deepEqual(Object.keys(body), ['error'])
   199	  assert.equal(body.error.code, code)
   200	  assert.ok(typeof body.error.message === 'string' && body.error.message.length > 0)
   201	  if (root) assert.ok(!response.body.includes(root), `absolute paths are never disclosed: ${response.body}`)
   202	  assert.ok(!/\/(?:home|tmp|var)\//.test(body.error.message), `message leaks a path: ${body.error.message}`)
   203	}

   313	        const response = await app.inject({ method, url: route, payload: method === 'OPTIONS' ? undefined : { approve: true } })
   314	        assert.equal(response.statusCode, 405, `${method} ${route}: ${response.body}`)
   315	        assert.equal(response.headers.allow, 'GET, HEAD')
   316	        assert.equal(response.json().error.code, 'METHOD_NOT_ALLOWED')
   317	      }
   318	      const head = await app.inject({ method: 'HEAD', url: route })
   319	      assert.equal(head.statusCode, 200, `HEAD ${route}`)
   320	      assert.equal((await get(app, route)).status, 200, route)
   321	    }
   322	    assert.deepEqual(await snapshotTree(runsRoot('alpha', 'main')), before, 'reads never create, modify or delete run storage')
   323	  })
   324	})

   326	async function snapshotTree(dir: string): Promise<[string, number, number][]> {
   327	  const entries: [string, number, number][] = []
   328	  for (const entry of await readdir(dir, { withFileTypes: true, recursive: true })) {
   329	    const path = join(entry.parentPath ?? dir, entry.name)
   330	    const info = await stat(path)
   331	    entries.push([path, info.size, info.mtimeMs])
   332	  }
   333	  return entries.sort()
   334	}

   661	test('runs paginate newest-first with opaque workflow-bound cursors, default 50 and maximum 100', async () => {
   662	  await harness(async ({ app, runsRoot }) => {
   663	    const rootDir = runsRoot('alpha', 'main')
   664	    const ids: string[] = []
   665	    for (let index = 0; index < 53; index += 1) {
   666	      const runId = `run-${String(index).padStart(3, '0')}`
   667	      ids.push(runId)
   668	      // Pairs share an updated_at so the tie-break by run_id is exercised across page boundaries.
   669	      const minute = String(Math.floor(index / 2)).padStart(2, '0')
   670	      await writeRun(rootDir, { runId, values: { ui: receipt('ui') }, next: ['launch_adapter'], updated: `2026-03-02T11:${minute}:00Z` })
   671	    }
   672	    const expected = [...ids].sort((a, b) => {
   673	      const ma = Math.floor(Number(a.slice(4)) / 2), mb = Math.floor(Number(b.slice(4)) / 2)
   674	      return mb - ma || (a < b ? -1 : 1)
   675	    })
   676	    const first = projectSchemas.runList.parse((await get(app, url('alpha', 'main'))).json())
   677	    assert.equal(first.runs.length, 50)
   678	    assert.ok(first.next_cursor)
   679	    const second = projectSchemas.runList.parse((await get(app, url('alpha', 'main', undefined, `?cursor=${encodeURIComponent(first.next_cursor!)}`))).json())
   680	    assert.equal(second.runs.length, 3)
   681	    assert.equal(second.next_cursor, null)
   682	    assert.deepEqual([...first.runs, ...second.runs].map(run => run.run_id), expected)
   683	    const all = projectSchemas.runList.parse((await get(app, url('alpha', 'main', undefined, '?limit=100'))).json())
   684	    assert.equal(all.runs.length, 53)
   685	    assert.equal(all.next_cursor, null)
   686	    // Walk in pages of 7 without overlap or omission; cursors are never paths and cannot cross workflows.
   687	    const walked: string[] = []
   688	    let cursor: string | null = null
   689	    let pages = 0
   690	    do {
   691	      const page: { runs: { run_id: string }[]; next_cursor: string | null } = projectSchemas.runList.parse((await get(app, url('alpha', 'main', undefined, `?limit=7${cursor ? `&cursor=${encodeURIComponent(cursor)}` : ''}`))).json())
   692	      walked.push(...page.runs.map(run => run.run_id))
   693	      assert.ok(cursor === null || !Buffer.from(cursor, 'base64url').toString().includes(rootDir))
   694	      cursor = page.next_cursor
   695	      pages += 1
   696	    } while (cursor)
   697	    assert.equal(pages, 8)
   698	    assert.deepEqual(walked, expected)
   699	    assertError(await get(app, url('alpha', 'other', undefined, `?cursor=${encodeURIComponent(first.next_cursor!)}`)), 400, 'INVALID_CURSOR')
   700	    assertError(await get(app, url('alpha', 'main', undefined, '?cursor=')), 400, 'INVALID_CURSOR')
   701	  }, [{ id: 'alpha', workflows: [{ id: 'main' }, { id: 'other' }] }])
   702	})
