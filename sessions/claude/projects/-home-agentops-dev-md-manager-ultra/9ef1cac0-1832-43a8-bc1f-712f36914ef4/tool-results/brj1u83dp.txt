    23	DEFAULTS = {"finish": "verified-feature-branch", "permission_mode": "bypassPermissions",
    24	            "worker_timeout_seconds": 4 * 3600, "review_timeout_seconds": 1800, "reviewer_transport": "native"}
    25	TIMEOUT_KEYS = ("worker_timeout_seconds", "review_timeout_seconds")
    26	REVIEWER_TRANSPORTS = ("native", "print")
    27	# Plans pinned before the native reviewer existed lack reviewer_transport; they mean native.
    28	REQUIRED_KEYS = frozenset(DEFAULTS) - {"reviewer_transport"}

    31	def automatic_settings(worker_timeout_seconds: int | None = None, review_timeout_seconds: int | None = None,
    32	                       reviewer_transport: str | None = None) -> dict:
    33	    """Run-scoped automatic configuration; deadlines and transport are pinned into plan.json at prepare."""
    34	    settings = dict(DEFAULTS)
    35	    for key, value in (("worker_timeout_seconds", worker_timeout_seconds), ("review_timeout_seconds", review_timeout_seconds),
    36	                       ("reviewer_transport", reviewer_transport)):
    37	        if value is not None:
    38	            settings[key] = value
    39	    validate_automatic({"automatic": settings, "source_branch": "feature/validation-only"})
    40	    return settings
    41

    43	def validate_automatic(plan: dict) -> None:
    44	    settings = plan.get("automatic")
    45	    if not isinstance(settings, dict) or not REQUIRED_KEYS <= set(settings) <= set(DEFAULTS):
    46	        raise ValueError("Malformed automatic run configuration")
    47	    if settings["finish"] != DEFAULTS["finish"] or settings["permission_mode"] != "bypassPermissions":
    48	        raise ValueError("Unsupported automatic authority")
    49	    for key in TIMEOUT_KEYS:
    50	        if type(settings[key]) is not int or not 1 <= settings[key] <= 86400:
    51	            raise ValueError("Automatic timeouts must be bounded positive seconds (at most 86400)")
    52	    if "reviewer_transport" in settings and settings["reviewer_transport"] not in REVIEWER_TRANSPORTS:
    53	        raise ValueError("Unsupported reviewer transport; expected native or print")
    54	    if not plan.get("source_branch", "").startswith("feature/"):
    55	        raise ValueError("Automatic completion is restricted to a feature/ branch")

    58	def reviewer_transport(plan: dict) -> str:
    59	    """Effective transport: plans that predate the setting are native, though they never launch a new reviewer."""
    60	    return plan["automatic"].get("reviewer_transport", "native")

    63	def completion_prompt(directory: Path, plan: dict, node: str) -> str:
    64	    example = {"version": "1.0.0", "run_id": plan["run_id"], "node_id": node,
    65	               "launch_token": plan["nodes"][node]["session_id"], "status": "completed",
    66	               "summary": "Describe actual work and checks executed", "open_assumptions": []}
    67	    return ("\n\nAUTOMATIC MODE: permission checks are bypassed and Bash is available. "
    68	            "Do not wait for a human handoff. Stay within assigned ownership; do not commit, merge, push, "
    69	            "launch agents, change runtime evidence or switch billing/provider. "
    70	            "On completion write the following JSON shape atomically (temporary file then rename) to "
    71	            f"{directory / (node + '.completion.json')}. This one output file is allowed outside your worktree. "
    72	            "Use status blocked if you cannot finish; never manufacture checks. Write it as your last action, "
    73	            "then finish your turn and do not modify more files. Controller checks and independent review "
    74	            "still determine acceptance.\n" + json.dumps(example))

    77	def read_completion(runtime, node: str) -> dict:
    78	    path = runtime.directory / f"{node}.completion.json"
    79	    if path.is_symlink() or not path.is_file() or path.stat().st_size > 65536:
    80	        raise ValueError(f"Invalid completion file for {node}")
    81	    item = read_json(path)
    82	    expected = {"version", "run_id", "node_id", "launch_token", "status", "summary", "open_assumptions"}
    83	    if not isinstance(item, dict) or set(item) != expected:
    84	        raise ValueError("Malformed completion signal")
    85	    if (item["version"] != "1.0.0" or item["run_id"] != runtime.plan["run_id"] or item["node_id"] != node
    86	            or item["launch_token"] != runtime.plan["nodes"][node]["session_id"]):
    87	        raise ValueError("Stale or foreign worker completion signal")
    88	    if item["status"] not in {"completed", "blocked"} or not isinstance(item["summary"], str) or not item["summary"].strip():
    89	        raise ValueError("Invalid completion status/summary")
    90	    if not isinstance(item["open_assumptions"], list) or any(not isinstance(x, str) for x in item["open_assumptions"]):
    91	        raise ValueError("Invalid completion assumptions")
    92	    if item["status"] == "blocked":
    93	        raise RuntimeError(f"Worker {node} explicitly blocked: {item['summary']}")
    94	    return {"summary": item["summary"], "open_assumptions": item["open_assumptions"]}

    97	def wait_handoffs(runtime, *, clock=time.time, sleep=time.sleep) -> None:
    98	    """Idle alone never means completion. Deadlines survive controller restart."""
    99	    validate_automatic(runtime.plan)
   100	    if any((runtime.directory / f"{node}.stop.json").exists() for node in NODES):
   101	        # Recover a controller crash after durable handoffs/stop intent, before snapshot.
   102	        for node in NODES:
   103	            if read_completion(runtime, node) != read_json(runtime.directory / f"{node}.handoff.json"):
   104	                raise RuntimeError("Handoff changed after stop intent")
   105	        return
   106	    while True:
   107	        rows = runtime.sessions.inventory()
   108	        handoffs = {}
   109	        for node in NODES:
   110	            receipt = read_json(runtime.directory / f"{node}.interactive.json")
   111	            started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   112	            if clock() >= started + runtime.plan["automatic"]["worker_timeout_seconds"]:
   113	                raise RuntimeError(f"Worker {node} deadline exhausted; no automatic relaunch")
   114	            row = runtime.sessions.locate(node, rows)
   115	            if row is None:
   116	                raise RuntimeError("Native worker missing; reconciliation required")
   117	            if row["state"] == "blocked":
   118	                raise RuntimeError(f"Worker {node} blocked; inspect quota or native error. No billing/provider fallback.")
   119	            path = runtime.directory / f"{node}.completion.json"
   120	            if row["state"] in {"idle", "done"} and path.exists():
   121	                handoffs[node] = read_completion(runtime, node)
   122	        if set(handoffs) == set(NODES):
   123	            for node, value in handoffs.items():
   124	                save_json(runtime.directory / f"{node}.handoff.json", value)
   125	            return
   126	        sleep(2)

   129	REVIEW_SCHEMA = {
   130	    "type": "object", "additionalProperties": False, "required": ["verdict", "findings"],
   131	    "properties": {
   132	        "verdict": {"enum": ["approved", "blocked"]},
   133	        "findings": {"type": "array", "items": {
   134	            "type": "object", "additionalProperties": False,
   135	            "required": ["severity", "message", "disposition", "worker", "requirement"],
   136	            "properties": {"severity": {"enum": ["P0", "P1", "P2"]}, "message": {"type": "string", "minLength": 1},
   137	                           "disposition": {"enum": ["open", "resolved", "accepted"]},
   138	                           "worker": {"enum": ["ui", "adapter", "both", "none"]},
   139	                           "requirement": {"type": ["string", "null"], "minLength": 1}}}}}

   160	def completion_protocol_prompt(runtime, launch_token: str, digest: str, candidate_commit: str) -> str:
   161	    """The native reviewer reports its verdict only through a bound completion file."""
   162	    completion = runtime.directory / "review.completion.json"
   163	    example = {"version": "1.0.0", "run_id": runtime.plan["run_id"], "node_id": "review", "launch_token": launch_token,
   164	               "bundle_sha256": digest, "candidate_commit": candidate_commit, "verdict": "approved",
   165	               "findings": [{"severity": "P2", "message": "Describe the concrete defect and where it is", "disposition": "open",
   166	                             "worker": "ui", "requirement": "a verbatim quote from that worker's task text, or null"}]}
   167	    return ("\n\nREVIEW COMPLETION PROTOCOL: you run as a native session. A human may type in this terminal; the transcript "
   168	            f"is the record, but your verdict is only the file {completion}. Write exactly this JSON shape there "
   169	            "(schema: contracts/workflow/reviewCompletion.schema.json in this checkout when present):\n" + json.dumps(example) + "\n"
   170	            "Keep version, run_id, node_id, launch_token, bundle_sha256 and candidate_commit exactly as shown; the controller "
   171	            "rejects any other binding without launching another reviewer. verdict is approved or blocked. Each finding "
   172	            "has severity P0, P1 or P2 (P2 is the lowest; there is no P3, use P2 for minor items), disposition open, "
   173	            "resolved or accepted, worker (ui, adapter, both or none) and requirement (a verbatim quote from that "
   174	            "worker's task text in plan.json under nodes.<worker>.task, or null); no other keys. A file that does not "
   175	            "match this shape exactly is rejected as a whole. This completion file is the only write you are allowed. It cannot "

   181	def read_review_completion(runtime) -> dict:
   182	    """Accept the reviewer's file only when it validates and is bound to this exact run, launch, bundle and candidate."""
   183	    from jsonschema.exceptions import ValidationError
   184	    from .verification import validate_schema
   185	    path = runtime.directory / "review.completion.json"
   186	    if path.is_symlink() or not path.is_file() or path.stat().st_size > REVIEW_COMPLETION_LIMIT:
   187	        raise RuntimeError("Invalid review completion file")
   188	    try:
   189	        item = read_json(path)
   190	    except ValueError as error:
   191	        raise RuntimeError(f"Malformed review completion signal: {error}") from None
   192	    try:
   193	        validate_schema("reviewCompletion", item)
   194	    except ValidationError as error:
   195	        raise RuntimeError(f"Review completion signal violates the schema: {error.message}") from None
   196	    receipt = read_json(runtime.directory / "automatic-review.json")
   197	    bundle, digest = runtime.validate_bundle()
   198	    if (item["run_id"] != runtime.plan["run_id"] or item["node_id"] != "review" or item["launch_token"] != receipt.get("launch_token")
   199	            or item["bundle_sha256"] != digest or item["candidate_commit"] != bundle["candidate_commit"]):
   200	        raise RuntimeError("Stale or foreign review completion signal")
   201	    return {"verdict": item["verdict"], "findings": item["findings"]}

   204	def wait_review(runtime, *, clock=None, sleep=None) -> dict:
   205	    """Idle alone never means a verdict. The deadline counts from the reviewer's launch and survives restarts."""
   206	    clock = clock or time.time
   207	    sleep = sleep or time.sleep
   208	    validate_automatic(runtime.plan)
   209	    receipt = read_json(runtime.directory / "review.interactive.json")
   210	    started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   211	    path = runtime.directory / "review.completion.json"
   212	    attention = False
   213	    while True:
   214	        if clock() >= started + runtime.plan["automatic"]["review_timeout_seconds"]:
   215	            raise RuntimeError("Reviewer deadline exhausted; no second reviewer is launched")
   216	        row = runtime.sessions.locate("review", runtime.sessions.inventory())
   217	        if row is None:
   218	            raise RuntimeError("Native reviewer missing; reconciliation required")
   219	        if row["state"] == "blocked" and not attention:
   220	            # A native session reports `blocked` when it needs a human: a question or a prompt
   221	            # it cannot answer itself. The operator may answer in the pane; the deadline bounds it.
   222	            attention = True
   223	            runtime.event("review", "interactive", "Reviewer needs attention in its pane (native state blocked); waiting until the deadline")
   224	        if row["state"] in {"idle", "done"} and path.exists():
   225	            return read_review_completion(runtime)
   226	        sleep(2)

   229	def review_candidate(runtime) -> dict:
   230	    """Called only by the LangGraph review node. Ambiguous invocations never replay."""
   231	    validate_automatic(runtime.plan)
   232	    bundle, digest = runtime.validate_bundle()
   233	    receipt_path = runtime.directory / "automatic-review.json"
   234	    if receipt_path.exists():
   235	        receipt = read_json(receipt_path)
   236	        if receipt.get("bundle_sha256") == digest and receipt.get("status") == "succeeded":
   237	            runtime.validate_review(receipt["review"])
   238	            if receipt.get("transport") == "native" and not reviewer_stopped(runtime):
   239	                # Accepted earlier, but the stop was not confirmed: retry it (the stop intent makes it idempotent).
   240	                runtime.stop_reviewer()
   241	            return receipt["review"]
   242	        if receipt.get("bundle_sha256") == digest and receipt.get("status") == "running" and receipt.get("transport") == "native":
   243	            if not receipt.get("session_id"):
   244	                # Interrupted inside the launch window: bind the one session that launch produced, never another.
   245	                try:
   246	                    bound = runtime.reconcile_reviewer()
   247	                except Exception as error:
   248	                    receipt.update(status="needs_reconciliation", error=str(error))
   249	                    save_json(receipt_path, receipt)
   250	                    raise
   251	                receipt.update(session_id=bound["session_id"], background_id=bound.get("background_id"))
   252	                save_json(receipt_path, receipt)
   253	            # The reviewer kept running while the controller was away; keep waiting for its file.
   254	            return _accept_native(runtime, bundle, digest, receipt_path, receipt)
   255	        raise RuntimeError("Prior reviewer invocation needs reconciliation; no automatic relaunch")
   256	    cwd = runtime.directory / "review-worktree"
   257	    if cwd.exists():
   258	        raise RuntimeError("Partial review worktree exists; reconcile rather than overwrite")
   259	    subprocess.run(["git", "-C", runtime.plan["repository"], "worktree", "add", "--detach", str(cwd), bundle["candidate_commit"]],
   260	                   check=True, capture_output=True)
   261	    patch = runtime.directory / "review.diff"
   262	    with patch.open("w") as handle:
   263	        subprocess.run(["git", "-C", str(cwd), "diff", "--binary", runtime.plan["base_commit"], bundle["candidate_commit"]], stdout=handle, check=True)
   264	    if reviewer_transport(runtime.plan) == "print":
   265	        return _review_print(runtime, bundle, digest, cwd, patch, receipt_path)
   266	    return _review_native(runtime, bundle, digest, patch, receipt_path)

   269	def reviewer_stopped(runtime) -> bool:
   270	    """Only a confirmed stop intent counts; an absent or unconfirmed one means the stop is still owed."""
   271	    marker = runtime.directory / "review.stop.json"
   272	    return marker.exists() and read_json(marker).get("stopped") is True

   275	def _decide(runtime, bundle: dict, digest: str, reviewer: str, decision: dict, receipt: dict) -> dict:
   276	    """Persist review.json for any verdict; only an approved review without blocking findings passes."""
   277	    from .pipeline import blocking_findings
   278	    review = {"run_id": bundle["run_id"], "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   279	              "reviewer": reviewer, "independent": True, "verdict": decision["verdict"], "findings": decision["findings"]}
   280	    blocked = decision["verdict"] != "approved" or bool(blocking_findings(decision["findings"]))
   281	    if blocked:
   282	        review["verdict"] = "blocked"  # An approval with an unresolved P0/P1 is contradictory; the run is blocked.
   283	    save_json(runtime.directory / "review.json", review)
   284	    receipt["decision"] = decision
   285	    if blocked:
   286	        raise RuntimeError("Independent reviewer blocked the candidate")
   287	    runtime.validate_review(review)
   288	    receipt.update(status="succeeded", review=review, accepted_at=now())
   289	    return review

   292	def _review_native(runtime, bundle: dict, digest: str, patch: Path, receipt_path: Path) -> dict:
   293	    from .pipeline import digest_file
   294	    launch_token = str(uuid.uuid4())
   295	    receipt = {"transport": "native", "launch_token": launch_token, "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   296	               "patch_sha256": digest_file(patch), "status": "launching"}
   297	    save_json(receipt_path, receipt)
   298	    prompt = review_prompt(runtime, patch) + completion_protocol_prompt(runtime, launch_token, digest, bundle["candidate_commit"])
   299	    try:
   300	        launched = runtime.launch_reviewer(prompt, launch_token, bundle["candidate_commit"])
   301	    except KeyboardInterrupt:
   302	        issued = runtime.directory / "review.interactive.json"
   303	        if not issued.exists():
   304	            # No launch command was issued yet. Nothing is guessed either way: the operator reconciles.
   305	            receipt.update(status="needs_reconciliation", error="Interrupted before the reviewer launch was issued")
   306	            save_json(receipt_path, receipt)
   307	            raise
   308	        # `claude --bg` was issued (settle poll or pane attach interrupted): the session exists or is
   309	        # registering and keeps running. Record what is bound so far; resume binds the rest through
   310	        # reconciliation, never through a relaunch.
   311	        bound = read_json(issued)
   312	        receipt.update({key: bound[key] for key in ("session_id", "background_id") if bound.get(key)}, status="running")
   313	        save_json(receipt_path, receipt)
   314	        runtime.event("review", "interrupted", REVIEW_RESUME_NOTE.format(directory=runtime.directory))
   315	        raise
   316	    except BaseException as error:
   317	        # The launch may or may not have produced a session; only an operator can tell. Never launch again.
   318	        receipt.update(status="needs_reconciliation", error=str(error))
   319	        save_json(receipt_path, receipt)
   320	        raise
   321	    receipt.update(session_id=launched["session_id"], background_id=launched.get("background_id"), status="running")
   322	    save_json(receipt_path, receipt)
   323	    return _accept_native(runtime, bundle, digest, receipt_path, receipt)

   326	def _accept_native(runtime, bundle: dict, digest: str, receipt_path: Path, receipt: dict) -> dict:
   327	    from .pipeline import digest_file
   328	    cwd = runtime.directory / "review-worktree"
   329	    patch = runtime.directory / "review.diff"
   330	    try:
   331	        decision = wait_review(runtime)
   332	        row = runtime.sessions.locate("review", runtime.sessions.inventory())
   333	        worker_ids = {item["session_id"] for item in bundle["snapshots"].values()}
   334	        if row is None or row.get("sessionId") != receipt.get("session_id") or receipt.get("session_id") in worker_ids:
   335	            raise RuntimeError("Reviewer identity changed or is not independent; refusing the verdict")
   336	        if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   337	            raise RuntimeError("Reviewer worktree changed")
   338	        if runtime.validate_bundle()[1] != digest or digest_file(patch) != receipt["patch_sha256"]:
   339	            raise RuntimeError("Evidence changed during review")
   340	        review = _decide(runtime, bundle, digest, receipt["session_id"], decision, receipt)
   341	        save_json(receipt_path, receipt)
   342	    except KeyboardInterrupt:
   343	        # Operator/terminal interruption is not a reviewer failure: the native session keeps
   344	        # running and `automatic --live` resumes waiting for its completion file.
   345	        runtime.event("review", "interrupted", REVIEW_RESUME_NOTE.format(directory=runtime.directory))
   346	        raise
   347	    except BaseException as error:
   348	        # Deadline, blocked session, rejected file or blocked verdict: stop the one reviewer
   349	        # so it consumes no usage for a run that cannot continue. Nothing is relaunched.
   350	        receipt.update(status="blocked", error=str(error))
   351	        save_json(receipt_path, receipt)
   352	        try:
   353	            runtime.stop_reviewer()
   354	        except Exception as cleanup_error:
   355	            runtime.event("review", "blocked", f"Could not confirm reviewer stop: {cleanup_error}")
   356	        raise
   357	    try:
   358	        runtime.stop_reviewer()
   359	    except Exception as error:
   360	        # The verdict is durable (receipt succeeded, review.json written); only the stop is unconfirmed.
   361	        # review_candidate retries it on resume before handing the accepted review back.
   362	        runtime.event("review", "running", f"Could not confirm reviewer stop: {error}; resume retries the stop")
   363	        raise
   364	    return review

   367	def _review_print(runtime, bundle: dict, digest: str, cwd: Path, patch: Path, receipt_path: Path) -> dict:
   368	    """Headless fallback (--reviewer-transport print): one `claude --print` job, no pane, no human input."""
   369	    from jsonschema import validate
   370	    from .pipeline import digest_file
   371	    session_id = str(uuid.uuid4())
   372	    receipt = {"transport": "print", "session_id": session_id, "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   373	               "status": "launching", "patch_sha256": digest_file(patch)}
   374	    save_json(receipt_path, receipt)
   375	    prompt = review_prompt(runtime, patch) + " Return the requested JSON schema."
   376	    command = [runtime.sessions.executable, "--print", "--output-format", "json", "--session-id", session_id,
   377	               "--safe-mode", "--strict-mcp-config", "--mcp-config", '{"mcpServers":{}}',
   378	               "--tools", "Read,Glob,Grep", "--permission-mode", "dontAsk", "--permission-prompts", "none",
   379	               "--add-dir", str(runtime.directory), "--json-schema", json.dumps(REVIEW_SCHEMA)]
   380	    env = {key: value for key, value in os.environ.items() if not key.startswith("HERDR_")}
   381	    process = None
   382	    try:
   383	        with (runtime.directory / "review.stdout.json").open("w") as output, (runtime.directory / "review.stderr.log").open("w") as errors:
   384	            process = subprocess.Popen(command, cwd=cwd, env=env, stdin=subprocess.PIPE, stdout=output, stderr=errors,
   385	                                       text=True, start_new_session=True)
   386	            receipt.update(status="running", pid=process.pid)
   387	            save_json(receipt_path, receipt)
   388	            process.communicate(prompt, timeout=runtime.plan["automatic"]["review_timeout_seconds"])
   389	        result = read_json(runtime.directory / "review.stdout.json")
   390	        if process.returncode != 0 or result.get("session_id") != session_id or result.get("is_error") is not False or result.get("subtype") != "success":
   391	            raise RuntimeError("Reviewer did not succeed; inspect retained output. No automatic retry/provider switch.")
   392	        decision = result.get("structured_output")
   393	        validate(decision, REVIEW_SCHEMA)
   394	        if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   395	            raise RuntimeError("Reviewer worktree changed")
   396	        if runtime.validate_bundle()[1] != digest or digest_file(patch) != receipt["patch_sha256"]:
   397	            raise RuntimeError("Evidence changed during review")
   398	        return _decide(runtime, bundle, digest, session_id, decision, receipt)
   399	    except BaseException as error:
   400	        if process is not None and process.poll() is None:
   401	            terminate(process)
   402	        receipt.update(status="blocked", error=str(error))
   403	        raise
   404	    finally:
   405	        save_json(receipt_path, receipt)

   408	def advance_failed_checks(runtime, state) -> bool:
   409	    """Retry only recorded failing verification packets, never launches or review."""
   410	    # A checkpoint can carry an error from an earlier attempt of a task that has since
   411	    # succeeded (its writes are applied and it is no longer pending). Only pending
   412	    # tasks with errors are failures to classify.
   413	    failures = [task.name for task in state.tasks if task.error and task.name in state.next]
   414	    if not failures or any(name not in {"verify_ui", "verify_adapter", "candidate"} for name in failures):
   415	        return False
   416	    targets = []
   417	    for name in failures:
   418	        phase = "candidate" if name == "candidate" else "worker"
   419	        stage_targets = []
   420	        for node in NODES if phase == "candidate" else (name.removeprefix("verify_"),):
   421	            attempt = runtime.attempt(phase, node)
   422	            path = runtime.directory / "verification" / phase / node / str(attempt) / "packet.json"
   423	            if path.exists() and read_json(path)["gate"]["status"] != "passed":
   424	                previous = runtime.directory / "verification" / phase / node / str(attempt - 1) / "packet.json"
   425	                if attempt > 1 and previous.exists() and read_json(previous)["gate"]["reasons"] == read_json(path)["gate"]["reasons"]:
   426	                    # Retries rerun immutable code; two identical failures mean the cause is
   427	                    # deterministic (code or environment), and more attempts only burn time.
   428	                    raise RuntimeError(f"{phase}/{node} failed identically on attempts {attempt - 1} and {attempt}; "
   429	                                       f"not transient, inspect {path}")
   430	                stage_targets.append((phase, node))
   431	        if not stage_targets:
   432	            return False
   433	        targets.extend(stage_targets)
   434	    if not targets:
   435	        return False
   436	    # Check all bounds before changing any counters.
   437	    if any(runtime.attempt(p, n) >= runtime.policy.get("max_verification_attempts", 3) for p, n in targets):
   438	        raise RuntimeError("Verification retry limit exhausted; work and evidence retained")
   439	    for phase, node in targets:
   440	        runtime.retry_check(phase, node)
   441	    return True

   444	def supervise(directory: Path) -> None:
   445	    """Each recovery uses a new controller process, not just an in-memory replay."""
   446	    validate_automatic(read_json(directory / "plan.json"))
   447	    with run_lock(directory, "automatic-supervisor.lock"):
   448	        for _ in range(45):
   449	            try:
   450	                result = subprocess.run([sys.executable, "-m", "workflow", "automatic-step", str(directory), "--live"],
   451	                                        cwd=Path(__file__).resolve().parents[1])
   452	            except KeyboardInterrupt:
   453	                raise RuntimeError(RESUME_NOTE.format(directory=directory)) from None
   454	            if result.returncode == 0:
   455	                return
   456	            if result.returncode != 75:
   457	                raise RuntimeError(f"Automatic controller blocked (exit {result.returncode}); inspect retained run")
   458	        raise RuntimeError("Automatic controller restart limit exhausted")
   459	
   460	
   461	RESUME_NOTE = ("Supervisor interrupted. Native workers were NOT stopped and keep running; "
   462	               "resume with: python -m workflow automatic {directory} --live")
   463	REVIEW_STOP_NOTE = ("Reviewer stop not confirmed after acceptance: {error}. The verdict is kept; inspect the reviewer "
   464	                    "session, then resume retries the stop with: python -m workflow automatic {directory} --live")
   465	
   466	
   467	def reviewer_stop_pending(runtime, state) -> bool:
   468	    """The review node failed after its verdict was accepted durably, before the reviewer stop was confirmed.
   469	
   470	    Re-entering the node only retries that stop (review_candidate returns the persisted review)
   471	    and launches nothing, so a resumed controller may do it; nothing else about review is retried.
   472	    """
   473	    if [task.name for task in state.tasks if task.error and task.name in state.next] != ["review"]:
   474	        return False
   475	    receipt_path = runtime.directory / "automatic-review.json"
   476	    if not receipt_path.exists():
   477	        return False
   478	    receipt = read_json(receipt_path)
   479	    return receipt.get("transport") == "native" and receipt.get("status") == "succeeded" and not reviewer_stopped(runtime)

   482	def drive(runtime, *, single_step=False) -> str | None:
   483	    """Advance persisted graph state; CLI supervision restarts this process at joins."""
   484	    from .pipeline import build_pipeline, report
   485	    validate_automatic(runtime.plan)
   486	    if git(Path(runtime.plan["repository"]), "symbolic-ref", "--short", "HEAD") != runtime.plan["source_branch"]:
   487	        raise RuntimeError("Source feature branch changed; no automatic continuation")
   488	    runtime.event("controller", "running", f"Automatic checkpoint controller PID {os.getpid()}")
   489	    config = {"configurable": {"thread_id": runtime.plan["run_id"]}, "max_concurrency": 2}
   490	    while True:
   491	        with SqliteSaver.from_conn_string(str(runtime.directory / "pipeline.sqlite")) as saver:
   492	            graph = build_pipeline(saver, runtime)
   493	            state = graph.get_state(config)
   494	            if not state.values or any(name.startswith("launch_") for name in state.next):
   495	                raise RuntimeError("Automatic supervision requires a completed start; reconcile uncertain launches explicitly")
   496	            if not state.next:
   497	                commit = state.values.get("integrated_commit")
   498	                if (not commit or git(Path(runtime.plan["repository"]), "rev-parse", "HEAD") != commit
   499	                        or git(Path(runtime.plan["repository"]), "status", "--porcelain")):
   500	                    raise RuntimeError("No verified feature-branch completion")
   501	                runtime.validate_review(read_json(runtime.directory / "review.json"))
   502	                report(runtime, state)
   503	                return commit
   504	            pending = [item.value.get("kind") for task in state.tasks for item in task.interrupts]
   505	            value = None
   506	            if pending == ["worker_handoff"]:
   507	                try:
   508	                    wait_handoffs(runtime)
   509	                except KeyboardInterrupt:
   510	                    # Operator/terminal interruption is not a worker failure: leave the
   511	                    # native sessions running so `automatic --live` can resume polling.
   512	                    runtime.event("controller", "interrupted", RESUME_NOTE.format(directory=runtime.directory))
   513	                    report(runtime, state)
   514	                    raise
   515	                except BaseException as error:
   516	                    # Deadline, quota block, missing/blocked completion: stop the workers so
   517	                    # no session keeps consuming usage for a run that cannot continue.
   518	                    runtime.event("controller", "blocked", str(error))
   519	                    try:
   520	                        runtime.stop_workers()
   521	                    except Exception as cleanup_error:
   522	                        runtime.event("freeze", "blocked", f"Could not confirm worker stop: {cleanup_error}")
   523	                    report(runtime, state)
   524	                    raise
   525	                value = Command(resume={"freeze": True})
   526	            elif pending:
   527	                raise RuntimeError("Unexpected manual gate in automatic run; inspect state")
   528	            elif any(task.error for task in state.tasks) and not (advance_failed_checks(runtime, state) or reviewer_stop_pending(runtime, state)):
   529	                raise RuntimeError("Non-retryable graph failure; inspect retained evidence")
   530	            try:
   531	                graph.invoke(value, config)
   532	            except Exception as error:
   533	                failed = graph.get_state(config)
   534	                if not any(task.error for task in failed.tasks):
   535	                    raise
   536	                if reviewer_stop_pending(runtime, failed):
   537	                    # Not retried in this loop: the operator inspects the session first; a resumed
   538	                    # controller retries the stop once before continuing.
   539	                    raise RuntimeError(REVIEW_STOP_NOTE.format(error=error, directory=runtime.directory)) from error
   540	                # Next loop reopens the checkpointer and classifies the exact failure.
   541	            finally:
   542	                report(runtime, graph.get_state(config))
   543	            if single_step:
   544	                return None  # Exit 75: supervisor reopens this checkpoint in a fresh process.
