   110	        from .automatic import advance_failed_checks
   111	        policy = {"max_verification_attempts": 3}
   112	        attempts = {"worker:adapter": 2}
   113	        bumped = []
   114	        runtime = SimpleNamespace(directory=self.root, policy=policy,
   115	                                  attempt=lambda phase, node: attempts.get(f"{phase}:{node}", 1),
   116	                                  retry_check=lambda phase, node: bumped.append((phase, node)))
   117	        state = SimpleNamespace(next=("verify_adapter",), tasks=[SimpleNamespace(name="verify_adapter", error="blocked")])
   118	        reasons = ["backend-unit: exit 1", "backend-unit: no passing test evidence or failed tests"]

   124	            advance_failed_checks(runtime, state)
   125	        self.assertEqual(bumped, [])
   126	        # A different failure on the latest attempt is still retried within the cap.
   127	        save_json(self.root / "verification" / "worker" / "adapter" / "2" / "packet.json",
   128	                  {"gate": {"status": "blocked", "reasons": ["browser: Playwright reported global errors"]}})

   130	        self.assertEqual(bumped, [("worker", "adapter")])

   241	    def test_wait_accepts_only_an_idle_reviewer_with_a_file_and_never_relaunches(self):
   242	        from .automatic import wait_review
   243	        ticks = iter([1, 1, 1, DEFAULTS["review_timeout_seconds"] + 1])
   244	        with self.assertRaisesRegex(RuntimeError, "deadline exhausted; no second reviewer"):
   245	            wait_review(self.runtime, clock=lambda: next(ticks), sleep=lambda _: None)
   246	        save_json(self.root / "review.completion.json", self.completion())
   247	        self.rows[0]["state"] = "working"
   248	        ticks = iter([1, 1, DEFAULTS["review_timeout_seconds"] + 1])
   249	        with self.assertRaisesRegex(RuntimeError, "deadline exhausted"):
   250	            wait_review(self.runtime, clock=lambda: next(ticks), sleep=lambda _: None)
   251	        # `blocked` means the session needs a human (a question in its pane): the wait continues
   252	        # until the deadline, recorded once, and never treated as a verdict or a failure.
   253	        self.rows[0]["state"] = "blocked"
   254	        ticks = iter([1, 1, 1, DEFAULTS["review_timeout_seconds"] + 1])
   255	        with self.assertRaisesRegex(RuntimeError, "deadline exhausted"):
   256	            wait_review(self.runtime, clock=lambda: next(ticks), sleep=lambda _: None)
   257	        self.assertEqual([event[1] for event in self.events], ["interactive"])
   258	        self.assertIn("needs attention", self.events[0][2])
   259	        # Answered in the pane, the reviewer finishes: the file is accepted once the session is idle.
   260	        states = iter(["blocked", "blocked", "idle"])
   261	        self.rows[0]["state"] = "blocked"
   262	        def settle(_seconds):
   263	            self.rows[0]["state"] = next(states)
   264	        self.assertEqual(wait_review(self.runtime, clock=lambda: 1, sleep=settle)["verdict"], "approved")
   265	        self.rows[0]["state"] = "done"
   266	        self.assertEqual(wait_review(self.runtime, clock=lambda: 1, sleep=lambda _: self.fail("Unexpected wait"))["verdict"], "approved")
   267	        self.rows.clear()
   268	        with self.assertRaisesRegex(RuntimeError, "missing; reconciliation"):
   269	            wait_review(self.runtime, clock=lambda: 1, sleep=lambda _: None)

   474	    def test_interrupted_reviewer_wait_keeps_the_reviewer_and_resumes_in_a_new_process(self):
   475	        f = self.fixture
   476	        marker = f.root / "interrupt-once"
   477	        marker.touch()
   478	        script = '''import sys
   479	from pathlib import Path
   480	from workflow import automatic
   481	from workflow.sessions import read_json
   482	from workflow.test_pipeline import OfflinePipeline, FakeSessions
   483	directory, marker = Path(sys.argv[1]), Path(sys.argv[2])
   484	sessions = FakeSessions(directory, read_json(directory / 'plan.json'))
   485	runtime = OfflinePipeline(directory, sessions)
   486	automatic.wait_handoffs = lambda runtime: None
   487	real_wait = automatic.wait_review
   488	def interrupted_once(runtime, **kwargs):
   489	    if marker.exists():
   490	        marker.unlink()
   491	        raise KeyboardInterrupt
   492	    return real_wait(runtime, **kwargs)
   493	automatic.wait_review = interrupted_once
   494	try:
   495	    commit = automatic.drive(runtime, single_step=True)
   496	except KeyboardInterrupt:
   497	    sys.exit(130)
   498	assert not [node for node in sessions.starts if node != 'review'], 'Restart launched workers again'
   499	sys.exit(0 if commit else 75)
   500	'''
   501	        codes = []
   502	        for _ in range(5):
   503	            result = subprocess.run([sys.executable, "-c", script, str(f.directory), str(marker)],
   504	                                    cwd=Path(__file__).resolve().parents[1], capture_output=True, text=True, timeout=90)
   505	            codes.append(result.returncode)
   506	            self.assertIn(result.returncode, (0, 75, 130), result.stderr)
   507	            if result.returncode == 130:
   508	                # The interrupted controller left the reviewer running: receipt still running, no stop, no error.
   509	                receipt = read_json(f.directory / "automatic-review.json")
   510	                self.assertEqual((receipt["transport"], receipt["status"]), ("native", "running"))
   511	                self.assertNotIn("error", receipt)
   512	                self.assertFalse((f.directory / "review.stop.json").exists())
   513	                self.assertFalse((f.directory / "review.json").exists())
   514	                self.assertTrue(any(event["node"] == "review" and event["status"] == "interrupted" and "NOT stopped" in event["message"] for event in self.events()))
   515	                self.assertTrue(f.runtime.sessions.locate("review", f.runtime.sessions.inventory()))
   516	            if result.returncode == 0:
   517	                break
   518	        self.assertEqual(codes, [75, 130, 75, 0])
   519	        self.assertEqual(self.reviewer_launches(), 1)
   520	        self.assertEqual(read_json(f.directory / "review.json")["verdict"], "approved")
   521	        self.assertEqual(read_json(f.directory / "automatic-review.json")["status"], "succeeded")
   522	        self.assertTrue(read_json(f.directory / "review.stop.json")["stopped"])
   523	        self.assertEqual(git(f.repo, "rev-parse", self.original_branch), f.plan["base_commit"])
