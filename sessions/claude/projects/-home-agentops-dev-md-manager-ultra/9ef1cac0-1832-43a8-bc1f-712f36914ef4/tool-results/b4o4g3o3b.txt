    23	DEFAULTS = {"finish": "verified-feature-branch", "permission_mode": "bypassPermissions",
    24	            "worker_timeout_seconds": 4 * 3600, "review_timeout_seconds": 1800, "reviewer_transport": "native"}
    25	TIMEOUT_KEYS = ("worker_timeout_seconds", "review_timeout_seconds")
    26	REVIEWER_TRANSPORTS = ("native", "print")
    27	# Plans pinned before the native reviewer existed lack reviewer_transport; they mean native.
    28	REQUIRED_KEYS = frozenset(DEFAULTS) - {"reviewer_transport"}

    31	def automatic_settings(worker_timeout_seconds: int | None = None, review_timeout_seconds: int | None = None,
    32	                       reviewer_transport: str | None = None) -> dict:
    33	    """Run-scoped automatic configuration; deadlines and transport are pinned into plan.json at prepare."""
    34	    settings = dict(DEFAULTS)
    35	    for key, value in (("worker_timeout_seconds", worker_timeout_seconds), ("review_timeout_seconds", review_timeout_seconds),
    36	                       ("reviewer_transport", reviewer_transport)):
    37	        if value is not None:
    38	            settings[key] = value
    39	    validate_automatic({"automatic": settings, "source_branch": "feature/validation-only"})
    40	    return settings
    41

    43	def validate_automatic(plan: dict) -> None:
    44	    settings = plan.get("automatic")
    45	    if not isinstance(settings, dict) or not REQUIRED_KEYS <= set(settings) <= set(DEFAULTS):
    46	        raise ValueError("Malformed automatic run configuration")
    47	    if settings["finish"] != DEFAULTS["finish"] or settings["permission_mode"] != "bypassPermissions":
    48	        raise ValueError("Unsupported automatic authority")
    49	    for key in TIMEOUT_KEYS:
    50	        if type(settings[key]) is not int or not 1 <= settings[key] <= 86400:
    51	            raise ValueError("Automatic timeouts must be bounded positive seconds (at most 86400)")
    52	    if "reviewer_transport" in settings and settings["reviewer_transport"] not in REVIEWER_TRANSPORTS:
    53	        raise ValueError("Unsupported reviewer transport; expected native or print")
    54	    if not plan.get("source_branch", "").startswith("feature/"):
    55	        raise ValueError("Automatic completion is restricted to a feature/ branch")

    58	def reviewer_transport(plan: dict) -> str:
    59	    """Effective transport: plans that predate the setting are native, though they never launch a new reviewer."""
    60	    return plan["automatic"].get("reviewer_transport", "native")

    63	def completion_prompt(directory: Path, plan: dict, node: str) -> str:
    64	    example = {"version": "1.0.0", "run_id": plan["run_id"], "node_id": node,
    65	               "launch_token": plan["nodes"][node]["session_id"], "status": "completed",
    66	               "summary": "Describe actual work and checks executed", "open_assumptions": []}
    67	    return ("\n\nAUTOMATIC MODE: permission checks are bypassed and Bash is available. "
    68	            "Do not wait for a human handoff. Stay within assigned ownership; do not commit, merge, push, "
    69	            "launch agents, change runtime evidence or switch billing/provider. "
    70	            "On completion write the following JSON shape atomically (temporary file then rename) to "
    71	            f"{directory / (node + '.completion.json')}. This one output file is allowed outside your worktree. "
    72	            "Use status blocked if you cannot finish; never manufacture checks. Write it as your last action, "
    73	            "then finish your turn and do not modify more files. Controller checks and independent review "
    74	            "still determine acceptance.\n" + json.dumps(example))

    77	def read_completion(runtime, node: str) -> dict:
    78	    path = runtime.directory / f"{node}.completion.json"
    79	    if path.is_symlink() or not path.is_file() or path.stat().st_size > 65536:
    80	        raise ValueError(f"Invalid completion file for {node}")
    81	    item = read_json(path)
    82	    expected = {"version", "run_id", "node_id", "launch_token", "status", "summary", "open_assumptions"}
    83	    if not isinstance(item, dict) or set(item) != expected:
    84	        raise ValueError("Malformed completion signal")
    85	    if (item["version"] != "1.0.0" or item["run_id"] != runtime.plan["run_id"] or item["node_id"] != node
    86	            or item["launch_token"] != runtime.plan["nodes"][node]["session_id"]):
    87	        raise ValueError("Stale or foreign worker completion signal")
    88	    if item["status"] not in {"completed", "blocked"} or not isinstance(item["summary"], str) or not item["summary"].strip():
    89	        raise ValueError("Invalid completion status/summary")
    90	    if not isinstance(item["open_assumptions"], list) or any(not isinstance(x, str) for x in item["open_assumptions"]):
    91	        raise ValueError("Invalid completion assumptions")
    92	    if item["status"] == "blocked":
    93	        raise RuntimeError(f"Worker {node} explicitly blocked: {item['summary']}")
    94	    return {"summary": item["summary"], "open_assumptions": item["open_assumptions"]}

    97	def wait_handoffs(runtime, *, clock=time.time, sleep=time.sleep) -> None:
    98	    """Idle alone never means completion. Deadlines survive controller restart."""
    99	    validate_automatic(runtime.plan)
   100	    if any((runtime.directory / f"{node}.stop.json").exists() for node in NODES):
   101	        # Recover a controller crash after durable handoffs/stop intent, before snapshot.
   102	        for node in NODES:
   103	            if read_completion(runtime, node) != read_json(runtime.directory / f"{node}.handoff.json"):
   104	                raise RuntimeError("Handoff changed after stop intent")
   105	        return
   106	    while True:
   107	        rows = runtime.sessions.inventory()
   108	        handoffs = {}
   109	        for node in NODES:
   110	            receipt = read_json(runtime.directory / f"{node}.interactive.json")
   111	            started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   112	            if clock() >= started + runtime.plan["automatic"]["worker_timeout_seconds"]:
   113	                raise RuntimeError(f"Worker {node} deadline exhausted; no automatic relaunch")
   114	            row = runtime.sessions.locate(node, rows)
   115	            if row is None:
   116	                raise RuntimeError("Native worker missing; reconciliation required")
   117	            if row["state"] == "blocked":
   118	                raise RuntimeError(f"Worker {node} blocked; inspect quota or native error. No billing/provider fallback.")
   119	            path = runtime.directory / f"{node}.completion.json"
   120	            if row["state"] in {"idle", "done"} and path.exists():
   121	                handoffs[node] = read_completion(runtime, node)
   122	        if set(handoffs) == set(NODES):
   123	            for node, value in handoffs.items():
   124	                save_json(runtime.directory / f"{node}.handoff.json", value)
   125	            return
   126	        sleep(2)

   129	REVIEW_SCHEMA = {
   130	    "type": "object", "additionalProperties": False, "required": ["verdict", "findings"],
   131	    "properties": {
   132	        "verdict": {"enum": ["approved", "blocked"]},
   133	        "findings": {"type": "array", "items": {
   134	            "type": "object", "additionalProperties": False,

   137	                           "disposition": {"enum": ["open", "resolved", "accepted"]},
   138	                           "worker": {"enum": ["ui", "adapter", "both", "none"]},
   139	                           "requirement": {"type": ["string", "null"], "minLength": 1}}}}}

   167	    return ("\n\nREVIEW COMPLETION PROTOCOL: you run as a native session. A human may type in this terminal; the transcript "

   169	            "(schema: contracts/workflow/reviewCompletion.schema.json in this checkout when present):\n" + json.dumps(example) + "\n"

   179	def read_review_completion(runtime) -> dict:
   180	    """Accept the reviewer's file only when it validates and is bound to this exact run, launch, bundle and candidate."""
   181	    from jsonschema.exceptions import ValidationError
   182	    from .verification import validate_schema
   183	    path = runtime.directory / "review.completion.json"
   184	    if path.is_symlink() or not path.is_file() or path.stat().st_size > REVIEW_COMPLETION_LIMIT:
   185	        raise RuntimeError("Invalid review completion file")
   186	    try:
   187	        item = read_json(path)
   188	    except ValueError as error:
   189	        raise RuntimeError(f"Malformed review completion signal: {error}") from None
   190	    try:
   191	        validate_schema("reviewCompletion", item)
   192	    except ValidationError as error:
   193	        raise RuntimeError(f"Review completion signal violates the schema: {error.message}") from None
   194	    receipt = read_json(runtime.directory / "automatic-review.json")
   195	    bundle, digest = runtime.validate_bundle()
   196	    if (item["run_id"] != runtime.plan["run_id"] or item["node_id"] != "review" or item["launch_token"] != receipt.get("launch_token")
   197	            or item["bundle_sha256"] != digest or item["candidate_commit"] != bundle["candidate_commit"]):
   198	        raise RuntimeError("Stale or foreign review completion signal")
   199	    return {"verdict": item["verdict"], "findings": item["findings"]}

   202	def wait_review(runtime, *, clock=None, sleep=None) -> dict:
   203	    """Idle alone never means a verdict. The deadline counts from the reviewer's launch and survives restarts."""
   204	    clock = clock or time.time
   205	    sleep = sleep or time.sleep
   206	    validate_automatic(runtime.plan)
   207	    receipt = read_json(runtime.directory / "review.interactive.json")
   208	    started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   209	    path = runtime.directory / "review.completion.json"
   210	    attention = False
   211	    while True:
   212	        if clock() >= started + runtime.plan["automatic"]["review_timeout_seconds"]:
   213	            raise RuntimeError("Reviewer deadline exhausted; no second reviewer is launched")
   214	        row = runtime.sessions.locate("review", runtime.sessions.inventory())
   215	        if row is None:
   216	            raise RuntimeError("Native reviewer missing; reconciliation required")
   217	        if row["state"] == "blocked" and not attention:
   218	            # A native session reports `blocked` when it needs a human: a question or a prompt
   219	            # it cannot answer itself. The operator may answer in the pane; the deadline bounds it.
   220	            attention = True
   221	            runtime.event("review", "interactive", "Reviewer needs attention in its pane (native state blocked); waiting until the deadline")
   222	        if row["state"] in {"idle", "done"} and path.exists():
   223	            return read_review_completion(runtime)
   224	        sleep(2)

   227	def review_candidate(runtime) -> dict:
   228	    """Called only by the LangGraph review node. Ambiguous invocations never replay."""
   229	    validate_automatic(runtime.plan)
   230	    bundle, digest = runtime.validate_bundle()
   231	    receipt_path = runtime.directory / "automatic-review.json"
   232	    if receipt_path.exists():
   233	        receipt = read_json(receipt_path)
   234	        if receipt.get("bundle_sha256") == digest and receipt.get("status") == "succeeded":
   235	            runtime.validate_review(receipt["review"])
   236	            return receipt["review"]
   237	        if receipt.get("bundle_sha256") == digest and receipt.get("status") == "running" and receipt.get("transport") == "native":
   238	            # The reviewer kept running while the controller was away; keep waiting for its file.
   239	            return _accept_native(runtime, bundle, digest, receipt_path, receipt)
   240	        raise RuntimeError("Prior reviewer invocation needs reconciliation; no automatic relaunch")
   241	    cwd = runtime.directory / "review-worktree"
   242	    if cwd.exists():
   243	        raise RuntimeError("Partial review worktree exists; reconcile rather than overwrite")
   244	    subprocess.run(["git", "-C", runtime.plan["repository"], "worktree", "add", "--detach", str(cwd), bundle["candidate_commit"]],
   245	                   check=True, capture_output=True)
   246	    patch = runtime.directory / "review.diff"
   247	    with patch.open("w") as handle:
   248	        subprocess.run(["git", "-C", str(cwd), "diff", "--binary", runtime.plan["base_commit"], bundle["candidate_commit"]], stdout=handle, check=True)
   249	    if reviewer_transport(runtime.plan) == "print":
   250	        return _review_print(runtime, bundle, digest, cwd, patch, receipt_path)
   251	    return _review_native(runtime, bundle, digest, patch, receipt_path)

   254	def _decide(runtime, bundle: dict, digest: str, reviewer: str, decision: dict, receipt: dict) -> dict:
   255	    """Persist review.json for any verdict; only an approved review without blocking findings passes."""
   256	    from .pipeline import blocking_findings
   257	    review = {"run_id": bundle["run_id"], "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   258	              "reviewer": reviewer, "independent": True, "verdict": decision["verdict"], "findings": decision["findings"]}
   259	    blocked = decision["verdict"] != "approved" or bool(blocking_findings(decision["findings"]))
   260	    if blocked:
   261	        review["verdict"] = "blocked"  # An approval with an unresolved P0/P1 is contradictory; the run is blocked.
   262	    save_json(runtime.directory / "review.json", review)
   263	    receipt["decision"] = decision
   264	    if blocked:
   265	        raise RuntimeError("Independent reviewer blocked the candidate")
   266	    runtime.validate_review(review)
   267	    receipt.update(status="succeeded", review=review, accepted_at=now())
   268	    return review

   271	def _review_native(runtime, bundle: dict, digest: str, patch: Path, receipt_path: Path) -> dict:
   272	    from .pipeline import digest_file
   273	    launch_token = str(uuid.uuid4())
   274	    receipt = {"transport": "native", "launch_token": launch_token, "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   275	               "patch_sha256": digest_file(patch), "status": "launching"}
   276	    save_json(receipt_path, receipt)
   277	    prompt = review_prompt(runtime, patch) + completion_protocol_prompt(runtime, launch_token, digest, bundle["candidate_commit"])
   278	    try:
   279	        launched = runtime.launch_reviewer(prompt, launch_token, bundle["candidate_commit"])
   280	    except BaseException as error:
   281	        # The launch may or may not have produced a session; only an operator can tell. Never launch again.
   282	        receipt.update(status="needs_reconciliation", error=str(error))
   283	        save_json(receipt_path, receipt)
   284	        raise
   285	    receipt.update(session_id=launched["session_id"], background_id=launched.get("background_id"), status="running")
   286	    save_json(receipt_path, receipt)
   287	    return _accept_native(runtime, bundle, digest, receipt_path, receipt)

   290	def _accept_native(runtime, bundle: dict, digest: str, receipt_path: Path, receipt: dict) -> dict:
   291	    from .pipeline import digest_file
   292	    cwd = runtime.directory / "review-worktree"
   293	    patch = runtime.directory / "review.diff"
   294	    try:
   295	        decision = wait_review(runtime)
   296	        row = runtime.sessions.locate("review", runtime.sessions.inventory())
   297	        worker_ids = {item["session_id"] for item in bundle["snapshots"].values()}
   298	        if row is None or row.get("sessionId") != receipt.get("session_id") or receipt.get("session_id") in worker_ids:
   299	            raise RuntimeError("Reviewer identity changed or is not independent; refusing the verdict")
   300	        if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   301	            raise RuntimeError("Reviewer worktree changed")
   302	        if runtime.validate_bundle()[1] != digest or digest_file(patch) != receipt["patch_sha256"]:
   303	            raise RuntimeError("Evidence changed during review")
   304	        review = _decide(runtime, bundle, digest, receipt["session_id"], decision, receipt)
   305	        save_json(receipt_path, receipt)
   306	    except KeyboardInterrupt:
   307	        # Operator/terminal interruption is not a reviewer failure: the native session keeps
   308	        # running and `automatic --live` resumes waiting for its completion file.
   309	        runtime.event("review", "interrupted", REVIEW_RESUME_NOTE.format(directory=runtime.directory))
   310	        raise
   311	    except BaseException as error:
   312	        # Deadline, blocked session, rejected file or blocked verdict: stop the one reviewer
   313	        # so it consumes no usage for a run that cannot continue. Nothing is relaunched.
   314	        receipt.update(status="blocked", error=str(error))
   315	        save_json(receipt_path, receipt)
   316	        try:
   317	            runtime.stop_reviewer()
   318	        except Exception as cleanup_error:
   319	            runtime.event("review", "blocked", f"Could not confirm reviewer stop: {cleanup_error}")
   320	        raise
   321	    runtime.stop_reviewer()
   322	    return review

   325	def _review_print(runtime, bundle: dict, digest: str, cwd: Path, patch: Path, receipt_path: Path) -> dict:
   326	    """Headless fallback (--reviewer-transport print): one `claude --print` job, no pane, no human input."""
   327	    from jsonschema import validate
   328	    from .pipeline import digest_file
   329	    session_id = str(uuid.uuid4())
   330	    receipt = {"transport": "print", "session_id": session_id, "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   331	               "status": "launching", "patch_sha256": digest_file(patch)}
   332	    save_json(receipt_path, receipt)
   333	    prompt = review_prompt(runtime, patch) + " Return the requested JSON schema."
   334	    command = [runtime.sessions.executable, "--print", "--output-format", "json", "--session-id", session_id,
   335	               "--safe-mode", "--strict-mcp-config", "--mcp-config", '{"mcpServers":{}}',
   336	               "--tools", "Read,Glob,Grep", "--permission-mode", "dontAsk", "--permission-prompts", "none",
   337	               "--add-dir", str(runtime.directory), "--json-schema", json.dumps(REVIEW_SCHEMA)]
   338	    env = {key: value for key, value in os.environ.items() if not key.startswith("HERDR_")}
   339	    process = None
   340	    try:
   341	        with (runtime.directory / "review.stdout.json").open("w") as output, (runtime.directory / "review.stderr.log").open("w") as errors:
   342	            process = subprocess.Popen(command, cwd=cwd, env=env, stdin=subprocess.PIPE, stdout=output, stderr=errors,
   343	                                       text=True, start_new_session=True)
   344	            receipt.update(status="running", pid=process.pid)
   345	            save_json(receipt_path, receipt)
   346	            process.communicate(prompt, timeout=runtime.plan["automatic"]["review_timeout_seconds"])
   347	        result = read_json(runtime.directory / "review.stdout.json")
   348	        if process.returncode != 0 or result.get("session_id") != session_id or result.get("is_error") is not False or result.get("subtype") != "success":
   349	            raise RuntimeError("Reviewer did not succeed; inspect retained output. No automatic retry/provider switch.")
   350	        decision = result.get("structured_output")
   351	        validate(decision, REVIEW_SCHEMA)
   352	        if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   353	            raise RuntimeError("Reviewer worktree changed")
   354	        if runtime.validate_bundle()[1] != digest or digest_file(patch) != receipt["patch_sha256"]:
   355	            raise RuntimeError("Evidence changed during review")
   356	        return _decide(runtime, bundle, digest, session_id, decision, receipt)
   357	    except BaseException as error:
   358	        if process is not None and process.poll() is None:
   359	            terminate(process)
   360	        receipt.update(status="blocked", error=str(error))
   361	        raise
   362	    finally:
   363	        save_json(receipt_path, receipt)

   366	def advance_failed_checks(runtime, state) -> bool:
   367	    """Retry only recorded failing verification packets, never launches or review."""
   368	    # A checkpoint can carry an error from an earlier attempt of a task that has since
   369	    # succeeded (its writes are applied and it is no longer pending). Only pending
   370	    # tasks with errors are failures to classify.
   371	    failures = [task.name for task in state.tasks if task.error and task.name in state.next]
   372	    if not failures or any(name not in {"verify_ui", "verify_adapter", "candidate"} for name in failures):
   373	        return False
   374	    targets = []
   375	    for name in failures:
   376	        phase = "candidate" if name == "candidate" else "worker"
   377	        stage_targets = []
   378	        for node in NODES if phase == "candidate" else (name.removeprefix("verify_"),):
   379	            attempt = runtime.attempt(phase, node)
   380	            path = runtime.directory / "verification" / phase / node / str(attempt) / "packet.json"
   381	            if path.exists() and read_json(path)["gate"]["status"] != "passed":
   382	                previous = runtime.directory / "verification" / phase / node / str(attempt - 1) / "packet.json"
   383	                if attempt > 1 and previous.exists() and read_json(previous)["gate"]["reasons"] == read_json(path)["gate"]["reasons"]:
   384	                    # Retries rerun immutable code; two identical failures mean the cause is
   385	                    # deterministic (code or environment), and more attempts only burn time.
   386	                    raise RuntimeError(f"{phase}/{node} failed identically on attempts {attempt - 1} and {attempt}; "
   387	                                       f"not transient, inspect {path}")
   388	                stage_targets.append((phase, node))
   389	        if not stage_targets:
   390	            return False
   391	        targets.extend(stage_targets)
   392	    if not targets:
   393	        return False
   394	    # Check all bounds before changing any counters.
   395	    if any(runtime.attempt(p, n) >= runtime.policy.get("max_verification_attempts", 3) for p, n in targets):
   396	        raise RuntimeError("Verification retry limit exhausted; work and evidence retained")
   397	    for phase, node in targets:
   398	        runtime.retry_check(phase, node)
   399	    return True

   402	def supervise(directory: Path) -> None:
   403	    """Each recovery uses a new controller process, not just an in-memory replay."""
   404	    validate_automatic(read_json(directory / "plan.json"))
   405	    with run_lock(directory, "automatic-supervisor.lock"):
   406	        for _ in range(45):
   407	            try:
   408	                result = subprocess.run([sys.executable, "-m", "workflow", "automatic-step", str(directory), "--live"],
   409	                                        cwd=Path(__file__).resolve().parents[1])
   410	            except KeyboardInterrupt:
   411	                raise RuntimeError(RESUME_NOTE.format(directory=directory)) from None
   412	            if result.returncode == 0:
   413	                return
   414	            if result.returncode != 75:
   415	                raise RuntimeError(f"Automatic controller blocked (exit {result.returncode}); inspect retained run")
   416	        raise RuntimeError("Automatic controller restart limit exhausted")
   417	
   418	
   419	RESUME_NOTE = ("Supervisor interrupted. Native workers were NOT stopped and keep running; "
   420	               "resume with: python -m workflow automatic {directory} --live")

   422	
   423	def drive(runtime, *, single_step=False) -> str | None:
   424	    """Advance persisted graph state; CLI supervision restarts this process at joins."""
   425	    from .pipeline import build_pipeline, report
   426	    validate_automatic(runtime.plan)
   427	    if git(Path(runtime.plan["repository"]), "symbolic-ref", "--short", "HEAD") != runtime.plan["source_branch"]:
   428	        raise RuntimeError("Source feature branch changed; no automatic continuation")
   429	    runtime.event("controller", "running", f"Automatic checkpoint controller PID {os.getpid()}")
   430	    config = {"configurable": {"thread_id": runtime.plan["run_id"]}, "max_concurrency": 2}
   431	    while True:
   432	        with SqliteSaver.from_conn_string(str(runtime.directory / "pipeline.sqlite")) as saver:
   433	            graph = build_pipeline(saver, runtime)
   434	            state = graph.get_state(config)
   435	            if not state.values or any(name.startswith("launch_") for name in state.next):
   436	                raise RuntimeError("Automatic supervision requires a completed start; reconcile uncertain launches explicitly")
   437	            if not state.next:
   438	                commit = state.values.get("integrated_commit")
   439	                if (not commit or git(Path(runtime.plan["repository"]), "rev-parse", "HEAD") != commit
   440	                        or git(Path(runtime.plan["repository"]), "status", "--porcelain")):
   441	                    raise RuntimeError("No verified feature-branch completion")
   442	                runtime.validate_review(read_json(runtime.directory / "review.json"))
   443	                report(runtime, state)
   444	                return commit
   445	            pending = [item.value.get("kind") for task in state.tasks for item in task.interrupts]
   446	            value = None
   447	            if pending == ["worker_handoff"]:
   448	                try:
   449	                    wait_handoffs(runtime)
   450	                except KeyboardInterrupt:
   451	                    # Operator/terminal interruption is not a worker failure: leave the
   452	                    # native sessions running so `automatic --live` can resume polling.
   453	                    runtime.event("controller", "interrupted", RESUME_NOTE.format(directory=runtime.directory))
   454	                    report(runtime, state)
   455	                    raise
   456	                except BaseException as error:
   457	                    # Deadline, quota block, missing/blocked completion: stop the workers so
   458	                    # no session keeps consuming usage for a run that cannot continue.
   459	                    runtime.event("controller", "blocked", str(error))
   460	                    try:
   461	                        runtime.stop_workers()
   462	                    except Exception as cleanup_error:
   463	                        runtime.event("freeze", "blocked", f"Could not confirm worker stop: {cleanup_error}")
   464	                    report(runtime, state)
   465	                    raise
   466	                value = Command(resume={"freeze": True})
   467	            elif pending:
   468	                raise RuntimeError("Unexpected manual gate in automatic run; inspect state")
   469	            elif any(task.error for task in state.tasks) and not advance_failed_checks(runtime, state):
   470	                raise RuntimeError("Non-retryable graph failure; inspect retained evidence")
   471	            try:
   472	                graph.invoke(value, config)
   473	            except Exception:
   474	                failed = graph.get_state(config)
   475	                if not any(task.error for task in failed.tasks):
   476	                    raise
   477	                # Next loop reopens the checkpointer and classifies the exact failure.
   478	            finally:
   479	                report(runtime, graph.get_state(config))
   480	            if single_step:
   481	                return None  # Exit 75: supervisor reopens this checkpoint in a fresh process.
