    52	/** Reads the rendered text and asserts no absolute filesystem path (temporary root or home) leaked into it. */

    59	/** Fetches a URL from inside the page, so worker-mode route mocks apply exactly as they do for the app. */
    60	export async function fetchFromPage(page: Page, url: string): Promise<{ status: number; contentType: string | null; text: string }> {

    63	    return { status: response.status, contentType: response.headers.get('content-type'), text: await response.text() }
    64	  }, url)
