   109	        from .automatic import advance_failed_checks
   110	        policy = {"max_verification_attempts": 3}
   111	        attempts = {"worker:adapter": 2}
   112	        bumped = []
   113	        runtime = SimpleNamespace(directory=self.root, policy=policy,
   114	                                  attempt=lambda phase, node: attempts.get(f"{phase}:{node}", 1),
   115	                                  retry_check=lambda phase, node: bumped.append((phase, node)))
   116	        state = SimpleNamespace(next=("verify_adapter",), tasks=[SimpleNamespace(name="verify_adapter", error="blocked")])
   117	        reasons = ["backend-unit: exit 1", "backend-unit: no passing test evidence or failed tests"]

   119	            (self.root / "verification" / "worker" / "adapter" / str(attempt)).mkdir(parents=True)

   123	            advance_failed_checks(runtime, state)
   124	        self.assertEqual(bumped, [])
   125	        # A different failure on the latest attempt is still retried within the cap.
   126	        save_json(self.root / "verification" / "worker" / "adapter" / "2" / "packet.json",
   127	                  {"gate": {"status": "blocked", "reasons": ["browser: Playwright reported global errors"]}})
