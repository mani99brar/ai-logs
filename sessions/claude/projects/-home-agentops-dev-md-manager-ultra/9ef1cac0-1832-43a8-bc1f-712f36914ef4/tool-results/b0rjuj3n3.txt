    43	/** No view action may launch, approve, retry, delete or edit a run: no such real controls exist (graph nodes are read-only selectors). */
    44	async function expectNoExecutionControls(page: Page) {
    45	  const controls = workspace(page).locator('button, input, select, textarea, [role="menuitem"]')
    46	  await expect(controls.filter({ hasText: /approve|retry|launch|start|resume|delete|cancel|integrate|edit|save/i })).toHaveCount(0)
    47	  await expect(page.locator('form')).toHaveCount(0)
    48	}

    26	function freePort(): number {
    27	  const script = "const s=require('net').createServer();s.listen(0,'127.0.0.1',()=>{process.stdout.write(String(s.address().port));s.close()})"
    28	  return Number(execFileSync(process.execPath, ['-e', script], { encoding: 'utf8' }).trim())
    29	}
