    80	/** Definition revision exactly as the contract specifies: SHA-256 of canonical JSON without `definition_revision`. */
    81	export function definitionRevision(definition: { contract_version: string; project_id: string; workflow_id: string; name: string; nodes: DefinitionNode[] }): string {
    82	  const canonical = (value: unknown): string => {
    83	    if (Array.isArray(value)) return `[${value.map(canonical).join(',')}]`
    84	    if (value && typeof value === 'object') {
    85	      return `{${Object.keys(value as Record<string, unknown>).sort().map(key => `${JSON.stringify(key)}:${canonical((value as Record<string, unknown>)[key])}`).join(',')}}`
    86	    }
    87	    return JSON.stringify(value)
    88	  }
    89	  const { contract_version, project_id, workflow_id, name, nodes } = definition
    90	  return createHash('sha256').update(canonical({ contract_version, project_id, workflow_id, name, nodes }), 'utf8').digest('hex')
    91	}

   117	function crc32(bytes: Buffer): number {
   118	  let crc = ~0
   119	  for (const byte of bytes) {
   120	    crc ^= byte
   121	    for (let bit = 0; bit < 8; bit += 1) crc = (crc >>> 1) ^ (0xedb88320 & -(crc & 1))
   122	  }
   123	  return ~crc >>> 0
   124	}

   126	function chunk(type: string, data: Buffer): Buffer {
   127	  const length = Buffer.alloc(4)
   128	  length.writeUInt32BE(data.length)
   129	  const body = Buffer.concat([Buffer.from(type, 'ascii'), data])
   130	  const crc = Buffer.alloc(4)
   131	  crc.writeUInt32BE(crc32(body))
   132	  return Buffer.concat([length, body, crc])
   133	}

   135	export function solidPng(width: number, height: number, rgb: [number, number, number]): Buffer {
   136	  const header = Buffer.alloc(13)
   137	  header.writeUInt32BE(width, 0)
   138	  header.writeUInt32BE(height, 4)
   139	  header[8] = 8 // bit depth
   140	  header[9] = 2 // colour type: RGB
   141	  const rows: Buffer[] = []
   142	  for (let y = 0; y < height; y += 1) {
   143	    const row = Buffer.alloc(1 + width * 3)
   144	    for (let x = 0; x < width; x += 1) row.set(rgb, 1 + x * 3)
   145	    rows.push(row)
   146	  }
   147	  return Buffer.concat([
   148	    Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]),
   149	    chunk('IHDR', header),
   150	    chunk('IDAT', deflateSync(Buffer.concat(rows))),
   151	    chunk('IEND', Buffer.alloc(0)),
   152	  ])
   153	}

   232	function runDetail(runId: string, status: RunDetail['summary']['status'], pinned: WorkflowDefinition, createdAt: string, updatedAt: string, states: Record<string, NodeState>, lastSequence: number): RunDetail {
   233	  const resultPath = (node: string, attempt: number) => `${apiRunPath(runId)}/results/${node}/${attempt}`
   234	  return validateRunDetail({
   235	    summary: {
   236	      contract_version: '1.0.0', project_id: PROJECT.project_id, workflow_id: WORKFLOW_ID,
   237	      definition_revision: pinned.definition_revision, run_id: runId, status, created_at: createdAt, updated_at: updatedAt,
   238	    },
   239	    definition: pinned,
   240	    snapshot: {
   241	      contract_version: '1.0.0', run_id: runId, status, last_sequence: lastSequence,
   242	      nodes: pinned.nodes.map(node => {
   243	        const state = states[node.node_id] ?? { status: 'pending', attempt: 0 }
   244	        return {
   245	          node_id: node.node_id, kind: node.kind, depends_on: node.depends_on, status: state.status, attempt: state.attempt,
   246	          session_id: state.session ?? null,
   247	          result_uri: state.review ? `${apiRunPath(runId)}/reviews/${state.review}` : state.result ? resultPath(state.result, state.attempt) : null,
   248	        }
   249	      }),
   250	    },
   251	  })
   252	}

   500	function rawWorker(lane: 'ui' | 'adapter', runId: string, leak: string, requestedAt: string): RawWorkerInput {
   501	  const ui = lane === 'ui'
   502	  const failedAdapter = !ui && runId === RUN_FAILED
   503	  return {
   504	    role: ui ? 'frontend' : 'backend',
   505	    task: ui ? uiTask(leak) : adapterTask(),
   506	    prompt: ui ? uiPrompt(leak) : null,
   507	    owned_paths: ui ? ['src/App.tsx', 'src/App.css', 'src/projects', 'tests/project-workflows'] : ['server', 'config/projects.example.json'],
   508	    checks: ui ? UI_CHECKS : ADAPTER_CHECKS,
   509	    launch: {
   510	      session_id: ui ? UI_SESSION : ADAPTER_SESSION, launch_token: launchToken(lane), launch_requested_at: offset(requestedAt),
   511	      native_started_at: Date.parse(requestedAt) + 2000, observed_state: failedAdapter ? 'working' : 'done', status: 'attached_session_available',
   512	      launcher_invocations: 1, background_id: ui ? '10001' : '10002',
   513	    },
   514	    completion: failedAdapter ? null : ui
   515	      ? { status: 'completed', summary: uiCompletionSummary(leak), open_assumptions: [UI_ASSUMPTION] }
   516	      : { status: 'completed', summary: ADAPTER_COMPLETION_SUMMARY, open_assumptions: [] },
   517	    // The UI handoff repeats the completion; the adapter's was amended when it was accepted, so it differs.
   518	    handoff: failedAdapter ? null : ui
   519	      ? { summary: uiCompletionSummary(leak), open_assumptions: [UI_ASSUMPTION] }
   520	      : { summary: ADAPTER_HANDOFF_SUMMARY, open_assumptions: [ADAPTER_HANDOFF_ASSUMPTION] },
   521	    stop: failedAdapter ? { stopped: false, confirmed_at: null } : { stopped: true, confirmed_at: offset(T2) },
   522	  }
   523	}

   525	/** The export's `inputs` section for a run, or null for the legacy export that predates it. */
   526	export function rawInputsSection(runId: string, leak = PATH_TOKEN): RawInputsSection | null {
   527	  if (runId === RUN_LEGACY) return null
   528	  const requestedAt = runId === RUN_SUCCEEDED ? T0 : runId === RUN_AWAITING ? T2 : T1
   529	  const manual = runId === RUN_AWAITING
   530	  return {
   531	    feature: FEATURE_NAME, policy_version: '1.1.0', base_commit: BASE_COMMIT, source_branch: sourceBranch(runId),
   532	    mode: manual ? 'manual' : 'automatic',
   533	    automatic: manual ? null : {
   534	      finish: 'verified-feature-branch', permission_mode: 'bypassPermissions',
   535	      worker_timeout_seconds: runId === RUN_SUCCEEDED ? 14400 : 3600, review_timeout_seconds: 1800,
   536	      reviewer_transport: runId === RUN_BLOCKED ? 'print' : 'native',
   537	    },
   538	    setup: [{ argv: ['npm', 'ci'], command: 'npm ci', timeout_seconds: 600 }],
   539	    max_verification_attempts: 3,
   540	    failure_drill: runId === RUN_FAILED ? { node_id: 'adapter', phase: 'worker', attempt: 1 } : null,
   541	    workers: { ui: rawWorker('ui', runId, leak, requestedAt), adapter: rawWorker('adapter', runId, leak, requestedAt) },
   542	  }
   543	}

   553	function projectReview(runId: string, section: RawReviewSection, tasks: Record<'ui' | 'adapter', string>): ReviewResult {
   554	  return validateReviewResult({
   555	    contract_version: '1.2.0', run_id: runId, node_id: 'review', attempt: section.attempt,
   556	    reviewer: { session_id: section.reviewer_session_id, transport: section.transport, independent: true },
   557	    bundle_sha256: section.bundle_sha256, candidate_commit: section.candidate_commit, verdict: section.verdict,
   558	    findings: section.findings.map(finding => ({ ...finding, requirement_found_in: lanesQuoting(finding.requirement, tasks) })),
   559	    reviewed_at: zulu(section.reviewed_at),
   560	    diff: section.diff === null ? null : { artifact_id: REVIEW_DIFF_ARTIFACT_ID, kind: 'patch', uri: `${apiRunPath(runId)}/artifacts/${REVIEW_DIFF_ARTIFACT_ID}`, sha256: section.diff.sha256 },
   561	  })

   564	function projectInputs(runId: string, section: RawInputsSection): RunInputs {
   565	  return validateRunInputs({
   566	    contract_version: '1.2.0', run_id: runId, feature: section.feature, base_commit: section.base_commit, source_branch: section.source_branch,
   567	    mode: section.mode, automatic: section.automatic,
   568	    setup: section.setup.map(step => ({ command: step.command, timeout_seconds: step.timeout_seconds })),
   569	    max_verification_attempts: section.max_verification_attempts,
   570	    workers: (['ui', 'adapter'] as const).map(lane => {
   571	      const worker = section.workers[lane]
   572	      return {
   573	        node_id: lane, launch_node_id: `launch_${lane}`, role: worker.role,
   574	        task: { text: worker.task, truncated: false },
   575	        prompt: worker.prompt === null ? null : { text: worker.prompt, truncated: false },
   576	        owned_paths: worker.owned_paths,
   577	        checks: worker.checks.map(({ id, kind, command, timeout_seconds, scenarios }) => ({ id, kind, command, timeout_seconds, scenarios })),
   578	        launch: worker.launch === null ? null : {
   579	          session_id: worker.launch.session_id, launch_requested_at: zulu(worker.launch.launch_requested_at),
   580	          native_started_at: worker.launch.native_started_at === null ? null : new Date(worker.launch.native_started_at).toISOString(),

   587	  })
