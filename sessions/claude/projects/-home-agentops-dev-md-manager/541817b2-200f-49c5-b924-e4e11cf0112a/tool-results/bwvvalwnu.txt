    56	/** The graph node that launched a lane: from the inputs when loaded, else `launch_<lane>` when the pinned graph has it, else the lane itself. */
    57	function launchNodeFor(lane: Lane, definitionNodes: DefinitionNode[], inputs: Resource<RunInputs | null>): string {
    58	  if (inputs.status === 'ready' && inputs.data !== null) {
    59	    const worker = inputs.data.workers.find(candidate => candidate.node_id === lane)
    60	    if (worker) return worker.launch_node_id
    61	  }
    62	  return definitionNodes.some(candidate => candidate.node_id === `launch_${lane}`) ? `launch_${lane}` : lane
    63	}

    71	function FindingRow({ finding, scope, definitionNodes, inputs, onOpenRequirement, onOpenFile, capturedFiles }: RowProps) {
    72	  const blocking = isBlockingFinding(finding)
    73	  // A finding links to every captured file its message names verbatim; the review result itself records no location.
    74	  const files = [...capturedFiles].filter(([path]) => findingsForFile([finding], path).length > 0)
    75	  return (
    76	    <tr
    77	      data-testid="finding"
    78	      data-severity={finding.severity}
    79	      data-disposition={finding.disposition}
    80	      data-worker={finding.worker ?? 'unrecorded'}
    81	      data-reviewer={finding.reviewer}
    82	      className={blocking ? 'finding-blocking' : undefined}
    83	    >
    84	      <td><span className="finding-severity">{finding.severity}</span>{blocking && <span className="visually-hidden"> (blocks integration)</span>}</td>
    85	      <td>
    86	        {finding.message}
    87	        {files.map(([path, nodeId]) => (
    88	          <AppLink
    89	            key={path}
    90	            href={runPathname(scope.projectId, scope.workflowId, scope.runId, nodeId)}
    91	            onNavigate={() => onOpenFile(nodeId, path)}
    92	            className="finding-file-link"
    93	            data-testid="finding-file-link"
    94	            data-path={path}
    95	            style={{ display: 'block' }}
    96	          >

    98	          </AppLink>
    99	        ))}
   100	      </td>
   101	      <td>{finding.worker === null ? <span className="projects-muted">not recorded</span> : workerWording(finding.worker)}</td>
   102	      <td data-testid="finding-reviewer">{finding.reviewer}</td>
   103	      <td>
   104	        {finding.requirement === null ? '—' : (
   105	          <>
   106	            <q data-testid="finding-requirement">{finding.requirement}</q>

   108	              <span className="projects-muted finding-task-note" data-testid="finding-task-unlinked">not found verbatim in the task</span>

   132	/** A reviewer's own verdict; a reviewer superseded before deciding, still pending, or blocked by the deadline or a rejected file has none. */

   134	  if (verdict === null) return <span className="projects-muted" data-testid="reviewer-verdict" data-status="none">No verdict</span>
   135	  const approved = verdict === 'approved'
   136	  return (
   137	    <span className={`status-badge ${approved ? 'status-succeeded' : 'status-failed'}`} data-testid="reviewer-verdict" data-status={verdict}>

   139	    </span>
   140	  )

   170	/**
   171	 * One group per reviewer of the run in declared order, labelled with its id, its own verdict and its severity counts, then a
   172	 * guard group for findings whose reviewer the run does not list (the contract forbids it, so it stays hidden when empty).
   173	 */
   174	function reviewerGroups(reviewers: readonly Reviewer[], visible: readonly ReviewFinding[]) {
   175	  const ids = new Set(reviewers.map(reviewer => reviewer.reviewer_id))
   176	  const verdictWording = (verdict: Reviewer['verdict']) => (verdict === null ? 'no verdict' : verdict === 'approved' ? 'approved' : 'blocked')
   177	  return [
   178	    ...reviewers.map(reviewer => ({
   179	      key: `reviewer-${reviewer.reviewer_id}`,
   180	      label: `${reviewer.reviewer_id} · ${verdictWording(reviewer.verdict)} · ${severityCounts(reviewer.findings)}`,
   181	      attributes: { 'data-reviewer-group': reviewer.reviewer_id, 'data-verdict': reviewer.verdict ?? 'none' },
   182	      findings: visible.filter(finding => finding.reviewer === reviewer.reviewer_id),
   183	    })),
   184	    {
   185	      key: `reviewer-${UNLISTED_REVIEWER}`,
   186	      label: 'Reviewer not listed for this run',
   187	      attributes: { 'data-reviewer-group': UNLISTED_REVIEWER, 'data-verdict': 'none' },
   188	      findings: visible.filter(finding => !ids.has(finding.reviewer)),
   189	    },
   190	  ]
   191	}

   193	function ReviewResultView({ review, scope, definitionNodes, snapshotNodes, inputs, refreshToken, onNavigate, onOpenRequirement, onOpenFile, onTransport }: { review: ReviewResult } & Omit<Props, 'node'>) {
   194	  const approved = review.verdict === 'approved'
   195	  const reviewers = review.reviewers
   196	  const several = reviewers.length > 1
   197	  const blockedBy = blockedByWording(review)
   198	  const candidateNode = definitionNodes.some(candidate => candidate.node_id === 'candidate') ? 'candidate' : null
   199	  const bundle = <code title={review.bundle_sha256}>{shortRevision(review.bundle_sha256)}</code>
   200	  // Like every other artifact link, the diff is only ever fetched through this run's own artifact route by its ID.
   201	  const diffHref = review.diff === null ? null : paths.artifact(scope, review.diff.artifact_id)
   202	  const diffScoped = review.diff !== null && review.diff.uri === diffHref
   203	  // Several reviewers read best as one group each; a single reviewer's findings group by disposition.
   204	  const [groupBy, setGroupBy] = useState<GroupBy>(several ? 'reviewer' : 'disposition')
   205	  const [reviewerFilter, setReviewerFilter] = useState<ReviewerFilter>(null)
   206	  const lanes = runLanes(definitionNodes, inputs)
   207	  const capturedFiles = useRunCapturedFiles(scope, snapshotNodes, refreshToken)
   208	  const transport = review.reviewer.transport
   209	  useEffect(() => { onTransport?.(transport) }, [onTransport, transport])
   210	  // The filter narrows the union to one reviewer's findings; grouping then applies to what is left.
   211	  const visible = reviewerFilter === null ? review.findings : review.findings.filter(finding => finding.reviewer === reviewerFilter)
   212	  const countFor = (reviewerId: string) => review.findings.filter(finding => finding.reviewer === reviewerId).length
   213	  const groups = groupBy === 'disposition'
   214	    ? DISPOSITIONS.map(disposition => ({ key: disposition, label: DISPOSITION_LABEL[disposition], attributes: { 'data-disposition': disposition }, findings: visible.filter(finding => finding.disposition === disposition) }))
   215	    : groupBy === 'reviewer'
   216	      ? reviewerGroups(reviewers, visible)
   217	      : workerGroups(lanes, visible).map(group => ({ ...group, attributes: { 'data-worker-group': group.key }, findings: visible.filter(finding => workerGroupOf(finding) === group.key) }))
   218	  const populated = groups.filter(group => group.findings.length > 0)
   219	  return (
   220	    <div className="review-result" data-testid="review-result" data-verdict={review.verdict} data-reviewer-count={reviewers.length}>
   221	      <p className="review-verdict-line">
   222	        <span className={`status-badge ${approved ? 'status-succeeded' : 'status-failed'}`} data-testid="review-verdict" data-status={review.verdict}>
   223	          <span>{approved ? 'Approved' : 'Blocked'}</span>
   224	        </span>
   225	        <span className="projects-muted">
   226	          Combined verdict of {several ? `${reviewers.length} reviewers` : 'the one reviewer'} for review attempt {review.attempt}: every reviewer must approve; approval and integration are separate nodes.
   227	        </span>
   228	      </p>
   229	      {blockedBy !== null && <p className="projects-error-inline" role="status" data-testid="review-blocked-by">{blockedBy}</p>}
   230	      <p className="review-summary" data-testid="review-summary">{reviewSummary(review)}</p>
   231	      <dl className="projects-facts">
   232	        <div>
   233	          <dt>{several ? 'Reviewers' : 'Reviewer'}</dt>
   234	          <dd data-testid="review-reviewer">
   235	            {reviewers.map((reviewer, index) => (
   236	              <span key={reviewer.reviewer_id} data-reviewer={reviewer.reviewer_id}>
   237	                {index > 0 && ', '}
   238	                {several && `${reviewer.reviewer_id}: `}{reviewer.session_id ? <code>{reviewer.session_id}</code> : <span className="projects-muted">no session recorded</span>} · {TRANSPORT_WORDING[reviewer.transport]}
   239	              </span>
   240	            ))}
   241	            {several ? '; independent of every worker lane and of one another' : ', independent of every worker lane'}
   242	          </dd>
   243	        </div>
   244	        <div><dt>Reviewed at</dt><dd>{formatTime(review.reviewed_at)}</dd></div>
   245	        <div>
   246	          <dt>Bundle reviewed</dt>
   247	          <dd data-testid="review-bundle">
   248	            {candidateNode === null ? bundle : (
   249	              <AppLink href={runPathname(scope.projectId, scope.workflowId, scope.runId, candidateNode)} onNavigate={onNavigate} title="Open the candidate node this bundle was built from">{bundle}</AppLink>
   250	            )}
   251	            {' '}· candidate commit <code title={review.candidate_commit}>{shortRevision(review.candidate_commit)}</code>
   252	          </dd>
   253	        </div>
   254	        <div>
   255	          <dt>Diff the {several ? 'reviewers' : 'reviewer'} saw</dt>
   256	          <dd data-testid="review-diff">
   257	            {review.diff === null ? 'No diff artifact was recorded.' : diffScoped && diffHref !== null ? (
   258	              <>
   259	                <a href={diffHref} target="_blank" rel="noopener noreferrer">Open {review.diff.artifact_id}</a>
   260	                {' '}<span className="projects-muted">(plain text, new tab) · sha256 {review.diff.sha256.slice(0, 12)}…</span>
   261	              </>
   262	            ) : (
   263	              <span className="projects-error-inline" role="alert">The diff link <code>{review.diff.uri}</code> is outside this run's artifact route and was not linked.</span>
   264	            )}
   265	          </dd>
   266	        </div>
   267	      </dl>
   268	
   269	      <ReviewerStrip reviewers={reviewers} />
   270	
   271	      <section className="evidence-section" aria-labelledby="review-findings-title" data-testid="review-findings" data-group-by={groupBy} data-reviewer-filter={reviewerFilter ?? 'all'}>
   272	        <div className="review-findings-head">
   273	          <h4 id="review-findings-title">Findings</h4>

   276	              <div className="review-group-toggle" role="group" aria-label="Show findings from">
   277	                <span className="projects-muted">Reviewer</span>

   279	                {reviewers.map(reviewer => (
   280	                  <button
   281	                    key={reviewer.reviewer_id}
   282	                    type="button"
   283	                    className="button button-small"
   284	                    aria-pressed={reviewerFilter === reviewer.reviewer_id}
   285	                    data-testid="filter-reviewer"

   289	                    {reviewer.reviewer_id} ({countFor(reviewer.reviewer_id)})

   292	              </div>

   294	                <span className="projects-muted">Group by</span>

Workers are the lanes this run had{lanes.length > 0 ? ` (${lanes.join(', ')})` : ''}; “multiple workers” covers findings that concern more than one lane.    325	        </p>

   331	/**
   332	 * The Result section of a review node: the persisted combined verdict, every reviewer's identity and outcome, the bundle,
   333	 * the unioned findings and the diff. A run whose export predates review results (404 REVIEW_NOT_FOUND) is a "not recorded"
   334	 * state, not an error.
   335	 */
   336	export function ReviewPanel({ scope, node, definitionNodes, snapshotNodes, inputs, refreshToken, onNavigate, onOpenRequirement, onOpenFile, onTransport }: Props) {
   337	  // The snapshot links the recorded review; an older export has no link, so the first attempt is asked for once the node ran.
   338	  const reviewPath = node.result_uri !== null
   339	    ? scopedReviewPath(scope, node.result_uri)
   340	    : node.status === 'pending' ? null : paths.review(scope, Math.max(node.attempt, 1))
   341	  const unscoped = node.result_uri !== null && reviewPath === null
   342	  const load = useCallback((signal: AbortSignal) => fetchReviewResult(scope, reviewPath!, signal), [scope, reviewPath])
   343	  // The fallback path before a review is recorded equals the linked one after, so the key also names what the node shows.
   344	  const { state, reload } = useResource(reviewPath === null ? null : `${reviewPath}|${node.result_uri ?? node.status}`, load, refreshToken)
   345	  const none = (
   346	    <p className="projects-muted" data-testid="review-none">
   347	      No review recorded for this run: either the review has not happened or the run's export predates review results (re-export it with the workflow CLI).
   348	    </p>
   349	  )
   350	  if (unscoped) {
   351	    return (
   352	      <p className="projects-error-inline" role="alert" data-testid="result-unscoped">
   353	        The review link <code>{node.result_uri}</code> is outside this run's reviews route and was not fetched.
   354	      </p>
   355	    )
   356	  }
   357	  if (reviewPath === null) return none
   358	  if (state.status === 'loading' || state.status === 'idle') return <LoadingPanel>Loading the recorded review…</LoadingPanel>
   359	  if (state.status === 'error') {
   360	    if (isNotRecorded(state.error, NOT_RECORDED.review)) return none
   361	    return <ErrorPanel error={state.error} what="The recorded review" onRetry={reload} />
   362	  }
   363	  return (
   364	    <ReviewResultView
   365	      review={state.data}
   366	      scope={scope}
   367	      definitionNodes={definitionNodes}
   368	      snapshotNodes={snapshotNodes}
   369	      inputs={inputs}
   370	      refreshToken={refreshToken}
   371	      onNavigate={onNavigate}
   372	      onOpenRequirement={onOpenRequirement}
   373	      onOpenFile={onOpenFile}
   374	      onTransport={onTransport}
   375	    />
   376	  )
   377	}

   402	/**
   403	 * The design challenge node's page (PRD_PORTABLE_WORKFLOW 4.5 and 4.8): the outcome, the concerns grouped by severity with
   404	 * their consequences, the strongest simpler alternative, the cheap experiment, the attempts and the operator's accepted
   405	 * reason. It is read from the run inputs, which pin the latest `challenge.json`.
   406	 */
   407	export function ChallengePanel({ inputs, onRetry }: { inputs: Resource<RunInputs | null>; onRetry: () => void }) {
   408	  if (inputs.status === 'loading' || inputs.status === 'idle') return <LoadingPanel>Loading the design challenge…</LoadingPanel>
   409	  if (inputs.status === 'error') return <ErrorPanel error={inputs.error} what="The run inputs" onRetry={onRetry} />
   410	  const challenge = inputs.data?.challenge ?? null
   411	  if (challenge === null) {
   412	    return <p className="projects-muted" data-testid="challenge-none">No design challenge was recorded for this run.</p>
   413	  }

   611	/** A failed request: what failed, why, and what the reader can do. Never replaced by sample data. */
   612	export function ErrorPanel({ error, what, onRetry, children }: { error: unknown; what: string; onRetry?: () => void; children?: ReactNode }) {
   613	  const api = error instanceof ProjectsApiError ? error : null
   614	  return (
   615	    <div className="projects-error" role="alert" data-testid="projects-error" data-error-kind={api?.kind ?? 'unknown'}>
   616	      <p><strong>{what} could not be loaded.</strong> {describeApiError(error)}</p>
   617	      {api?.kind === 'http' && api.status !== null && api.status >= 500 && <p>The server reported a failure; nothing was substituted for the missing data. Retry once it has recovered.</p>}
   618	      {api?.kind === 'malformed' && <p>The response was rejected rather than rendered, because it might be incomplete or from an incompatible server version.</p>}
   619	      {api && <p className="projects-error-path">Request: <code>{api.path}</code></p>}
   620	      <div className="projects-actions">
   621	        {onRetry && <button type="button" className="button" onClick={onRetry}>Retry</button>}
   622	        {children}
   623	      </div>
   624	    </div>
   625	  )
   626	}

   637	type AppLinkProps = Omit<AnchorHTMLAttributes<HTMLAnchorElement>, 'href' | 'onClick'> & {
   638	  href: string
   639	  onNavigate: (pathname: string) => void
   640	  children: ReactNode
   641	  current?: boolean
   642	}
   643	
   644	/** A link that navigates inside the app through history rather than reloading. Other anchor attributes (title, data-*) pass through. */
   645	export function AppLink({ href, onNavigate, className, children, current = false, ...rest }: AppLinkProps) {
   646	  return (
   647	    <a
   648	      {...rest}
   649	      href={href}
   650	      className={className}
   651	      aria-current={current ? 'page' : undefined}
   652	      onClick={event => {
   653	        if (event.metaKey || event.ctrlKey || event.shiftKey || event.altKey || event.button !== 0) return
   654	        event.preventDefault()
   655	        onNavigate(href)
   656	      }}
   657	    >
   658	      {children}
   659	    </a>
   660	  )
   661	}
