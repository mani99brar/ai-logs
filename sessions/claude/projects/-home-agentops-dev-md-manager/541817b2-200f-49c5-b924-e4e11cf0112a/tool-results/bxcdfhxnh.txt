    22	/**
    23	 * The pinned task of a worker: rendered Markdown or the exact source, the ownership boundary and the
    24	 * required checks. When a finding link hands over a quote, the panel opens on the source with the quote
    25	 * marked and scrolled into view. The quote is captured when the panel mounts (the node detail remounts per
    26	 * node), so clearing the pending hand-off afterwards does not remove the mark until the reader leaves.
    27	 */
    28	export function TaskPanel({ worker, result, highlight, onHighlightApplied, checksNode, onNavigate }: TaskProps) {
    29	  const [view, setView] = useState<'rendered' | 'source'>(() => (highlight === null ? 'rendered' : 'source'))
    30	  const [activeQuote] = useState(() => highlight)
    31	  const markRef = useRef<HTMLElement>(null)
    32	  useEffect(() => {
    33	    if (activeQuote === null) return
    34	    const stop = markRef.current ? keepInView(markRef.current, { block: 'center' }) : undefined
    35	    onHighlightApplied()
    36	    return stop
    37	  }, [activeQuote, onHighlightApplied])
    38	
    39	  const text = worker.task.text
    40	  const index = activeQuote === null ? -1 : text.indexOf(activeQuote)
    41	  const executed = result.status === 'ready' ? result.data.checks : null
    42	
    43	  return (
    44	    <section className="evidence-section task-panel" aria-labelledby="task-title" data-testid="task-panel">
    45	      <div className="task-head">
    46	        <h4 id="task-title">Task</h4>
    47	        <div role="group" aria-label="Task view" className="task-toggle">
    48	          <button type="button" className="button button-small" aria-pressed={view === 'rendered'} onClick={() => setView('rendered')}>Rendered</button>
    49	          <button type="button" className="button button-small" aria-pressed={view === 'source'} onClick={() => setView('source')}>Source</button>
    50	        </div>
    51	      </div>
    52	      <p className="projects-muted">The assignment pinned in the run plan: the authored task plus the appended ownership and checks.</p>
    53	      {worker.task.truncated && (
    54	        <p className="projects-notice" role="note" data-testid="task-truncated">The task text was truncated by the viewer API; the marker at its end says how many characters were left out.</p>
    55	      )}
    56	      {activeQuote !== null && index === -1 && (
    57	        <p className="projects-muted" data-testid="task-highlight-missing">The quoted requirement was not found in this task text, so nothing is highlighted.</p>
    58	      )}
    59	      {view === 'rendered' ? (
    60	        <div className="task-rendered"><Markdown content={text} inert /></div>
    61	      ) : (
    62	        <pre className="task-source" data-testid="task-source" tabIndex={0}>
    63	          {index === -1 ? text : (
    64	            <>
    65	              {text.slice(0, index)}
    66	              <mark ref={markRef} className="task-highlight" data-testid="task-highlight">{activeQuote}</mark>
    67	              {text.slice(index + activeQuote!.length)}
    68	            </>
    69	          )}
    70	        </pre>
    71	      )}
    72	
    73	      <h5 id="task-owned-title">Owned paths</h5>
    74	      {worker.owned_paths.length === 0 ? (
    75	        <p className="projects-muted">No owned paths were pinned.</p>
    76	      ) : (
    77	        <ul className="evidence-list evidence-files" data-testid="task-owned-paths" aria-labelledby="task-owned-title">
    78	          {worker.owned_paths.map(path => <li key={path}><code>{path}</code></li>)}
    79	        </ul>
    80	      )}
    81	
    82	      <h5 id="task-checks-title">Required checks</h5>
    83	      <p className="projects-muted" data-testid="task-required-kinds">
    84	        Required check kinds for this lane: {worker.required_check_kinds.length === 0 ? 'none pinned' : worker.required_check_kinds.join(', ')}
    85	      </p>
    86	      {worker.checks.length === 0 ? (
    87	        <p className="projects-muted">No checks were required.</p>
    88	      ) : (
    89	        <ul className="evidence-list task-checks" data-testid="task-checks" aria-labelledby="task-checks-title">
    90	          {worker.checks.map(check => {
    91	            const executedIndex = executed === null ? -1 : executed.findIndex(candidate => candidate.command === check.command)
    92	            return (
    93	              <li key={check.id} data-check-id={check.id}>
    94	                <code>{check.command}</code>
    95	                <span className="projects-muted"> · {check.id} · {check.kind} · timeout {formatDuration(check.timeout_seconds)}</span>
    96	                {check.scenarios.length > 0 && (
    97	                  <span className="projects-muted"> · {check.scenarios.length} {check.scenarios.length === 1 ? 'scenario' : 'scenarios'}: {check.scenarios.map(scenario => scenario.id).join(', ')}</span>
    98	                )}
    99	                <div className="task-check-status">
   100	                  {executed === null ? (
   101	                    <span className="projects-muted">{result.status === 'loading' ? 'Loading the published result to compare…' : 'No published result to compare against.'}</span>
   102	                  ) : executedIndex === -1 ? (
   103	                    <span className="projects-muted">not executed in this result</span>

   161	/**
   162	 * The falsifying check the worker named: a declared check id links to the verify node that shows that check executed; a
   163	 * command equal to a declared check's command is treated the same; anything else is shown as the worker wrote it.
   164	 */
   165	function FalsifyingCheck({ value, worker, result, checksNode, onNavigate }: {
   166	  value: string
   167	  worker: RunInputWorker
   168	  result: Resource<WorkerResult>
   169	  checksNode: CheckTarget | null
   170	  onNavigate: (pathname: string) => void
   171	}) {
   172	  const check = worker.checks.find(candidate => candidate.id === value) ?? worker.checks.find(candidate => candidate.command === value) ?? null
   173	  if (check === null) {
   174	    return <span data-testid="falsifying-check-text"><code>{value}</code> <span className="projects-muted">(not a check id declared for this lane)</span></span>
   175	  }
   176	  const executed = result.status === 'ready' ? result.data.checks.findIndex(candidate => candidate.command === check.command) : -1
   177	  const where = executed === -1 ? '' : `, executed as check ${executed + 1}`
   178	  const label = <><code>{check.id}</code> · <code>{check.command}</code></>
   179	  if (checksNode === null) return <span data-testid="falsifying-check-text" data-check-id={check.id}>{label}</span>
   180	  return (
   181	    <AppLink
   182	      href={checksNode.href}
   183	      onNavigate={onNavigate}
   184	      data-testid="falsifying-check-link"
   185	      data-check-id={check.id}
   186	      data-check-index={executed === -1 ? undefined : executed}
   187	      title={`The executed check and its log are shown on ${checksNode.label}`}
   188	    >
   189	      {label} (on {checksNode.label}{where})
   190	    </AppLink>
   191	  )
   192	}

   194	/**
   195	 * The evidence a 1.1.0 completion carries (PRD_PORTABLE_WORKFLOW 4.6): what no executed check covers, the check that would
   196	 * fail if the implementation were wrong, and one assumption to verify independently. A 1.0.0 completion serves all three
   197	 * as null, which is stated rather than shown as empty; so is a 1.1.0 `blocked` one that recorded none. A question the
   198	 * controller has not recorded is shown with its text: still pending, or a fourth one the controller treated as blocked.
   199	 */
   200	function CompletionEvidence({ completion, worker, result, checksNode, onNavigate }: {
   201	  completion: Completion
   202	  worker: RunInputWorker
   203	  result: Resource<WorkerResult>
   204	  checksNode: CheckTarget | null
   205	  onNavigate: (pathname: string) => void
   206	}) {
   207	  const { untested, falsifying_check: falsifying, verify_yourself: verify } = completion
   208	  if (completion.question !== null) {
   209	    const pending = completion.status === 'question'
   210	    return (
   211	      <div data-testid={pending ? 'completion-question-pending' : 'completion-question-refused'}>
   212	        <p className={pending ? 'projects-muted' : 'projects-notice'} role={pending ? undefined : 'note'}>
   213	          {pending
   214	            ? 'No completion evidence yet: the worker ended its turn with this question, which the controller has not recorded as a question to the operator yet:'
   215	            : 'Treated as blocked: the worker asked a fourth question, and at most three are answered, so the controller blocked this lane. The question:'}
   216	        </p>
   217	        <p className="worker-question-text" data-testid="completion-question-text">{completion.question}</p>
   218	      </div>
   219	    )
   220	  }
   221	  if (completion.version === '1.0.0') {
   222	    return (
   223	      <p className="projects-muted" data-testid="completion-evidence-none">
   224	        Completion evidence was not recorded for this run: its completion predates the untested, falsifying-check and verify-yourself fields.
   225	      </p>
   226	    )
   227	  }
   228	  if (completion.status === 'blocked' && untested === null && falsifying === null && verify === null) {
   229	    return (
   230	      <p className="projects-muted" data-testid="completion-evidence-blocked">
   231	        No completion evidence recorded: the worker blocked, and a blocked completion need not carry it.
   232	      </p>
   233	    )
   234	  }
   235	  return (
   236	    <dl className="projects-facts completion-evidence" data-testid="completion-evidence">
   237	      <div data-testid="evidence-untested">
   238	        <dt>Untested</dt>
   239	        <dd>
   240	          {untested === null ? <span className="projects-muted">Not recorded</span>
   241	            : untested.length === 0 ? <span className="projects-muted">Nothing: the worker names no behaviour outside its executed checks.</span>
   242	              : <ul className="evidence-list">{untested.map((item, index) => <li key={index}>{item}</li>)}</ul>}
   243	        </dd>
   244	      </div>
   245	      <div data-testid="evidence-falsifying-check">
   246	        <dt>Falsifying check</dt>
   247	        <dd>
   248	          {falsifying === null || falsifying === ''
   249	            ? <span className="projects-muted">Not recorded</span>
   250	            : <FalsifyingCheck value={falsifying} worker={worker} result={result} checksNode={checksNode} onNavigate={onNavigate} />}
   251	        </dd>
   252	      </div>
   253	      <div data-testid="evidence-verify-yourself">
   254	        <dt>Verify yourself</dt>
   255	        <dd>{verify === null || verify === '' ? <span className="projects-muted">Not recorded</span> : verify}</dd>
   256	      </div>
   257	    </dl>
   258	  )
   259	}

   301	/**
   302	 * The worker's own completion signal with its evidence, and the accepted handoff only when it differs from it. `worker`,
   303	 * `result` and `checksNode` let the falsifying check link to the executed check.
   304	 */
   305	export function WorkerSignals({ completion, handoff, worker, result, checksNode, onNavigate }: {
   306	  completion: RunInputWorker['completion']
   307	  handoff: RunInputWorker['handoff']
   308	  worker: RunInputWorker
   309	  result: Resource<WorkerResult>
   310	  checksNode: CheckTarget | null
   311	  onNavigate: (pathname: string) => void
   312	}) {
   313	  const handoffDiffers = handoff !== null && (
   314	    completion === null || handoff.summary !== completion.summary || JSON.stringify(handoff.open_assumptions) !== JSON.stringify(completion.open_assumptions)
   315	  )
   316	  return (
   317	    <>
   318	      <section className="evidence-section" aria-labelledby="worker-completion-title" data-testid="worker-completion">
   319	        <h4 id="worker-completion-title">Reported by the worker</h4>
   320	        {completion === null ? (
   321	          <p className="projects-muted">No completion signal recorded.</p>
   322	        ) : (
   323	          <>
   324	            <p>
   325	              <span className={`status-badge ${COMPLETION_BADGE[completion.status]}`} data-status={completion.status}><span>{completion.status}</span></span>
   326	              {' '}<span className="projects-muted">as signalled by the session itself, not a verified result.</span>
   327	            </p>
   328	            <p className="worker-summary">{completion.summary}</p>
   329	            <Assumptions items={completion.open_assumptions} />
   330	            <CompletionEvidence completion={completion} worker={worker} result={result} checksNode={checksNode} onNavigate={onNavigate} />
   331	          </>
   332	        )}
   333	      </section>
   334	      {handoffDiffers && (
   335	        <section className="evidence-section" aria-labelledby="worker-handoff-title" data-testid="worker-handoff">
   336	          <h4 id="worker-handoff-title">Handoff as accepted</h4>
   337	          <p className="projects-muted">The accepted handoff differs from what the worker reported.</p>
   338	          <p className="worker-summary">{handoff.summary}</p>
   339	          <Assumptions items={handoff.open_assumptions} />
   340	        </section>
   341	      )}
   342	    </>
   343	  )
   344	}

   401	/** The review findings that name this file: severity, reviewer, disposition and message, and a control to mark the lines a finding names. */
   402	function FileFindings({ findings, path, onShowLines }: { findings: Resource<ReviewFinding[] | null>; path: string; onShowLines: (ranges: Range[]) => void }) {
   403	  if (findings.status === 'idle') return <p className="projects-muted" data-testid="file-findings-no-review">No review is recorded for this run, so no findings are shown on its files.</p>
   404	  if (findings.status === 'loading') return <LoadingPanel>Loading the review findings…</LoadingPanel>
   405	  if (findings.status === 'error') return <p className="projects-error-inline" role="alert">The recorded review could not be loaded, so findings on this file are not shown.</p>
   406	  if (findings.data === null) return <p className="projects-muted" data-testid="file-findings-no-review">No review is recorded for this run, so no findings are shown on its files.</p>
   407	  const named = findingsForFile(findings.data, path)
   408	  if (named.length === 0) return <p className="projects-muted" data-testid="file-findings-none">No review finding names this file.</p>
   409	  return (
   410	    <ul className="evidence-list" data-testid="file-findings">
   411	      {named.map((finding, index) => {
   412	        const ranges = linesNamed(finding.message, path)
   413	        return (
   414	          <li key={index} data-testid="file-finding" data-severity={finding.severity} data-reviewer={finding.reviewer} data-disposition={finding.disposition}>
   415	            <span className="finding-severity">{finding.severity}</span>
   416	            {' '}· <span data-testid="file-finding-reviewer">{finding.reviewer}</span> · {finding.disposition} — {finding.message}
   417	            {ranges.length > 0 && (
   418	              <>
   419	                {' '}
   420	                <button type="button" className="button button-small" data-testid="show-lines" onClick={() => onShowLines(ranges)}>
   421	                  Show {ranges.map(rangeText).join(', ')}
   422	                </button>
   423	              </>
   424	            )}
   425	          </li>
   426	        )
   427	      })}
   428	    </ul>
   429	  )

   432	/** The exact file content, one element per line; lines a finding names are marked. */
   433	function FileSource({ text, marked, preRef }: { text: string; marked: Range[]; preRef: RefObject<HTMLPreElement | null> }) {
   434	  const lines = text.endsWith('\n') ? text.slice(0, -1).split('\n') : text.split('\n')
   435	  const isMarked = (line: number) => marked.some(([from, to]) => line >= from && line <= to)
   436	  return (

   518	/**
   519	 * "Files created or changed" on a launch node: every changed file of the result in order, captured ones with their content
   520	 * and the findings that name them, the others with the reason they were not captured. A result recorded before capture
   521	 * lists the paths and says the files were not captured.

   523	export function CreatedFiles({ scope, result, reviewNode, refreshToken, focusPath, onFocusApplied }: Props) {
   524	  const captured = capturedFiles(result)
   525	  const byPath = new Map(captured.map(file => [file.path, file]))
   526	  const reasons = new Map(notCapturedFiles(result).map(entry => [entry.path, entry.reason]))
   527	  const recorded = recordsCapture(result)
   528	  // Every changed file in the result's order, then any captured or listed path the changed files do not name (never dropped).
   529	  const listed = [...result.changed_files]
   530	  for (const path of [...byPath.keys(), ...reasons.keys()]) if (!listed.includes(path)) listed.push(path)
   531	
   532	  const reviewPath = captured.length === 0 ? null : reviewPathOf(scope, reviewNode)
   533	  const loadReview = useCallback(
   534	    (signal: AbortSignal) => orNotRecorded(fetchReviewResult(scope, reviewPath!, signal), NOT_RECORDED.review).then(review => review?.findings ?? null),
   535	    [scope, reviewPath],
   536	  )
   537	  const { state: findings } = useResource(reviewPath === null ? null : `${reviewPath}|${reviewNode?.result_uri ?? reviewNode?.status}`, loadReview, refreshToken)
   538	
   539	  // A file link is applied once: the target panel scrolls into view when it mounts, and the run view forgets the hand-over.
   540	  const [focus] = useState(() => focusPath)
   541	  useEffect(() => {
   542	    if (focus !== null) onFocusApplied()
   543	  }, [focus, onFocusApplied])
   544	
   545	  return (
   546	    <section className="evidence-section" aria-labelledby="evidence-created-files" data-testid="created-files" data-captured={captured.length}>
   547	      <h4 id="evidence-created-files">Files created or changed</h4>
   548	      {listed.length === 0 ? (
   549	        <p className="projects-muted" data-testid="changed-files-empty">No changed files were recorded.</p>
   550	      ) : (
   551	        <>
   552	          {!recorded && (
   553	            <p className="projects-muted" data-testid="files-not-captured">
   554	              Created files were not captured for this run (its results predate file capture), so only their paths are listed.
   555	            </p>
   556	          )}
   557	          <ul className="evidence-list evidence-files" data-testid="changed-files">
   558	            {listed.map(path => {
   559	              const file = byPath.get(path)
   560	              const reason = reasons.get(path)
   561	              const state = file ? 'captured' : reason ? 'not-captured' : recorded ? 'unlisted' : 'not-recorded'
   562	              return (
   563	                <li key={path} data-testid="created-file" data-path={path} data-state={state} data-reason={reason}>
   564	                  {file ? (
   565	                    <CapturedFilePanel scope={scope} file={file} findings={findings} focused={focus === path} />
   566	                  ) : (
   567	                    <>
   568	                      <code>{path}</code>
   569	                      {reason && <span className="projects-muted" data-testid="not-captured-reason"> — {NOT_CAPTURED_WORDING[reason]}</span>}
   570	                      {!reason && recorded && <span className="projects-muted"> — not captured, and no reason was recorded</span>}
   571	                    </>
   572	                  )}
   573	                </li>
   574	              )
   575	            })}
   576	          </ul>
   577	          {focus !== null && !byPath.has(focus) && (
   578	            <p className="projects-muted" data-testid="file-focus-missing">The linked file <code>{focus}</code> is not among this node's captured files.</p>
   579	          )}
   580	        </>
   581	      )}
   582	    </section>
   583	  )
   584	}

   639	export function describeApiError(error: unknown): string {
   640	  if (error instanceof ProjectsApiError) {
   641	    switch (error.kind) {
   642	      case 'network': return 'The API could not be reached. Check that the server is running, then retry.'
   643	      case 'malformed': return `The API returned data that does not match the projects contract: ${error.message}`
   644	      case 'http': return error.code ? `${error.message} (${error.code}, HTTP ${error.status})` : `${error.message} (HTTP ${error.status})`

   646	  }
   647	  return error instanceof Error && error.message ? error.message : 'An unexpected error occurred.'

   676	async function request(path: string, signal: AbortSignal | undefined, accept: string): Promise<Response> {
   677	  let response: Response
   678	  try {
   679	    response = await fetch(path, { signal, headers: { Accept: accept } })
   680	  } catch (error) {
   681	    if (signal?.aborted) throw error
   682	    throw new ProjectsApiError('network', error instanceof Error ? error.message : 'Network failure', path)
   683	  }
   684	  if (!response.ok) {
   685	    let body: unknown = null
   686	    try { body = await response.json() } catch { /* not JSON */ }
   687	    const parsed = errorBodySchema.safeParse(body)
   688	    const message = parsed.success ? parsed.data.error.message : `The API responded with status ${response.status}.`
   689	    throw new ProjectsApiError('http', message, path, { status: response.status, code: parsed.success ? parsed.data.error.code : null })
   690	  }
   691	  return response
   692	}
