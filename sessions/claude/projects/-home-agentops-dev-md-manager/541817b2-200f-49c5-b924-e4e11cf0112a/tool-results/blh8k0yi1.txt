   221	/**
   222	 * Projects routes: `/projects[/<project>[/workflows/<workflow>[/runs/<run>[/nodes/<node>]]]]`.
   223	 *
   224	 * Every ID is an opaque contract identifier: it must match the shared ID pattern and is percent-encoded
   225	 * as one path segment. Anything else is malformed rather than looked up. This domain is read-only and
   226	 * separate from the Pi/Claude skill sources; nothing here names a filesystem path.
   227	 */
   228	
   229	/** Same pattern as the shared projects contract (`contracts/projects/v1.ts`). */
   230	export const PROJECT_ID_PATTERN = /^[A-Za-z0-9][A-Za-z0-9._-]{0,127}$/
   231	
   232	export const PROJECTS_ROUTE = 'projects'
   233	
   234	export type ProjectsRoute =
   235	  | { level: 'projects' }
   236	  | { level: 'project'; projectId: string }
   237	  | { level: 'workflow'; projectId: string; workflowId: string }
   238	  | { level: 'run'; projectId: string; workflowId: string; runId: string; nodeId: string | null }
   239	
   240	export function isContractId(value: string): boolean {
   241	  return PROJECT_ID_PATTERN.test(value)
   242	}
   243	
   244	function encode(segments: string[]): string {
   245	  return `/${segments.map(encodeURIComponent).join('/')}`
   246	}
   247	
   248	export function projectsPathname(): string {
   249	  return `/${PROJECTS_ROUTE}`
   250	}
   251	
   252	export function projectPathname(projectId: string): string {
   253	  return encode([PROJECTS_ROUTE, projectId])
   254	}
   255	
   256	export function workflowPathname(projectId: string, workflowId: string): string {
   257	  return encode([PROJECTS_ROUTE, projectId, 'workflows', workflowId])
   258	}
   259	
   260	export function runPathname(projectId: string, workflowId: string, runId: string, nodeId: string | null = null): string {
   261	  const segments = [PROJECTS_ROUTE, projectId, 'workflows', workflowId, 'runs', runId]
   262	  if (nodeId !== null) segments.push('nodes', nodeId)
   263	  return encode(segments)
   264	}
   265	
   266	export function routeToPathname(route: ProjectsRoute): string {
   267	  switch (route.level) {
   268	    case 'projects': return projectsPathname()
   269	    case 'project': return projectPathname(route.projectId)
   270	    case 'workflow': return workflowPathname(route.projectId, route.workflowId)
   271	    case 'run': return runPathname(route.projectId, route.workflowId, route.runId, route.nodeId)
   272	  }
   273	}

   275	/** Whether a pathname belongs to the Projects domain at all (decided before any decoding). */
   276	export function isProjectsPathname(pathname: string): boolean {
   277	  const first = pathname.split('/').filter(segment => segment !== '')[0]
   278	  return first === PROJECTS_ROUTE
   279	}
   280	
   281	/**
   282	 * Parses a Projects pathname. Returns null when it is not a Projects pathname or when any segment is
   283	 * malformed (bad encoding, an invalid ID, an unexpected literal or trailing segments).
   284	 */
   285	export function parseProjectsPathname(pathname: string): ProjectsRoute | null {
   286	  const raw = pathname.split('/').filter(segment => segment !== '')
   287	  if (raw[0] !== PROJECTS_ROUTE) return null
   288	  let segments: string[]
   289	  try {
   290	    segments = raw.slice(1).map(decodeURIComponent)
   291	  } catch {
   292	    return null
   293	  }
   294	  if (segments.length === 0) return { level: 'projects' }
   295	  const [projectId, workflowsLiteral, workflowId, runsLiteral, runId, nodesLiteral, nodeId, ...rest] = segments
   296	  if (!isContractId(projectId)) return null
   297	  if (workflowsLiteral === undefined) return { level: 'project', projectId }
   298	  if (workflowsLiteral !== 'workflows' || workflowId === undefined || !isContractId(workflowId)) return null
   299	  if (runsLiteral === undefined) return { level: 'workflow', projectId, workflowId }
   300	  if (runsLiteral !== 'runs' || runId === undefined || !isContractId(runId)) return null
   301	  if (nodesLiteral === undefined) return { level: 'run', projectId, workflowId, runId, nodeId: null }
   302	  if (nodesLiteral !== 'nodes' || nodeId === undefined || !isContractId(nodeId) || rest.length > 0) return null
   303	  return { level: 'run', projectId, workflowId, runId, nodeId }
   304	}

   324	/**
   325	 * The Projects root: registered projects, their workflow definitions and runs, read-only. Every level
   326	 * loads from the scoped API and shows loading, empty, not-found and failure states in place; nothing
   327	 * is ever substituted from fixtures.
   328	 */
   329	export function ProjectsView({ route, refreshToken, onNavigate, onAnnounce }: Props) {
   330	  const projectId = route && route.level !== 'projects' ? route.projectId : null
   331	  const workflowId = route && (route.level === 'workflow' || route.level === 'run') ? route.workflowId : null
   332	  const runId = route && route.level === 'run' ? route.runId : null
   333	  const nodeId = route && route.level === 'run' ? route.nodeId : null
   334	
   335	  // The registry itself changes while the viewer runs (a launch registers its workflow); both lists are cheap to re-read.
   336	  const registryPoll = usePoll(route?.level === 'projects' || route?.level === 'project')
   337	  const { state: projects, reload: reloadProjects } = useResource('projects', fetchProjects, refreshToken, registryPoll)
   338	  const loadWorkflows = useCallback((signal: AbortSignal) => fetchWorkflows(projectId!, signal), [projectId])
   339	  const { state: workflows, reload: reloadWorkflows } = useResource(projectId === null ? null : `workflows:${projectId}`, loadWorkflows, refreshToken, registryPoll)
   340	  const loadRuns = useCallback((signal: AbortSignal) => fetchRuns(projectId!, workflowId!, {}, signal), [projectId, workflowId])
   341	  const runsKey = workflowId === null ? null : `runs:${projectId}/${workflowId}`
   342	  // Run lists and run details re-read themselves while shown (a finished run no longer changes). A list stops polling
   343	  // while it loads or shows appended pages, since a new first page would drop them.
   344	  const [morePages, setMorePages] = useState<{ base: RunPage | null; pages: RunPage[]; loading: boolean; error: unknown }>({ base: null, pages: [], loading: false, error: null })
   345	  const [finished, setFinished] = useState(false)
   346	  // Appended pages belong to one list generation (this workflow, this Refresh); a new one starts without them.
   347	  const listGeneration = `${runsKey}\u0000${refreshToken}`
   348	  useEffect(() => { setMorePages({ base: null, pages: [], loading: false, error: null }) }, [listGeneration])
   349	  const listPoll = usePoll(route?.level === 'workflow' && !morePages.loading && morePages.pages.length === 0)
   350	  const runPoll = usePoll(route?.level === 'run' && !finished)
   351	  const { state: firstPage, reload: reloadRuns } = useResource(runsKey, loadRuns, refreshToken, listPoll)
   352	  const scope = useMemo(() => (projectId === null || workflowId === null || runId === null ? null : { projectId, workflowId, runId }), [projectId, workflowId, runId])
   353	  const loadDetail = useCallback((signal: AbortSignal) => fetchRunDetail({ projectId: projectId!, workflowId: workflowId!, runId: runId! }, signal), [projectId, workflowId, runId])
   354	  const { state: detail, reload: reloadDetail } = useResource(scope === null ? null : `run:${scope.projectId}/${scope.workflowId}/${scope.runId}`, loadDetail, refreshToken, runPoll)
   355	  const detailStatus = detail.status === 'ready' ? detail.data.summary.status : null
   356	  useEffect(() => { setFinished(detailStatus === 'succeeded' || detailStatus === 'cancelled') }, [detailStatus])
   357	
   358	  // Additional run pages are appended on demand; they belong to exactly one loaded first page and are dropped with it.
   359	  const firstPageData = firstPage.status === 'ready' ? firstPage.data : null
   360	  const extra = morePages.base !== null && morePages.base === firstPageData ? morePages : { base: firstPageData, pages: [], loading: false, error: null }
   361	  const pages: RunPage[] = firstPageData ? [firstPageData, ...extra.pages] : []
   362	  const runs: RunSummary[] = pages.flatMap(page => page.runs)
   363	  const nextCursor = pages.length ? pages[pages.length - 1].nextCursor : null
   364	  const loadMore = async () => {
   365	    if (!nextCursor || projectId === null || workflowId === null || firstPageData === null) return
   366	    setMorePages({ ...extra, base: firstPageData, loading: true, error: null })
   367	    try {
   368	      const page = await fetchRuns(projectId, workflowId, { cursor: nextCursor })
   369	      setMorePages(previous => (previous.base === firstPageData ? { ...previous, pages: [...previous.pages, page], loading: false } : previous))
   370	    } catch (error) {
   371	      setMorePages(previous => (previous.base === firstPageData ? { ...previous, loading: false, error } : previous))
   372	    }
   373	  }
   374	
   375	  const projectName = projects.status === 'ready' && projectId !== null ? projects.data.find(project => project.project_id === projectId)?.name ?? projectId : projectId
   376	  const currentWorkflow = workflows.status === 'ready' && workflowId !== null ? workflows.data.find(workflow => workflow.workflow_id === workflowId) ?? null : null
   377	  const workflowName = currentWorkflow?.name ?? workflowId
   378	
   379	  // A background poll re-announces only what changed; a load the reader started (navigation, Refresh) always announces.
   380	  const announced = useRef<string | null>(null)
   381	  useEffect(() => {
   382	    let message: string | null = null
   383	    if (route === null) message = 'This Projects link is invalid.'
   384	    else if (route.level === 'projects' && projects.status === 'ready') message = `Loaded ${projects.data.length} ${projects.data.length === 1 ? 'project' : 'projects'}.`
   385	    else if (route.level === 'project' && workflows.status === 'ready') message = `Loaded ${workflows.data.length} ${workflows.data.length === 1 ? 'workflow' : 'workflows'} for ${projectName ?? route.projectId}.`
   386	    else if (route.level === 'workflow' && firstPage.status === 'ready') message = `Loaded ${firstPage.data.runs.length} ${firstPage.data.runs.length === 1 ? 'run' : 'runs'} for ${workflowName ?? route.workflowId}.`
   387	    else if (route.level === 'run' && detail.status === 'ready') message = `Loaded run ${route.runId}: ${detail.data.summary.status.replace('_', ' ')}.`
   388	    else if ((projects.status === 'error') || workflows.status === 'error' || firstPage.status === 'error' || detail.status === 'error') message = 'Loading failed.'
   389	    if (message === null) { announced.current = null; return }
   390	    if (message === announced.current) return
   391	    announced.current = message
   392	    onAnnounce(message)
   393	  }, [route, projects, workflows, firstPage, detail, projectName, workflowName, onAnnounce])
   394	
   395	  const crumbs: PathCrumb[] = [{ id: 'home', label: 'Home', pathname: '/' }, { id: 'projects', label: 'Projects', pathname: projectsPathname() }]
   396	  if (projectId !== null) crumbs.push({ id: `project:${projectId}`, label: projectName ?? projectId, pathname: projectPathname(projectId) })
   397	  if (projectId !== null && workflowId !== null) crumbs.push({ id: `workflow:${workflowId}`, label: workflowName ?? workflowId, pathname: workflowPathname(projectId, workflowId) })
   398	  if (projectId !== null && workflowId !== null && runId !== null) crumbs.push({ id: `run:${runId}`, label: runId, pathname: runPathname(projectId, workflowId, runId) })
   399	  if (route === null) crumbs.push({ id: 'invalid', label: 'Invalid link' })
   400	
   401	  const notFound = (error: unknown) => error instanceof ProjectsApiError && error.notFound
   402	
   403	  let info: string
   404	  if (route === null) info = 'This Projects link is invalid.'
   405	  else if (route.level === 'projects') info = 'Registered projects. Read-only: runs are started, approved and retried from the workflow CLI, never from this page.'
   406	  else if (route.level === 'project') info = 'Workflow definitions registered for this project.'
   407	  else if (route.level === 'workflow') info = 'Runs of this workflow, newest first. Each run keeps the definition it was started with.'
   408	  else info = nodeId === null ? 'Run detail with its pinned definition graph.' : `Run detail, inspecting node ${nodeId}.`
   409	
   410	  const busy = projects.status === 'loading' || workflows.status === 'loading' || firstPage.status === 'loading' || detail.status === 'loading'
   411	
   412	  let content: React.ReactNode
   413	  if (route === null) {
   414	    content = (
   415	      <div className="projects-error" role="alert">
   416	        <p>This Projects link is invalid and could not be read. Links look like /projects/&lt;project&gt;/workflows/&lt;workflow&gt;/runs/&lt;run&gt;.</p>
   417	        <p className="projects-actions"><AppLink href={projectsPathname()} onNavigate={onNavigate}>Go to Projects</AppLink></p>
   418	      </div>
   419	    )
   420	  } else if (route.level === 'projects') {
   421	    if (projects.status === 'loading' || projects.status === 'idle') content = <LoadingPanel>Loading registered projects…</LoadingPanel>
   422	    else if (projects.status === 'error') content = <ErrorPanel error={projects.error} what="The project list" onRetry={reloadProjects} />
   423	    else if (projects.data.length === 0) {
   424	      content = (
   425	        <EmptyPanel title="No projects are registered." testId="empty-projects">
   426	          <p>The backend's project registry is empty. Projects are registered by the operator in the server configuration, not from this page.</p>
   427	        </EmptyPanel>
   428	      )
   429	    } else {
   430	      content = (
   431	        <section aria-labelledby="projects-title">
   432	          <h2 id="projects-title">Projects</h2>
   433	          <ul className="projects-list" data-testid="projects-list">
   434	            {projects.data.map(project => (
   435	              <li key={project.project_id}>
   436	                <AppLink href={projectPathname(project.project_id)} onNavigate={onNavigate} className="projects-card">
   437	                  <span className="projects-card-title">{project.name}</span>
   438	                  <span className="projects-muted">{project.project_id}</span>
   439	                </AppLink>
   440	              </li>
   441	            ))}
   442	          </ul>
   443	        </section>
   444	      )
   445	    }
   446	  } else if (route.level === 'project') {
   447	    if (workflows.status === 'loading' || workflows.status === 'idle') content = <LoadingPanel>Loading workflows of {projectName}…</LoadingPanel>
   448	    else if (workflows.status === 'error' && notFound(workflows.error)) {
   449	      content = (
   450	        <div className="projects-error" role="alert" data-testid="not-found">
   451	          <p><strong>Project “{route.projectId}” is not registered.</strong> The API has no project with this ID; it may have been removed from the registry or the link may be wrong.</p>
   452	          <p className="projects-actions"><AppLink href={projectsPathname()} onNavigate={onNavigate}>Go to Projects</AppLink></p>
   453	        </div>
   454	      )
   455	    } else if (workflows.status === 'error') content = <ErrorPanel error={workflows.error} what={`The workflows of ${projectName}`} onRetry={reloadWorkflows} />
   456	    else if (workflows.data.length === 0) {
   457	      content = (
   458	        <EmptyPanel title={`${projectName} has no workflow definitions.`} testId="empty-workflows">
   459	          <p>The project is registered but no workflow is configured for it, so there is nothing to run or inspect here.</p>
   460	        </EmptyPanel>
   461	      )
   462	    } else {
   463	      content = (
   464	        <section aria-labelledby="workflows-title">
   465	          <h2 id="workflows-title">{projectName}: workflows</h2>
   466	          <ul className="projects-list" data-testid="workflows-list">
   467	            {workflows.data.map(workflow => (
   468	              <li key={workflow.workflow_id}>
   469	                <AppLink href={workflowPathname(route.projectId, workflow.workflow_id)} onNavigate={onNavigate} className="projects-card">
   470	                  <span className="projects-card-title">{workflow.name}</span>
   471	                  <span className="projects-muted">{workflow.workflow_id} · {workflow.nodes.length} {workflow.nodes.length === 1 ? 'node' : 'nodes'} · current revision {shortRevision(workflow.definition_revision)}</span>
   472	                </AppLink>
   473	              </li>
   474	            ))}
   475	          </ul>
   476	        </section>
   477	      )
   478	    }
   479	  } else if (route.level === 'workflow') {
   480	    if (firstPage.status === 'loading' || firstPage.status === 'idle') content = <LoadingPanel>Loading runs of {workflowName}…</LoadingPanel>
   481	    else if (firstPage.status === 'error' && notFound(firstPage.error)) {
   482	      content = (
   483	        <div className="projects-error" role="alert" data-testid="not-found">
   484	          <p><strong>Workflow “{route.workflowId}” was not found in project “{route.projectId}”.</strong> Either the project is not registered or this workflow does not belong to it.</p>
   485	          <p className="projects-actions">
   486	            <AppLink href={projectPathname(route.projectId)} onNavigate={onNavigate}>Go to the project</AppLink>
   487	            <AppLink href={projectsPathname()} onNavigate={onNavigate}>Go to Projects</AppLink>
   488	          </p>
   489	        </div>
   490	      )
   491	    } else if (firstPage.status === 'error') content = <ErrorPanel error={firstPage.error} what={`The runs of ${workflowName}`} onRetry={reloadRuns} />
   492	    else {
   493	      content = (
   494	        <div className="workflow-view">
   495	          <section aria-labelledby="runs-title">
   496	            <h2 id="runs-title">{workflowName}: runs</h2>
   497	            {runs.length === 0 ? (
   498	              <EmptyPanel title="No runs have been recorded for this workflow." testId="empty-runs">
   499	                <p>The workflow definition exists (see its current graph below) but it has never been run, or its runs are stored elsewhere. Runs are started from the workflow CLI.</p>
   500	              </EmptyPanel>
   501	            ) : (
   502	              <ul className="projects-list run-list" data-testid="run-list">
   503	                {runs.map(run => (
   504	                  <li key={run.run_id} data-run-id={run.run_id} data-status={run.status}>
   505	                    <AppLink href={runPathname(route.projectId, route.workflowId, run.run_id)} onNavigate={onNavigate} className="projects-card">
   506	                      <span className="projects-card-title">{run.run_id} <StatusBadge status={run.status} explain /></span>
   507	                      <span className="projects-muted">updated {formatTime(run.updated_at)} · created {formatTime(run.created_at)} · pinned revision {shortRevision(run.definition_revision)}</span>
   508	                    </AppLink>
   509	                  </li>
   510	                ))}
   511	              </ul>
   512	            )}
   513	            {nextCursor && (
   514	              <div className="projects-actions">
   515	                <button type="button" className="button" onClick={() => void loadMore()} disabled={extra.loading} aria-busy={extra.loading}>
   516	                  {extra.loading ? 'Loading more runs…' : 'Load more runs'}
   517	                </button>
   518	              </div>
   519	            )}
   520	            {extra.error !== null && <ErrorPanel error={extra.error} what="The next page of runs" onRetry={() => void loadMore()} />}
   521	          </section>
   522	          <section aria-labelledby="definition-title" className="workflow-definition">
   523	            <h3 id="definition-title">Current definition</h3>
   524	            {workflows.status === 'loading' && <LoadingPanel>Loading the current definition…</LoadingPanel>}
   525	            {workflows.status === 'error' && <ErrorPanel error={workflows.error} what="The current definition" onRetry={reloadWorkflows} />}
   526	            {workflows.status === 'ready' && currentWorkflow === null && <p className="projects-error-inline" role="alert">The workflow list does not contain “{route.workflowId}”, so its current definition cannot be shown.</p>}
   527	            {currentWorkflow && (
   528	              <>
   529	                <p className="projects-muted" data-testid="current-definition">{currentWorkflow.name} · revision <code>{shortRevision(currentWorkflow.definition_revision)}</code> · {currentWorkflow.nodes.length} nodes. Runs started before a definition change keep their own pinned graph.</p>
   530	                <WorkflowGraph title={`Current definition graph of ${currentWorkflow.name}`} nodes={currentWorkflow.nodes} selectedId={null} />
   531	              </>
   532	            )}
   533	          </section>
   534	        </div>
   535	      )
   536	    }
   537	  } else {
   538	    const loading = detail.status === 'loading' || detail.status === 'idle'
   539	    if (loading) content = <LoadingPanel>Loading run {route.runId}…</LoadingPanel>
   540	    else if (detail.status === 'error' && notFound(detail.error)) {
   541	      content = (
   542	        <div className="projects-error" role="alert" data-testid="not-found">
   543	          <p><strong>Run “{route.runId}” was not found in workflow “{route.workflowId}” of project “{route.projectId}”.</strong> Runs are only served under the project and workflow they belong to; a run of another project is not shown here.</p>
   544	          <p className="projects-actions">
   545	            <AppLink href={workflowPathname(route.projectId, route.workflowId)} onNavigate={onNavigate}>Go to the workflow's runs</AppLink>
   546	            <AppLink href={projectsPathname()} onNavigate={onNavigate}>Go to Projects</AppLink>
   547	          </p>
   548	        </div>
   549	      )
   550	    } else if (detail.status === 'error') content = <ErrorPanel error={detail.error} what={`Run ${route.runId}`} onRetry={reloadDetail} />
   551	    else content = <RunView scope={scope!} detail={detail.data} current={currentWorkflow} selectedNodeId={nodeId} refreshToken={refreshToken} pollToken={runPoll} onNavigate={onNavigate} />
   552	  }
   553	
   554	  return (
   555	    <>
   556	      <div className="navigation">
   557	        <Breadcrumbs custom={{ crumbs, onNavigate }} selected={null} index={null} onNavigate={() => undefined} />
   558	        <p className="folder-info" data-testid="projects-info">{info}</p>
   559	      </div>
   560	      <main className="workspace workspace-projects" aria-busy={busy} data-testid="projects-workspace">
   561	        {content}
   562	      </main>
   563	    </>
   564	  )
   565	}
