    32	/** One run: its summary and pinned inputs, its definition graph coloured by actual node statuses, the selected node, and the assignment. */
    33	export function RunView({ scope, detail, current, selectedNodeId, refreshToken, pollToken = 0, onNavigate }: Props) {
    34	  const { summary, definition, snapshot } = detail
    35	  const snapshotById = new Map(snapshot.nodes.map(node => [node.node_id, node]))
    36	  const graphNodes: GraphNodeView[] = definition.nodes.map(node => {
    37	    const state = snapshotById.get(node.node_id)!
    38	    return { node_id: node.node_id, label: node.label, kind: node.kind, depends_on: node.depends_on, status: state.status, attempt: state.attempt }
    39	  })
    40	  const selectedDefinition = selectedNodeId === null ? null : definition.nodes.find(node => node.node_id === selectedNodeId) ?? null
    41	  const selectedState = selectedNodeId === null ? null : snapshotById.get(selectedNodeId) ?? null
    42	
    43	  const runKey = `${scope.projectId}/${scope.workflowId}/${scope.runId}`
    44	  const loadEvents = useCallback((signal: AbortSignal) => fetchEvents(scope, signal), [scope])
    45	  const { state: events, reload: reloadEvents } = useResource(`events:${runKey}`, loadEvents, refreshToken, pollToken)
    46	  // The inputs are one resource per run; a 404 INPUTS_NOT_FOUND means "not recorded", which loads as null.
    47	  const loadInputs = useCallback((signal: AbortSignal) => orNotRecorded(fetchRunInputs(scope, signal), NOT_RECORDED.inputs), [scope])
    48	  const { state: inputs, reload: reloadInputs } = useResource(`inputs:${runKey}`, loadInputs, refreshToken, pollToken)
    49	
    50	  const [tab, setTab] = useState<Tab>('run')
    51	  // A requirement quote handed from a review finding to a worker's task; it applies to one node and is dropped once applied or when leaving it.
    52	  const [pendingHighlight, setPendingHighlight] = useState<Highlight | null>(null)
    53	  const clearHighlight = useCallback(() => setPendingHighlight(null), [])
    54	  const highlight = pendingHighlight !== null && pendingHighlight.nodeId === selectedNodeId ? pendingHighlight.quote : null
    55	  // A captured file handed from a review finding to the launch node that shows it, applied the same way.
    56	  const [pendingFile, setPendingFile] = useState<FileFocus | null>(null)
    57	  const clearFile = useCallback(() => setPendingFile(null), [])
    58	  const fileFocus = pendingFile !== null && pendingFile.nodeId === selectedNodeId ? pendingFile.path : null
    59	
    60	  const nodePathname = (nodeId: string | null = null) => runPathname(scope.projectId, scope.workflowId, scope.runId, nodeId)
    61	  const selectNode = (nodeId: string) => onNavigate(nodePathname(nodeId))
    62	  const navigateToNode = (pathname: string) => {
    63	    setTab('run')
    64	    onNavigate(pathname)
    65	  }
    66	  const openRequirement = useCallback((nodeId: string, quote: string) => {
    67	    setPendingHighlight({ nodeId, quote })
    68	    setTab('run')
    69	    onNavigate(runPathname(scope.projectId, scope.workflowId, scope.runId, nodeId))
    70	  }, [onNavigate, scope])
    71	  const openFile = useCallback((nodeId: string, path: string) => {
    72	    setPendingFile({ nodeId, path })
    73	    setTab('run')
    74	    onNavigate(runPathname(scope.projectId, scope.workflowId, scope.runId, nodeId))
    75	  }, [onNavigate, scope])
    76	  const definitionChanged = current !== null && current.definition_revision !== definition.definition_revision
    77	
    78	  const onTabKey = (event: KeyboardEvent<HTMLButtonElement>) => {
    79	    const index = TABS.findIndex(candidate => candidate.id === tab)
    80	    let next: number | null = null
    81	    if (event.key === 'ArrowRight' || event.key === 'ArrowDown') next = (index + 1) % TABS.length
    82	    else if (event.key === 'ArrowLeft' || event.key === 'ArrowUp') next = (index - 1 + TABS.length) % TABS.length
    83	    else if (event.key === 'Home') next = 0
    84	    else if (event.key === 'End') next = TABS.length - 1
    85	    if (next === null) return
    86	    event.preventDefault()
    87	    setTab(TABS[next].id)
    88	    event.currentTarget.parentElement?.querySelector<HTMLButtonElement>(`[data-testid="${TABS[next].testId}"]`)?.focus()
    89	  }
    90	
    91	  return (
    92	    <div className="run-view" data-testid="run-view" data-run-id={summary.run_id} data-run-status={summary.status}>
    93	      <section className="run-summary" aria-labelledby="run-summary-title">
    94	        <h2 id="run-summary-title">Run {summary.run_id}</h2>
    95	        <p className="run-status-line" data-testid="run-status">
    96	          <StatusBadge status={summary.status} /> <span data-testid="run-status-meaning">{RUN_STATUS_MEANING[summary.status]}</span>
    97	        </p>
    98	        <dl className="projects-facts">
    99	          <div><dt>Created</dt><dd>{formatTime(summary.created_at)}</dd></div>
   100	          <div><dt>Updated</dt><dd>{formatTime(summary.updated_at)}</dd></div>
   101	          <div><dt>Last event</dt><dd>{snapshot.last_sequence === 0 ? 'none' : `sequence ${snapshot.last_sequence}`}</dd></div>
   102	          <div>
   103	            <dt>Pinned definition</dt>
   104	            <dd data-testid="pinned-definition">
   105	              {definition.name} · revision <code>{shortRevision(definition.definition_revision)}</code>
   106	              {current === null
   107	                ? null
   108	                : definitionChanged
   109	                  ? <span className="projects-warning" data-testid="definition-changed"> — the current workflow definition (revision <code>{shortRevision(current.definition_revision)}</code>) differs; this run is shown against its own pinned graph.</span>
   110	                  : <span className="projects-muted" data-testid="definition-current"> — same as the current workflow definition.</span>}
   111	            </dd>
   112	          </div>
   113	        </dl>
   114	        {(inputs.status === 'loading' || inputs.status === 'idle') && <LoadingPanel>Loading the run inputs…</LoadingPanel>}
   115	        {inputs.status === 'error' && <ErrorPanel error={inputs.error} what="The run inputs" onRetry={reloadInputs} />}
   116	        {inputs.status === 'ready' && inputs.data === null && <p className="projects-muted" data-testid="inputs-none">{INPUTS_NONE_SENTENCE}</p>}
   117	        {inputs.status === 'ready' && inputs.data !== null && (
   118	          <dl className="projects-facts" data-testid="run-inputs-facts">
   119	            <div><dt>Feature</dt><dd>{inputs.data.feature}</dd></div>
   120	            <div><dt>Base commit</dt><dd><code title={inputs.data.base_commit}>{shortRevision(inputs.data.base_commit)}</code></dd></div>
   121	            <div><dt>Source branch</dt><dd>{inputs.data.source_branch === null ? 'None recorded' : <code>{inputs.data.source_branch}</code>}</dd></div>
   122	            <div><dt>Mode</dt><dd>{inputs.data.mode}</dd></div>
   123	            {inputs.data.automatic && (
   124	              <>
   125	                <div><dt>Deadlines</dt><dd>{deadlinesLabel(inputs.data.automatic)}</dd></div>
   126	                <div><dt>Permission mode</dt><dd><code>{inputs.data.automatic.permission_mode}</code></dd></div>
   127	                <div><dt>Finish</dt><dd>{inputs.data.automatic.finish}</dd></div>
   128	              </>
   129	            )}
   130	          </dl>
   131	        )}
   132	      </section>
   133	
   134	      <div className="tabs run-tabs" role="tablist" aria-label="Run views">
   135	        {TABS.map(candidate => (
   136	          <button
   137	            key={candidate.id}
   138	            type="button"
   139	            role="tab"
   140	            id={`run-tab-${candidate.id}`}
   141	            className="tab"
   142	            aria-selected={tab === candidate.id}
   143	            aria-controls={`run-panel-${candidate.id}`}
   144	            tabIndex={tab === candidate.id ? 0 : -1}
   145	            data-testid={candidate.testId}
   146	            onClick={() => setTab(candidate.id)}
   147	            onKeyDown={onTabKey}
   148	          >
   149	            {candidate.label}
   150	          </button>
   151	        ))}
   152	      </div>
   153	
   154	      {tab === 'assignment' ? (
   155	        <AssignmentPanel scope={scope} inputs={inputs} onRetry={reloadInputs} onNavigate={navigateToNode} panelId="run-panel-assignment" tabId="run-tab-assignment" />
   156	      ) : (
   157	        <div role="tabpanel" id="run-panel-run" aria-labelledby="run-tab-run" className="run-body">
   158	          <section className="run-graph" aria-labelledby="run-graph-title">
   159	            <h3 id="run-graph-title">Pinned graph</h3>
   160	            <WorkflowGraph title={`Pinned definition graph of run ${summary.run_id}`} nodes={graphNodes} selectedId={selectedNodeId} onSelect={selectNode} />
   161	            <p className="projects-muted">Keyboard: Tab or arrow keys move between nodes, Enter or Space opens a node. The list below is the same graph as text.</p>
   162	          </section>
   163	
   164	          <div className="run-columns">
   165	            <nav className="run-nodes" aria-labelledby="run-nodes-title">
   166	              <h3 id="run-nodes-title">Nodes</h3>
   167	              <ul className="run-node-list" data-testid="run-node-list">
   168	                {graphNodes.map(node => (
   169	                  <li key={node.node_id} data-node-id={node.node_id} data-status={node.status}>
   170	                    <AppLink href={nodePathname(node.node_id)} onNavigate={onNavigate} className="run-node-link" current={node.node_id === selectedNodeId}>
   171	                      <span className="run-node-label">{node.label}</span>
   172	                      <span className="run-node-meta">
   173	                        <StatusBadge status={node.status!} /> {KIND_LABEL[node.kind]} · attempt {node.attempt}
   174	                      </span>
   175	                    </AppLink>
   176	                  </li>
   177	                ))}
   178	              </ul>
   179	            </nav>
   180	
   181	            <div className="run-detail-panel">
   182	              {selectedNodeId === null && (
   183	                <p className="projects-muted" data-testid="node-hint">Select a node in the graph or the list to inspect its attempts, result, evidence and events.</p>
   184	              )}
   185	              {selectedNodeId !== null && (selectedDefinition === null || selectedState === null) && (
   186	                <div className="projects-error" role="alert" data-testid="node-missing">
   187	                  <p>Node <code>{selectedNodeId}</code> is not part of this run's pinned definition.</p>
   188	                  <p><AppLink href={nodePathname()} onNavigate={onNavigate}>Back to the run</AppLink></p>
   189	                </div>
   190	              )}
   191	              {selectedDefinition !== null && selectedState !== null && (
   192	                <NodeDetail
   193	                  key={selectedNodeId}
   194	                  scope={scope}
   195	                  definition={selectedDefinition}
   196	                  definitionNodes={definition.nodes}
   197	                  snapshotNodes={snapshot.nodes}
   198	                  node={selectedState}
   199	                  events={events}
   200	                  onRetryEvents={reloadEvents}
   201	                  inputs={inputs}
   202	                  onRetryInputs={reloadInputs}
   203	                  refreshToken={refreshToken}
   204	                  onNavigate={onNavigate}
   205	                  highlight={highlight}
   206	                  onHighlightApplied={clearHighlight}
   207	                  onOpenRequirement={openRequirement}
   208	                  fileFocus={fileFocus}
   209	                  onFileFocusApplied={clearFile}
   210	                  onOpenFile={openFile}
   211	                />
   212	              )}
   213	            </div>
   214	          </div>
   215	        </div>
   216	      )}
   217	      <p className="projects-muted run-footnote">Statuses: {Object.entries(STATUS_LABEL).map(([key, label]) => `${label} (${key})`).join(', ')}. Only a succeeded run is a completed workflow.</p>
   218	    </div>
   219	  )

   221	import { useCallback, useEffect, useMemo, useRef, useState } from 'react'

   232	type Props = {
   233	  /** Null when the pathname is under /projects but malformed. */
   234	  route: ProjectsRoute | null

   240	/**
   241	 * The Projects root: registered projects, their workflow definitions and runs, read-only. Every level
   242	 * loads from the scoped API and shows loading, empty, not-found and failure states in place; nothing
   243	 * is ever substituted from fixtures.
   244	 */
   245	export function ProjectsView({ route, refreshToken, onNavigate, onAnnounce }: Props) {
   246	  const projectId = route && route.level !== 'projects' ? route.projectId : null
   247	  const workflowId = route && (route.level === 'workflow' || route.level === 'run') ? route.workflowId : null
   248	  const runId = route && route.level === 'run' ? route.runId : null
   249	  const nodeId = route && route.level === 'run' ? route.nodeId : null
   250	
   251	  // The registry itself changes while the viewer runs (a launch registers its workflow); both lists are cheap to re-read.
   252	  const registryPoll = usePoll(route?.level === 'projects' || route?.level === 'project')
   253	  const { state: projects, reload: reloadProjects } = useResource('projects', fetchProjects, refreshToken, registryPoll)
   254	  const loadWorkflows = useCallback((signal: AbortSignal) => fetchWorkflows(projectId!, signal), [projectId])
   255	  const { state: workflows, reload: reloadWorkflows } = useResource(projectId === null ? null : `workflows:${projectId}`, loadWorkflows, refreshToken, registryPoll)
   256	  const loadRuns = useCallback((signal: AbortSignal) => fetchRuns(projectId!, workflowId!, {}, signal), [projectId, workflowId])
   257	  const runsKey = workflowId === null ? null : `runs:${projectId}/${workflowId}`
   258	  // Run lists and run details re-read themselves while shown (a finished run no longer changes). A list stops polling
   259	  // while it loads or shows appended pages, since a new first page would drop them.
   260	  const [morePages, setMorePages] = useState<{ base: RunPage | null; pages: RunPage[]; loading: boolean; error: unknown }>({ base: null, pages: [], loading: false, error: null })
   261	  const [finished, setFinished] = useState(false)
   262	  // Appended pages belong to one list generation (this workflow, this Refresh); a new one starts without them.
   263	  const listGeneration = `${runsKey}\u0000${refreshToken}`
   264	  useEffect(() => { setMorePages({ base: null, pages: [], loading: false, error: null }) }, [listGeneration])
   265	  const listPoll = usePoll(route?.level === 'workflow' && !morePages.loading && morePages.pages.length === 0)
   266	  const runPoll = usePoll(route?.level === 'run' && !finished)
   267	  const { state: firstPage, reload: reloadRuns } = useResource(runsKey, loadRuns, refreshToken, listPoll)
   268	  const scope = useMemo(() => (projectId === null || workflowId === null || runId === null ? null : { projectId, workflowId, runId }), [projectId, workflowId, runId])
   269	  const loadDetail = useCallback((signal: AbortSignal) => fetchRunDetail({ projectId: projectId!, workflowId: workflowId!, runId: runId! }, signal), [projectId, workflowId, runId])
   270	  const { state: detail, reload: reloadDetail } = useResource(scope === null ? null : `run:${scope.projectId}/${scope.workflowId}/${scope.runId}`, loadDetail, refreshToken, runPoll)
   271	  const detailStatus = detail.status === 'ready' ? detail.data.summary.status : null
   272	  useEffect(() => { setFinished(detailStatus === 'succeeded' || detailStatus === 'cancelled') }, [detailStatus])
   273	
   274	  // Additional run pages are appended on demand; they belong to exactly one loaded first page and are dropped with it.
   275	  const firstPageData = firstPage.status === 'ready' ? firstPage.data : null
   276	  const extra = morePages.base !== null && morePages.base === firstPageData ? morePages : { base: firstPageData, pages: [], loading: false, error: null }
   277	  const pages: RunPage[] = firstPageData ? [firstPageData, ...extra.pages] : []
   278	  const runs: RunSummary[] = pages.flatMap(page => page.runs)
   279	  const nextCursor = pages.length ? pages[pages.length - 1].nextCursor : null
   280	  const loadMore = async () => {
   281	    if (!nextCursor || projectId === null || workflowId === null || firstPageData === null) return
   282	    setMorePages({ ...extra, base: firstPageData, loading: true, error: null })
   283	    try {
   284	      const page = await fetchRuns(projectId, workflowId, { cursor: nextCursor })
   285	      setMorePages(previous => (previous.base === firstPageData ? { ...previous, pages: [...previous.pages, page], loading: false } : previous))
   286	    } catch (error) {
   287	      setMorePages(previous => (previous.base === firstPageData ? { ...previous, loading: false, error } : previous))
   288	    }
   289	  }
   290	
   291	  const projectName = projects.status === 'ready' && projectId !== null ? projects.data.find(project => project.project_id === projectId)?.name ?? projectId : projectId
   292	  const currentWorkflow = workflows.status === 'ready' && workflowId !== null ? workflows.data.find(workflow => workflow.workflow_id === workflowId) ?? null : null
   293	  const workflowName = currentWorkflow?.name ?? workflowId
   294	
   295	  // A background poll re-announces only what changed; a load the reader started (navigation, Refresh) always announces.
   296	  const announced = useRef<string | null>(null)
   297	  useEffect(() => {
   298	    let message: string | null = null
   299	    if (route === null) message = 'This Projects link is invalid.'
   300	    else if (route.level === 'projects' && projects.status === 'ready') message = `Loaded ${projects.data.length} ${projects.data.length === 1 ? 'project' : 'projects'}.`
   301	    else if (route.level === 'project' && workflows.status === 'ready') message = `Loaded ${workflows.data.length} ${workflows.data.length === 1 ? 'workflow' : 'workflows'} for ${projectName ?? route.projectId}.`
   302	    else if (route.level === 'workflow' && firstPage.status === 'ready') message = `Loaded ${firstPage.data.runs.length} ${firstPage.data.runs.length === 1 ? 'run' : 'runs'} for ${workflowName ?? route.workflowId}.`
   303	    else if (route.level === 'run' && detail.status === 'ready') message = `Loaded run ${route.runId}: ${detail.data.summary.status.replace('_', ' ')}.`
   304	    else if ((projects.status === 'error') || workflows.status === 'error' || firstPage.status === 'error' || detail.status === 'error') message = 'Loading failed.'
   305	    if (message === null) { announced.current = null; return }
   306	    if (message === announced.current) return
   307	    announced.current = message
   308	    onAnnounce(message)
   309	  }, [route, projects, workflows, firstPage, detail, projectName, workflowName, onAnnounce])
   310	
   311	  const crumbs: PathCrumb[] = [{ id: 'home', label: 'Home', pathname: '/' }, { id: 'projects', label: 'Projects', pathname: projectsPathname() }]
   312	  if (projectId !== null) crumbs.push({ id: `project:${projectId}`, label: projectName ?? projectId, pathname: projectPathname(projectId) })
   313	  if (projectId !== null && workflowId !== null) crumbs.push({ id: `workflow:${workflowId}`, label: workflowName ?? workflowId, pathname: workflowPathname(projectId, workflowId) })
   314	  if (projectId !== null && workflowId !== null && runId !== null) crumbs.push({ id: `run:${runId}`, label: runId, pathname: runPathname(projectId, workflowId, runId) })
   315	  if (route === null) crumbs.push({ id: 'invalid', label: 'Invalid link' })
   316	
   317	  const notFound = (error: unknown) => error instanceof ProjectsApiError && error.notFound
   318	
   319	  let info: string
   320	  if (route === null) info = 'This Projects link is invalid.'
   321	  else if (route.level === 'projects') info = 'Registered projects. Read-only: runs are started, approved and retried from the workflow CLI, never from this page.'
   322	  else if (route.level === 'project') info = 'Workflow definitions registered for this project.'
   323	  else if (route.level === 'workflow') info = 'Runs of this workflow, newest first. Each run keeps the definition it was started with.'
   324	  else info = nodeId === null ? 'Run detail with its pinned definition graph.' : `Run detail, inspecting node ${nodeId}.`
   325	
   326	  const busy = projects.status === 'loading' || workflows.status === 'loading' || firstPage.status === 'loading' || detail.status === 'loading'
   327	
   328	  let content: React.ReactNode
   329	  if (route === null) {
   330	    content = (
   331	      <div className="projects-error" role="alert">
   332	        <p>This Projects link is invalid and could not be read. Links look like /projects/&lt;project&gt;/workflows/&lt;workflow&gt;/runs/&lt;run&gt;.</p>
   333	        <p className="projects-actions"><AppLink href={projectsPathname()} onNavigate={onNavigate}>Go to Projects</AppLink></p>
   334	      </div>
   335	    )
   336	  } else if (route.level === 'projects') {
   337	    if (projects.status === 'loading' || projects.status === 'idle') content = <LoadingPanel>Loading registered projects…</LoadingPanel>
   338	    else if (projects.status === 'error') content = <ErrorPanel error={projects.error} what="The project list" onRetry={reloadProjects} />
   339	    else if (projects.data.length === 0) {
   340	      content = (
   341	        <EmptyPanel title="No projects are registered." testId="empty-projects">
   342	          <p>The backend's project registry is empty. Projects are registered by the operator in the server configuration, not from this page.</p>
   343	        </EmptyPanel>
   344	      )
   345	    } else {
   346	      content = (
   347	        <section aria-labelledby="projects-title">
   348	          <h2 id="projects-title">Projects</h2>
   349	          <ul className="projects-list" data-testid="projects-list">
   350	            {projects.data.map(project => (
   351	              <li key={project.project_id}>
   352	                <AppLink href={projectPathname(project.project_id)} onNavigate={onNavigate} className="projects-card">
   353	                  <span className="projects-card-title">{project.name}</span>
   354	                  <span className="projects-muted">{project.project_id}</span>
   355	                </AppLink>
   356	              </li>
   357	            ))}
   358	          </ul>
   359	        </section>
   360	      )
   361	    }
   362	  } else if (route.level === 'project') {
   363	    if (workflows.status === 'loading' || workflows.status === 'idle') content = <LoadingPanel>Loading workflows of {projectName}…</LoadingPanel>
   364	    else if (workflows.status === 'error' && notFound(workflows.error)) {
   365	      content = (
   366	        <div className="projects-error" role="alert" data-testid="not-found">
   367	          <p><strong>Project “{route.projectId}” is not registered.</strong> The API has no project with this ID; it may have been removed from the registry or the link may be wrong.</p>

   371	    } else if (workflows.status === 'error') content = <ErrorPanel error={workflows.error} what={`The workflows of ${projectName}`} onRetry={reloadWorkflows} />
   372	    else if (workflows.data.length === 0) {
   373	      content = (
   374	        <EmptyPanel title={`${projectName} has no workflow definitions.`} testId="empty-workflows">

   387	                  <span className="projects-muted">{workflow.workflow_id} · {workflow.nodes.length} {workflow.nodes.length === 1 ? 'node' : 'nodes'} · current revision {shortRevision(workflow.definition_revision)}</span>
   388	                </AppLink>

   390	            ))}
   391	          </ul>
   392	        </section>
   393	      )
   394	    }
   395	  } else if (route.level === 'workflow') {
   396	    if (firstPage.status === 'loading' || firstPage.status === 'idle') content = <LoadingPanel>Loading runs of {workflowName}…</LoadingPanel>
   397	    else if (firstPage.status === 'error' && notFound(firstPage.error)) {
   398	      content = (
   399	        <div className="projects-error" role="alert" data-testid="not-found">
   400	          <p><strong>Workflow “{route.workflowId}” was not found in project “{route.projectId}”.</strong> Either the project is not registered or this workflow does not belong to it.</p>
   401	          <p className="projects-actions">

   403	            <AppLink href={projectsPathname()} onNavigate={onNavigate}>Go to Projects</AppLink>
   404	          </p>
   405	        </div>
   406	      )
   407	    } else if (firstPage.status === 'error') content = <ErrorPanel error={firstPage.error} what={`The runs of ${workflowName}`} onRetry={reloadRuns} />
   408	    else {
   409	      content = (
   410	        <div className="workflow-view">
   411	          <section aria-labelledby="runs-title">
   412	            <h2 id="runs-title">{workflowName}: runs</h2>
   413	            {runs.length === 0 ? (
   414	              <EmptyPanel title="No runs have been recorded for this workflow." testId="empty-runs">
   415	                <p>The workflow definition exists (see its current graph below) but it has never been run, or its runs are stored elsewhere. Runs are started from the workflow CLI.</p>
   416	              </EmptyPanel>
   417	            ) : (
   418	              <ul className="projects-list run-list" data-testid="run-list">
   419	                {runs.map(run => (
   420	                  <li key={run.run_id} data-run-id={run.run_id} data-status={run.status}>

   423	                      <span className="projects-muted">updated {formatTime(run.updated_at)} · created {formatTime(run.created_at)} · pinned revision {shortRevision(run.definition_revision)}</span>
   424	                    </AppLink>
   425	                  </li>
   426	                ))}
   427	              </ul>
   428	            )}
   429	            {nextCursor && (
   430	              <div className="projects-actions">
   431	                <button type="button" className="button" onClick={() => void loadMore()} disabled={extra.loading} aria-busy={extra.loading}>
   432	                  {extra.loading ? 'Loading more runs…' : 'Load more runs'}
   433	                </button>
   434	              </div>
   435	            )}
   436	            {extra.error !== null && <ErrorPanel error={extra.error} what="The next page of runs" onRetry={() => void loadMore()} />}
   437	          </section>
   438	          <section aria-labelledby="definition-title" className="workflow-definition">
   439	            <h3 id="definition-title">Current definition</h3>
   440	            {workflows.status === 'loading' && <LoadingPanel>Loading the current definition…</LoadingPanel>}
   441	            {workflows.status === 'error' && <ErrorPanel error={workflows.error} what="The current definition" onRetry={reloadWorkflows} />}
   442	            {workflows.status === 'ready' && currentWorkflow === null && <p className="projects-error-inline" role="alert">The workflow list does not contain “{route.workflowId}”, so its current definition cannot be shown.</p>}
   443	            {currentWorkflow && (
   444	              <>
   445	                <p className="projects-muted" data-testid="current-definition">{currentWorkflow.name} · revision <code>{shortRevision(currentWorkflow.definition_revision)}</code> · {currentWorkflow.nodes.length} nodes. Runs started before a definition change keep their own pinned graph.</p>
   446	                <WorkflowGraph title={`Current definition graph of ${currentWorkflow.name}`} nodes={currentWorkflow.nodes} selectedId={null} />
   447	              </>
   448	            )}
   449	          </section>
   450	        </div>
   451	      )
   452	    }
   453	  } else {
   454	    const loading = detail.status === 'loading' || detail.status === 'idle'
   455	    if (loading) content = <LoadingPanel>Loading run {route.runId}…</LoadingPanel>
   456	    else if (detail.status === 'error' && notFound(detail.error)) {
   457	      content = (
   458	        <div className="projects-error" role="alert" data-testid="not-found">
   459	          <p><strong>Run “{route.runId}” was not found in workflow “{route.workflowId}” of project “{route.projectId}”.</strong> Runs are only served under the project and workflow they belong to; a run of another project is not shown here.</p>
   460	          <p className="projects-actions">
   461	            <AppLink href={workflowPathname(route.projectId, route.workflowId)} onNavigate={onNavigate}>Go to the workflow's runs</AppLink>
   462	            <AppLink href={projectsPathname()} onNavigate={onNavigate}>Go to Projects</AppLink>
   463	          </p>
   464	        </div>
   465	      )
   466	    } else if (detail.status === 'error') content = <ErrorPanel error={detail.error} what={`Run ${route.runId}`} onRetry={reloadDetail} />
   467	    else content = <RunView scope={scope!} detail={detail.data} current={currentWorkflow} selectedNodeId={nodeId} refreshToken={refreshToken} pollToken={runPoll} onNavigate={onNavigate} />
   468	  }
   469	
   470	  return (
   471	    <>
   472	      <div className="navigation">
   473	        <Breadcrumbs custom={{ crumbs, onNavigate }} selected={null} index={null} onNavigate={() => undefined} />
   474	        <p className="folder-info" data-testid="projects-info">{info}</p>
   475	      </div>
   476	      <main className="workspace workspace-projects" aria-busy={busy} data-testid="projects-workspace">
   477	        {content}
   478	      </main>
   479	    </>
   480	  )

   482	/**
   483	 * Projects routes: `/projects[/<project>[/workflows/<workflow>[/runs/<run>[/nodes/<node>]]]]`.
   484	 *
   485	 * Every ID is an opaque contract identifier: it must match the shared ID pattern and is percent-encoded
   486	 * as one path segment. Anything else is malformed rather than looked up. This domain is read-only and
   487	 * separate from the Pi/Claude skill sources; nothing here names a filesystem path.
   488	 */
   489	
   490	/** Same pattern as the shared projects contract (`contracts/projects/v1.ts`). */
   491	export const PROJECT_ID_PATTERN = /^[A-Za-z0-9][A-Za-z0-9._-]{0,127}$/
   492	
   493	export const PROJECTS_ROUTE = 'projects'
   494	
   495	export type ProjectsRoute =
   496	  | { level: 'projects' }
   497	  | { level: 'project'; projectId: string }
   498	  | { level: 'workflow'; projectId: string; workflowId: string }
   499	  | { level: 'run'; projectId: string; workflowId: string; runId: string; nodeId: string | null }
   500	
   501	export function isContractId(value: string): boolean {
   502	  return PROJECT_ID_PATTERN.test(value)
   503	}
   504	
   505	function encode(segments: string[]): string {
   506	  return `/${segments.map(encodeURIComponent).join('/')}`
   507	}
   508	
   509	export function projectsPathname(): string {
   510	  return `/${PROJECTS_ROUTE}`
   511	}
   512	
   513	export function projectPathname(projectId: string): string {
   514	  return encode([PROJECTS_ROUTE, projectId])
   515	}
   516	
   517	export function workflowPathname(projectId: string, workflowId: string): string {
   518	  return encode([PROJECTS_ROUTE, projectId, 'workflows', workflowId])
   519	}
   520	
   521	export function runPathname(projectId: string, workflowId: string, runId: string, nodeId: string | null = null): string {
   522	  const segments = [PROJECTS_ROUTE, projectId, 'workflows', workflowId, 'runs', runId]
   523	  if (nodeId !== null) segments.push('nodes', nodeId)
   524	  return encode(segments)
   525	}

   527	export function routeToPathname(route: ProjectsRoute): string {
   528	  switch (route.level) {
   529	    case 'projects': return projectsPathname()
   530	    case 'project': return projectPathname(route.projectId)
   531	    case 'workflow': return workflowPathname(route.projectId, route.workflowId)
   532	    case 'run': return runPathname(route.projectId, route.workflowId, route.runId, route.nodeId)
   533	  }
   534	}

   536	/** Whether a pathname belongs to the Projects domain at all (decided before any decoding). */
   537	export function isProjectsPathname(pathname: string): boolean {
   538	  const first = pathname.split('/').filter(segment => segment !== '')[0]
   539	  return first === PROJECTS_ROUTE
   540	}
   541	
   542	/**
   543	 * Parses a Projects pathname. Returns null when it is not a Projects pathname or when any segment is
   544	 * malformed (bad encoding, an invalid ID, an unexpected literal or trailing segments).
   545	 */
   546	export function parseProjectsPathname(pathname: string): ProjectsRoute | null {
   547	  const raw = pathname.split('/').filter(segment => segment !== '')
   548	  if (raw[0] !== PROJECTS_ROUTE) return null
   549	  let segments: string[]
   550	  try {
   551	    segments = raw.slice(1).map(decodeURIComponent)
   552	  } catch {
   553	    return null
   554	  }
   555	  if (segments.length === 0) return { level: 'projects' }
   556	  const [projectId, workflowsLiteral, workflowId, runsLiteral, runId, nodesLiteral, nodeId, ...rest] = segments
   557	  if (!isContractId(projectId)) return null
   558	  if (workflowsLiteral === undefined) return { level: 'project', projectId }
   559	  if (workflowsLiteral !== 'workflows' || workflowId === undefined || !isContractId(workflowId)) return null
   560	  if (runsLiteral === undefined) return { level: 'workflow', projectId, workflowId }
   561	  if (runsLiteral !== 'runs' || runId === undefined || !isContractId(runId)) return null
   562	  if (nodesLiteral === undefined) return { level: 'run', projectId, workflowId, runId, nodeId: null }
   563	  if (nodesLiteral !== 'nodes' || nodeId === undefined || !isContractId(nodeId) || rest.length > 0) return null
   564	  return { level: 'run', projectId, workflowId, runId, nodeId }
   565	}

   639	export function executorCategory(kind: NodeKind): Executor {
   640	  if (kind === 'worker' || kind === 'review') return 'agent'

   652	 * The executor wording for a node; a review names print jobs when the served review result's transport is `print`, and
   653	 * the design challenge is always one print job.

   655	export function executorOf(kind: NodeKind, transport?: ReviewResult['reviewer']['transport'], nodeId?: string): string {
   656	  if (nodeId !== undefined && isChallengeNode({ node_id: nodeId, kind })) return 'one print job'
   657	  if (kind === 'review' && transport === 'print') return 'one print job per reviewer'
   658	  return EXECUTOR_LABEL[kind]

   665	export function formatTime(iso: string): string {
   666	  const time = Date.parse(iso)
   667	  if (Number.isNaN(time)) return iso
   668	  return new Date(time).toISOString().replace('T', ' ').replace(/\.\d{3}Z$/, ' UTC')

   671	/** Compact duration for deadlines and timeouts: 14400 → "4h", 5400 → "1h30m", 45 → "45s". */
   672	export function formatDuration(seconds: number): string {
   673	  const hours = Math.floor(seconds / 3600)
   674	  const minutes = Math.floor((seconds % 3600) / 60)
   675	  const rest = seconds % 60
   676	  const parts: string[] = []
   677	  if (hours > 0) parts.push(`${hours}h`)
   678	  if (minutes > 0) parts.push(`${minutes}m`)
   679	  if (rest > 0 || parts.length === 0) parts.push(`${rest}s`)
   680	  return parts.join('')
   681	}

   688	const DISPOSITIONS = ['open', 'resolved', 'accepted'] as const
   689	
   690	/** One line for a recorded review: verdict, finding counts by disposition and how many block integration (unresolved P0/P1). */
   691	export function reviewSummary(review: ReviewResult): string {
   692	  const { verdict, findings } = review
   693	  if (findings.length === 0) return `${verdict} with no findings`
   694	  const counts = DISPOSITIONS
   695	    .map(disposition => [disposition, findings.filter(finding => finding.disposition === disposition).length] as const)
   696	    .filter(([, count]) => count > 0)
   697	    .map(([disposition, count]) => `${count} ${disposition}`)
   698	  const blocking = findings.filter(isBlockingFinding).length
   699	  return `${verdict} with ${findings.length} ${findings.length === 1 ? 'finding' : 'findings'}: ${counts.join(', ')}, ${blocking === 0 ? 'none blocking' : `${blocking} blocking`}`
   700	}
