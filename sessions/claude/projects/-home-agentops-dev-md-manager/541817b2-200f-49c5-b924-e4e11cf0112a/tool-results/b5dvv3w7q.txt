    32	/** One run: its summary and pinned inputs, its definition graph coloured by actual node statuses, the selected node, and the assignment. */
    33	export function RunView({ scope, detail, current, selectedNodeId, refreshToken, pollToken = 0, onNavigate }: Props) {
    34	  const { summary, definition, snapshot } = detail
    35	  const snapshotById = new Map(snapshot.nodes.map(node => [node.node_id, node]))
    36	  const graphNodes: GraphNodeView[] = definition.nodes.map(node => {
    37	    const state = snapshotById.get(node.node_id)!
    38	    return { node_id: node.node_id, label: node.label, kind: node.kind, depends_on: node.depends_on, status: state.status, attempt: state.attempt }
    39	  })
    40	  const selectedDefinition = selectedNodeId === null ? null : definition.nodes.find(node => node.node_id === selectedNodeId) ?? null
    41	  const selectedState = selectedNodeId === null ? null : snapshotById.get(selectedNodeId) ?? null
    42	
    43	  const runKey = `${scope.projectId}/${scope.workflowId}/${scope.runId}`
    44	  const loadEvents = useCallback((signal: AbortSignal) => fetchEvents(scope, signal), [scope])
    45	  const { state: events, reload: reloadEvents } = useResource(`events:${runKey}`, loadEvents, refreshToken, pollToken)
    46	  // The inputs are one resource per run; a 404 INPUTS_NOT_FOUND means "not recorded", which loads as null.
    47	  const loadInputs = useCallback((signal: AbortSignal) => orNotRecorded(fetchRunInputs(scope, signal), NOT_RECORDED.inputs), [scope])
    48	  const { state: inputs, reload: reloadInputs } = useResource(`inputs:${runKey}`, loadInputs, refreshToken, pollToken)
    49	
    50	  const [tab, setTab] = useState<Tab>('run')
    51	  // A requirement quote handed from a review finding to a worker's task; it applies to one node and is dropped once applied or when leaving it.
    52	  const [pendingHighlight, setPendingHighlight] = useState<Highlight | null>(null)
    53	  const clearHighlight = useCallback(() => setPendingHighlight(null), [])
    54	  const highlight = pendingHighlight !== null && pendingHighlight.nodeId === selectedNodeId ? pendingHighlight.quote : null
    55	  // A captured file handed from a review finding to the launch node that shows it, applied the same way.
    56	  const [pendingFile, setPendingFile] = useState<FileFocus | null>(null)
    57	  const clearFile = useCallback(() => setPendingFile(null), [])
    58	  const fileFocus = pendingFile !== null && pendingFile.nodeId === selectedNodeId ? pendingFile.path : null
    59	
    60	  const nodePathname = (nodeId: string | null = null) => runPathname(scope.projectId, scope.workflowId, scope.runId, nodeId)
    61	  const selectNode = (nodeId: string) => onNavigate(nodePathname(nodeId))
    62	  const navigateToNode = (pathname: string) => {
    63	    setTab('run')
    64	    onNavigate(pathname)
    65	  }
    66	  const openRequirement = useCallback((nodeId: string, quote: string) => {
    67	    setPendingHighlight({ nodeId, quote })
    68	    setTab('run')
    69	    onNavigate(runPathname(scope.projectId, scope.workflowId, scope.runId, nodeId))
    70	  }, [onNavigate, scope])
    71	  const openFile = useCallback((nodeId: string, path: string) => {
    72	    setPendingFile({ nodeId, path })
    73	    setTab('run')
    74	    onNavigate(runPathname(scope.projectId, scope.workflowId, scope.runId, nodeId))
    75	  }, [onNavigate, scope])
    76	  const definitionChanged = current !== null && current.definition_revision !== definition.definition_revision
    77	
    78	  const onTabKey = (event: KeyboardEvent<HTMLButtonElement>) => {
    79	    const index = TABS.findIndex(candidate => candidate.id === tab)
    80	    let next: number | null = null
    81	    if (event.key === 'ArrowRight' || event.key === 'ArrowDown') next = (index + 1) % TABS.length
    82	    else if (event.key === 'ArrowLeft' || event.key === 'ArrowUp') next = (index - 1 + TABS.length) % TABS.length
    83	    else if (event.key === 'Home') next = 0
    84	    else if (event.key === 'End') next = TABS.length - 1
    85	    if (next === null) return
    86	    event.preventDefault()
    87	    setTab(TABS[next].id)
    88	    event.currentTarget.parentElement?.querySelector<HTMLButtonElement>(`[data-testid="${TABS[next].testId}"]`)?.focus()
    89	  }
    90	
    91	  return (
    92	    <div className="run-view" data-testid="run-view" data-run-id={summary.run_id} data-run-status={summary.status}>
    93	      <section className="run-summary" aria-labelledby="run-summary-title">
    94	        <h2 id="run-summary-title">Run {summary.run_id}</h2>
    95	        <p className="run-status-line" data-testid="run-status">
    96	          <StatusBadge status={summary.status} /> <span data-testid="run-status-meaning">{RUN_STATUS_MEANING[summary.status]}</span>
    97	        </p>
    98	        <dl className="projects-facts">
    99	          <div><dt>Created</dt><dd>{formatTime(summary.created_at)}</dd></div>
   100	          <div><dt>Updated</dt><dd>{formatTime(summary.updated_at)}</dd></div>
   101	          <div><dt>Last event</dt><dd>{snapshot.last_sequence === 0 ? 'none' : `sequence ${snapshot.last_sequence}`}</dd></div>
   102	          <div>
   103	            <dt>Pinned definition</dt>
   104	            <dd data-testid="pinned-definition">
   105	              {definition.name} · revision <code>{shortRevision(definition.definition_revision)}</code>
   106	              {current === null
   107	                ? null
   108	                : definitionChanged
   109	                  ? <span className="projects-warning" data-testid="definition-changed"> — the current workflow definition (revision <code>{shortRevision(current.definition_revision)}</code>) differs; this run is shown against its own pinned graph.</span>
   110	                  : <span className="projects-muted" data-testid="definition-current"> — same as the current workflow definition.</span>}
   111	            </dd>

   121	            <div><dt>Source branch</dt><dd>{inputs.data.source_branch === null ? 'None recorded' : <code>{inputs.data.source_branch}</code>}</dd></div>
   122	            <div><dt>Mode</dt><dd>{inputs.data.mode}</dd></div>
   123	            {inputs.data.automatic && (
   124	              <>
   125	                <div><dt>Deadlines</dt><dd>{deadlinesLabel(inputs.data.automatic)}</dd></div>
   126	                <div><dt>Permission mode</dt><dd><code>{inputs.data.automatic.permission_mode}</code></dd></div>
   127	                <div><dt>Finish</dt><dd>{inputs.data.automatic.finish}</dd></div>
   128	              </>
   129	            )}
   130	          </dl>
   131	        )}
   132	      </section>
   133	
   134	      <div className="tabs run-tabs" role="tablist" aria-label="Run views">
   135	        {TABS.map(candidate => (
   136	          <button
   137	            key={candidate.id}
   138	            type="button"
   139	            role="tab"
   140	            id={`run-tab-${candidate.id}`}
   141	            className="tab"
   142	            aria-selected={tab === candidate.id}
   143	            aria-controls={`run-panel-${candidate.id}`}
   144	            tabIndex={tab === candidate.id ? 0 : -1}
   145	            data-testid={candidate.testId}
   146	            onClick={() => setTab(candidate.id)}
   147	            onKeyDown={onTabKey}
   148	          >
   149	            {candidate.label}
   150	          </button>
   151	        ))}
   152	      </div>
   153	
   154	      {tab === 'assignment' ? (
   155	        <AssignmentPanel scope={scope} inputs={inputs} onRetry={reloadInputs} onNavigate={navigateToNode} panelId="run-panel-assignment" tabId="run-tab-assignment" />

   157	        <div role="tabpanel" id="run-panel-run" aria-labelledby="run-tab-run" className="run-body">
   158	          <section className="run-graph" aria-labelledby="run-graph-title">
   159	            <h3 id="run-graph-title">Pinned graph</h3>

   161	            <p className="projects-muted">Keyboard: Tab or arrow keys move between nodes, Enter or Space opens a node. The list below is the same graph as text.</p>
   162	          </section>

   165	            <nav className="run-nodes" aria-labelledby="run-nodes-title">
   166	              <h3 id="run-nodes-title">Nodes</h3>

   185	              {selectedNodeId !== null && (selectedDefinition === null || selectedState === null) && (

   218	    </div>

   252	/** What a node in this status means, given its kind. */

   254	  switch (status) {
   255	    case 'pending': return 'Not started: no attempt has run.'
   256	    case 'running': return kind === 'worker' ? 'The native worker session is running or its turn has not been signalled complete.' : 'In progress.'
   257	    case 'awaiting_approval': return 'Waiting for an explicit decision recorded outside this viewer. Not complete.'
   258	    case 'paused': return 'Paused with unresolved evidence. Not complete.'
   259	    case 'succeeded':
   260	      return kind === 'worker'
   261	        ? 'The worker session launched and ended its turn. This is not workflow completion; verification, review and integration are separate nodes.'
   262	        : kind === 'integration'
   263	          ? 'This integration step completed.'
   264	          : 'This step completed.'
   265	    case 'failed': return 'This step failed. The run cannot succeed without intervention.'
   266	    case 'cancelled': return 'This step was cancelled.'
   267	  }

   321	  const time = Date.parse(iso)

   326	/** Compact duration for deadlines and timeouts: 14400 → "4h", 5400 → "1h30m", 45 → "45s". */
   327	export function formatDuration(seconds: number): string {
   328	  const hours = Math.floor(seconds / 3600)
   329	  const minutes = Math.floor((seconds % 3600) / 60)
   330	  const rest = seconds % 60
   331	  const parts: string[] = []
   332	  if (hours > 0) parts.push(`${hours}h`)
   333	  if (minutes > 0) parts.push(`${minutes}m`)
   334	  if (rest > 0 || parts.length === 0) parts.push(`${rest}s`)
   335	  return parts.join('')
   336	}

   343	const DISPOSITIONS = ['open', 'resolved', 'accepted'] as const
   344	
   345	/** One line for a recorded review: verdict, finding counts by disposition and how many block integration (unresolved P0/P1). */
   346	export function reviewSummary(review: ReviewResult): string {
   347	  const { verdict, findings } = review
   348	  if (findings.length === 0) return `${verdict} with no findings`
   349	  const counts = DISPOSITIONS
   350	    .map(disposition => [disposition, findings.filter(finding => finding.disposition === disposition).length] as const)
   351	    .filter(([, count]) => count > 0)
   352	    .map(([disposition, count]) => `${count} ${disposition}`)
   353	  const blocking = findings.filter(isBlockingFinding).length
   354	  return `${verdict} with ${findings.length} ${findings.length === 1 ? 'finding' : 'findings'}: ${counts.join(', ')}, ${blocking === 0 ? 'none blocking' : `${blocking} blocking`}`
   355	}
   356	/**
   357	 * The files a worker created or changed, as the trusted verifier captured them at freeze (PRD_VIEWER_CLARITY 4.1): every
   358	 * changed text file within the caps is a `file` artifact carrying its repo-relative `path`; every other changed file is
   359	 * listed in `files_not_captured` with the reason. Results recorded before capture carry neither. Pure helpers and one
   360	 * hook, kept out of the component files so those only export components.

   362	import { useCallback } from 'react'

   400	/**
   401	 * Every captured file of a run, by path, with the launch node that shows it. Read from the launch nodes' own served worker
   402	 * results through the run's scoped results route; nothing is inferred for a lane whose result is absent or failed to load.
   403	 */
   404	export function useRunCapturedFiles(scope: RunScope, snapshotNodes: SnapshotNode[], refreshToken: number): Map<string, string> {
   405	  const launches = snapshotNodes.flatMap(node => {
   406	    if (node.kind !== 'worker' || node.result_uri === null) return []
   407	    const path = scopedResultPath(scope, node.result_uri)
   408	    return path === null ? [] : [{ nodeId: node.node_id, path }]
   409	  })
   410	  // The launches are serialised into the resource key, so the loader only changes when the linked results do.
   411	  const spec = JSON.stringify(launches)
   412	  const load = useCallback(async (signal: AbortSignal) => {
   413	    const list = JSON.parse(spec) as typeof launches
   414	    const settled = await Promise.allSettled(list.map(async launch => ({ nodeId: launch.nodeId, result: await fetchWorkerResult(scope, launch.path, signal) })))
   415	    const byPath = new Map<string, string>()
   416	    for (const entry of settled) {
   417	      if (entry.status !== 'fulfilled') continue
   418	      for (const file of capturedFiles(entry.value.result)) if (!byPath.has(file.path)) byPath.set(file.path, entry.value.nodeId)
   419	    }
   420	    return byPath
   421	  }, [scope, spec])
   422	  const key = launches.length === 0 ? null : `captured:${spec}`
   423	  const { state } = useResource(key, load, refreshToken)
   424	  return state.status === 'ready' ? state.data : EMPTY
   425	}

   427	const EMPTY: Map<string, string> = new Map()
   428	/**
   429	 * How review findings are attributed to the lanes a run actually had. A finding's `worker` is a lane id, `multiple`
   430	 * or `none`; exports of two-lane runs recorded `both`, which reads as `multiple`; null predates the vocabulary.
   431	 * Pure helpers, kept out of the component file so it only exports components.
   432	 */
   433	import type { ReviewFinding, RunDetail, RunInputs } from './api.ts'
   434	import type { Resource } from './useResource.ts'
   435	
   436	type DefinitionNode = RunDetail['definition']['nodes'][number]
   437	
   438	const MULTIPLE = 'multiple'
   439	const LEGACY_MULTIPLE = 'both'
   440	const NONE = 'none'
   441	const UNRECORDED = 'unrecorded'
   442	const SPECIAL_GROUPS: readonly string[] = [MULTIPLE, NONE, UNRECORDED]
   443	const SPECIAL_LABEL: Record<string, string> = { [MULTIPLE]: 'Multiple workers', [NONE]: 'No worker (cross-cutting)', [UNRECORDED]: 'Worker not recorded' }
   444	
   445	/** The group a finding belongs to when grouped by worker: a lane id, `multiple` (also for legacy `both`), `none` or `unrecorded`. */
   446	export function workerGroupOf(finding: Pick<ReviewFinding, 'worker'>): string {
   447	  if (finding.worker === null) return UNRECORDED
   448	  if (finding.worker === LEGACY_MULTIPLE) return MULTIPLE
   449	  return finding.worker
   450	}
   451	
   452	/** The Worker column wording: lane ids verbatim, the multi-lane values as words. */
   453	export function workerWording(worker: string): string {
   454	  if (worker === MULTIPLE || worker === LEGACY_MULTIPLE) return 'multiple workers'
   455	  if (worker === NONE) return 'none'
   456	  return worker
   457	}
   458	
   459	/**
   460	 * The run's lanes in policy order: from the recorded inputs when loaded, else from the pinned graph's `launch_<id>`
   461	 * nodes (the naming the exporter guarantees), so findings group by lane even for a run whose inputs are not recorded.
   462	 */
   463	export function runLanes(definitionNodes: DefinitionNode[], inputs: Resource<RunInputs | null>): string[] {
   464	  if (inputs.status === 'ready' && inputs.data !== null) return inputs.data.workers.map(worker => worker.node_id)
   465	  return definitionNodes.filter(node => node.kind === 'worker' && node.node_id.startsWith('launch_')).map(node => node.node_id.slice('launch_'.length))
   466	}

   468	/**
   469	 * Worker groups in display order: the run's lanes, then any lane id a finding names that the run does not list
   470	 * (never dropped), then `multiple`, `none` and `unrecorded`. Only populated groups are rendered.
   471	 */
   472	export function workerGroups(lanes: readonly string[], findings: readonly Pick<ReviewFinding, 'worker'>[]): { key: string; label: string }[] {
   473	  const keys = [...lanes]
   474	  for (const finding of findings) {
   475	    const group = workerGroupOf(finding)
   476	    if (!keys.includes(group) && !SPECIAL_GROUPS.includes(group)) keys.push(group)
   477	  }
   478	  return [...keys.map(lane => ({ key: lane, label: `Worker ${lane}` })), ...SPECIAL_GROUPS.map(key => ({ key, label: SPECIAL_LABEL[key] }))]
   479	}

   481	/**
   482	 * Where a message names `path` verbatim: the index just past each occurrence. An occurrence counts only as a whole path, so
   483	 * `docs/a.md` is not named by `src/docs/a.md` or `docs/a.mdx`; a trailing sentence period or a `:N` line suffix still counts.
   484	 * Nothing is inferred from similar names or partial paths (PRD_VIEWER_CLARITY section 2).
   485	 */
   486	function occurrencesOf(message: string, path: string): number[] {
   487	  const ends: number[] = []
   488	  if (path === '') return ends
   489	  for (let index = message.indexOf(path); index !== -1; index = message.indexOf(path, index + 1)) {
   490	    const before = index === 0 ? '' : message[index - 1]
   491	    const end = index + path.length
   492	    const after = message.slice(end, end + 2)
   493	    if (/[\w./-]/.test(before)) continue
   494	    if (/^[\w/-]/.test(after) || /^\.[\w/-]/.test(after)) continue
   495	    ends.push(end)
   496	  }
   497	  return ends
   498	}
   499	
   500	/** The findings whose message names `path` verbatim, in review order. */
   501	export function findingsForFile<T extends Pick<ReviewFinding, 'message'>>(findings: readonly T[], path: string): T[] {
   502	  return findings.filter(finding => occurrencesOf(finding.message, path).length > 0)
   503	}
   504	
   505	/** The `[from, to]` line ranges a message names for `path`: one per `path:N` (N to N) or `path:N-M` occurrence. */
   506	export function linesNamed(message: string, path: string): [number, number][] {
   507	  const ranges: [number, number][] = []
   508	  for (const end of occurrencesOf(message, path)) {
   509	    const match = /^:(\d+)(?:[-–](\d+))?/.exec(message.slice(end))
   510	    if (!match) continue
   511	    const from = Number(match[1])
   512	    const to = match[2] === undefined ? from : Number(match[2])
   513	    if (from < 1) continue
   514	    ranges.push([from, Math.max(from, to)])
   515	  }
   516	  return ranges
   517	}
   518	/**
   519	 * Per-reviewer wording for the review node. A review result always carries `reviewers`, one entry per reviewer the run
   520	 * declared (one named `review` for exports recorded before the list existed, filled by the adapter), so nothing here has
   521	 * a single-reviewer case: the combined verdict is unanimous, and a blocked run is explained by the reviewers that caused it.
   522	 * Pure helpers, kept out of the component file so it only exports components.
   523	 */
   524	import { isBlockingFinding, type ReviewFinding, type ReviewResult } from './api.ts'
   525	
   526	export type Reviewer = ReviewResult['reviewers'][number]
   527	
   528	const SEVERITIES: readonly ReviewFinding['severity'][] = ['P0', 'P1', 'P2']
   529	
   530	/**
   531	 * How a reviewer ended, in the contract's vocabulary (`REVIEWER_STATUSES` in `contracts/projects/v1.ts`): `accepted` (its
   532	 * completion file was accepted), `blocked` (its verdict, an unresolved P0/P1, a rejected file or the review deadline
   533	 * blocked the run), `superseded` (stopped after the run was decided while it was still working) or `pending` (no verdict
   534	 * recorded yet). The contract records no separate status for the deadline or a rejected file: both are `blocked` with a
   535	 * null verdict, so the wording for them is derived from `verdict` in `blockingReason`, never from an invented status.
   536	 */
   537	const STATUS_WORDING: Readonly<Record<Reviewer['status'], string>> = {
   538	  accepted: 'completion file accepted',
   539	  blocked: 'blocked',
   540	  superseded: 'stopped after the combined decision',
   541	  pending: 'no verdict recorded yet',
   542	}
   543	
   544	export function statusWording(status: Reviewer['status']): string {
   545	  return STATUS_WORDING[status]
   546	}
   547	
   548	/** Whether the reviewer was stopped because of another reviewer's outcome rather than its own. */
   549	function superseded(reviewer: Pick<Reviewer, 'status'>): boolean {
   550	  return reviewer.status === 'superseded'
   551	}
   552	
   553	/**
   554	 * Why this reviewer's own outcome blocks the run, or null when it does not: a blocked verdict; an approval that left an
   555	 * unresolved P0/P1; or a blocked status without any verdict, which the contract records for both the expired review
   556	 * deadline and a rejected completion file (it does not say which).
   557	 */
   558	export function blockingReason(reviewer: Pick<Reviewer, 'status' | 'verdict' | 'findings'>): string | null {
   559	  if (reviewer.verdict === 'blocked') return 'blocked the candidate'
   560	  if (reviewer.verdict === 'approved') return reviewer.findings.some(isBlockingFinding) ? 'approved with an unresolved P0/P1 finding' : null
   561	  if (reviewer.status === 'blocked') return 'delivered no verdict: the review deadline expired or its completion file was rejected'
   562	  return null
   563	}
   564	
   565	/**
   566	 * One line for the reviewer strip: how the reviewer ended, with the blocking reason when it has one. A reviewer without a
   567	 * verdict says so: superseded before deciding, pending and not yet launched, or pending after its launch.
   568	 */
   569	export function outcomeWording(reviewer: Pick<Reviewer, 'status' | 'verdict' | 'findings' | 'launched_at'>): string {
   570	  const reason = blockingReason(reviewer)
   571	  if (reason !== null) return reason
   572	  if (superseded(reviewer)) return reviewer.verdict === null ? `${statusWording(reviewer.status)}, no verdict` : statusWording(reviewer.status)
   573	  if (reviewer.status === 'pending') return reviewer.launched_at === null ? 'not launched, no verdict recorded yet' : statusWording(reviewer.status)
   574	  return statusWording(reviewer.status)
   575	}

   577	/** "1 P1, 2 P2" in severity order, listing only the severities present; "no findings" when the list is empty. */
   578	export function severityCounts(findings: readonly Pick<ReviewFinding, 'severity'>[]): string {
   579	  if (findings.length === 0) return 'no findings'
   580	  return SEVERITIES
   581	    .map(severity => [severity, findings.filter(finding => finding.severity === severity).length] as const)
   582	    .filter(([, count]) => count > 0)
   583	    .map(([severity, count]) => `${count} ${severity}`)
   584	    .join(', ')
   585	}
   586	
   587	function list(items: readonly string[]): string {
   588	  if (items.length <= 1) return items.join('')
   589	  return `${items.slice(0, -1).join(', ')} and ${items[items.length - 1]}`
   590	}
   591	
   592	/**
   593	 * Why a blocked review is blocked, naming the reviewers whose own outcome caused it (a blocked verdict, an unresolved
   594	 * P0/P1, the deadline or a rejected file), then the reviewers stopped because of it. Null for an approved review.
   595	 */
   596	export function blockedByWording(review: Pick<ReviewResult, 'verdict' | 'reviewers'>): string | null {
   597	  if (review.verdict !== 'blocked') return null
   598	  const reasons = review.reviewers.flatMap(reviewer => {
   599	    const reason = blockingReason(reviewer)
   600	    return reason === null ? [] : [`${reviewer.reviewer_id} (${reason})`]
   601	  })
   602	  const stopped = review.reviewers.filter(superseded).map(reviewer => reviewer.reviewer_id)
   603	  const head = reasons.length === 0
   604	    ? 'Blocked: the combined verdict is blocked although no reviewer recorded a blocking outcome.'
   605	    : `Blocked by ${list(reasons)}.`
   606	  const tail = stopped.length === 0 ? '' : ` ${list(stopped)} ${stopped.length === 1 ? 'was' : 'were'} superseded: stopped after that decision, no verdict recorded.`
   607	  return head + tail
   608	}
   609	/**
   610	 * Scrolls a handed-over target (a highlighted task quote, a linked file) into view and keeps it there while the page
   611	 * above it is still loading: results, files and Markdown arrive after the target mounts and would push it away. It stops
   612	 * at the first sign of the reader moving (wheel, touch, key, pointer), after a quiet second without layout changes, or
   613	 * after ten seconds.
   614	 */
   615	const QUIET_MS = 1000
   616	const LIMIT_MS = 10_000
   617	const READER_EVENTS = ['wheel', 'touchstart', 'keydown', 'pointerdown'] as const
   618	
   619	export function keepInView(target: Element, options: ScrollIntoViewOptions): () => void {
   620	  const view = target.ownerDocument.defaultView
   621	  if (!view) return () => {}
   622	  let quiet = 0
   623	  const stop = () => {
   624	    observer.disconnect()
   625	    view.clearTimeout(quiet)
   626	    view.clearTimeout(limit)
   627	    for (const type of READER_EVENTS) view.removeEventListener(type, stop, true)
   628	  }
   629	  const scroll = () => {
   630	    if (!target.isConnected) return stop()
   631	    target.scrollIntoView(options)
   632	    view.clearTimeout(quiet)
   633	    quiet = view.setTimeout(stop, QUIET_MS)
   634	  }
   635	  // Content loading above the target grows one of its ancestors (the document itself may keep a fixed-height body).
   636	  const observer = new ResizeObserver(scroll)
   637	  for (let element = target.parentElement; element !== null; element = element.parentElement) observer.observe(element)
   638	  const limit = view.setTimeout(stop, LIMIT_MS)
   639	  for (const type of READER_EVENTS) view.addEventListener(type, stop, { capture: true, passive: true })
   640	  scroll()
   641	  return stop
   642	}
