    56	/** Text-like artifact kinds are shown as plain text on demand; screenshots are images; anything else is only listed. */
    57	function TextArtifact({ scope, artifactId }: { scope: RunScope; artifactId: string }) {
    58	  const [open, setOpen] = useState(false)
    59	  const load = useCallback((signal: AbortSignal) => fetchArtifactText(scope, artifactId, signal), [scope, artifactId])
    60	  const { state, reload } = useResource(open ? `artifact:${artifactId}` : null, load)
    61	  return (
    62	    <div className="artifact-text">
    63	      <button type="button" className="button button-small" aria-expanded={open} onClick={() => setOpen(previous => !previous)}>
    64	        {open ? 'Hide contents' : 'Show contents'}
    65	      </button>
    66	      {open && state.status === 'loading' && <LoadingPanel>Loading artifact {artifactId}…</LoadingPanel>}
    67	      {open && state.status === 'error' && <ErrorPanel error={state.error} what={`Artifact ${artifactId}`} onRetry={reload} />}
    68	      {open && state.status === 'ready' && (
    69	        state.data.length === 0
    70	          ? <p className="projects-muted">This artifact is empty.</p>
    71	          : <pre className="artifact-log" data-testid={`artifact-text:${artifactId}`} tabIndex={0}>{state.data}</pre>
    72	      )}
    73	    </div>
    74	  )
    75	}

   141	/**
   142	 * What the trusted verifier proved, on a verify node and for each lane of the combined candidate: the result facts, the
   143	 * gate with its deferred checks, the checks with their logs, screenshots, other artifacts and the ownership outcome.
   144	 * Captured files are the launch node's; they are not repeated here.
   145	 */
   146	function VerifiedEvidence({ scope, result, node, phase, testId = 'worker-result' }: { scope: RunScope; result: WorkerResult; node: SnapshotNode; phase: 'worker' | 'candidate'; testId?: string }) {
   147	  const artifactsById = new Map(result.artifacts.map(artifact => [artifact.artifact_id, artifact]))
   148	  const checkLogs = new Set(result.checks.map(check => check.log_artifact_id))
   149	  const screenshots = result.artifacts.filter(artifact => artifact.kind === 'screenshot')
   150	  // Check logs are shown with their check and captured files on the launch node; everything else that is not a screenshot is listed here.
   151	  const others = result.artifacts.filter(artifact => artifact.kind !== 'screenshot' && artifact.kind !== 'file' && !checkLogs.has(artifact.artifact_id))
   152	  // Checks the gate recorded on this isolated snapshot but gates only at the combined candidate (build, browser).
   153	  const deferred = new Map((result.deferred_checks ?? []).map(entry => [entry.check_index, entry.id]))
   154	  const passed = result.status === 'succeeded' && result.error === null
   155	  return (
   156	    <div className="worker-evidence" data-testid={testId} data-view="verified">
   157	      <ResultFacts result={result} node={node} />
   158	
   159	      <section className="evidence-section" aria-labelledby={`${testId}-gate`} data-testid="gate-outcome" data-passed={passed ? 'true' : 'false'}>
   160	        <h4 id={`${testId}-gate`}>Gate</h4>
   161	        <p>
   162	          {passed
   163	            ? `Passed: every gating check of this ${phase === 'worker' ? 'isolated lane snapshot' : 'combined candidate'} exited 0.`
   164	            : 'Did not pass: see the error above for the reasons the gate recorded.'}
   165	        </p>
   166	        {deferred.size > 0 && (
   167	          <p className="projects-notice-inline" data-testid="deferred-checks">
   168	            Deferred checks: {[...deferred.values()].join(', ')} — executed and recorded on this isolated lane snapshot, but gated only at the combined candidate, which verifies the whole application.
   169	          </p>
   170	        )}
   171	        <p className="projects-muted" data-testid="ownership-outcome">
   172	          {phase === 'candidate'
   173	            ? 'Ownership: enforced on each lane\'s isolated snapshot, not on the combined candidate.'
   174	            : passed
   175	              ? 'Ownership: the verifier checked the changed files against the lane\'s owned paths on this snapshot, and the gate passed.'
   176	              : 'Ownership: checked on this snapshot; an ownership violation, if there was one, is among the reasons in the error above.'}
   177	        </p>
   178	      </section>
   179	
   180	      <section className="evidence-section" aria-labelledby={`${testId}-checks`}>
   181	        <h4 id={`${testId}-checks`}>Checks actually executed</h4>
   182	        {result.checks.length === 0 ? (
   183	          <p className="projects-muted" data-testid="checks-empty">No checks were recorded for this result.</p>
   184	        ) : (
   185	          <ul className="evidence-list" data-testid="checks-list">
   186	            {result.checks.map((check, index) => {
   187	              const log = artifactsById.get(check.log_artifact_id)
   188	              const isDeferred = deferred.has(index)
   189	              const exitText = check.exit_code === 0 ? 'exit 0' : `exit ${check.exit_code}`
   190	              return (
   191	                <li key={`${check.log_artifact_id}-${index}`} id={`check-${index}`} tabIndex={-1} className={isDeferred ? 'check check-deferred' : check.exit_code === 0 ? 'check check-passed' : 'check check-failed'} data-exit-code={check.exit_code} data-deferred={isDeferred ? 'true' : undefined}>
   192	                  <div className="check-head">
   193	                    <code className="check-command">{check.command}</code>
   194	                    <span className="check-exit">{isDeferred ? `${exitText} · recorded, gated at the combined candidate` : check.exit_code === 0 ? exitText : `${exitText} (failed)`}</span>
   195	                  </div>
   196	                  <p className="projects-muted check-meta">
   197	                    {formatTime(check.started_at)} → {formatTime(check.finished_at)} · cwd <code>{check.cwd}</code>
   198	                  </p>
   199	                  <div className="check-log">
   200	                    <span>Log artifact <code>{check.log_artifact_id}</code>{log ? '' : ' (not listed among the result artifacts)'}</span>
   201	                    {log && <TextArtifact scope={scope} artifactId={log.artifact_id} />}
   202	                  </div>
   203	                </li>
   204	              )
   205	            })}
   206	          </ul>
   207	        )}
   208	      </section>
   209	
   210	      <section className="evidence-section" aria-labelledby={`${testId}-screenshots`}>
   211	        <h4 id={`${testId}-screenshots`}>Screenshots</h4>
   212	        {screenshots.length === 0 && deferred.size > 0 ? (
   213	          <p className="projects-muted" data-testid="screenshots-deferred">No screenshot artifacts were published for this isolated lane run; browser evidence is gated and shown at the combined candidate.</p>
   214	        ) : screenshots.length === 0 ? (
   215	          <p className="projects-muted" data-testid="screenshots-empty">No screenshot artifacts were published.</p>
   216	        ) : (
   217	          <ul className="evidence-list evidence-screenshots" data-testid="screenshots">
   218	            {screenshots.map(artifact => (
   219	              <li key={artifact.artifact_id}>
   220	                <p><code>{artifact.artifact_id}</code> <span className="projects-muted">sha256 {artifact.sha256.slice(0, 12)}…</span></p>
   221	                <ScreenshotArtifact scope={scope} artifactId={artifact.artifact_id} />
   222	              </li>
   223	            ))}
   224	          </ul>
   225	        )}
   226	      </section>
   227	
   228	      <section className="evidence-section" aria-labelledby={`${testId}-artifacts`}>
   229	        <h4 id={`${testId}-artifacts`}>Other artifacts</h4>
   230	        {others.length === 0 ? (
   231	          <p className="projects-muted">No artifacts beyond the check logs, screenshots and captured files were published.</p>
   232	        ) : (
   233	          <ul className="evidence-list" data-testid="artifacts">
   234	            {others.map(artifact => (
   235	              <li key={artifact.artifact_id}>
   236	                <p><code>{artifact.artifact_id}</code> · {artifact.kind} · <span className="projects-muted">sha256 {artifact.sha256.slice(0, 12)}…</span></p>
   237	                {artifact.kind === 'other'
   238	                  ? <p className="projects-muted">Not rendered: only logs, reports, patches and screenshots are displayed.</p>
   239	                  : <TextArtifact scope={scope} artifactId={artifact.artifact_id} />}
   240	              </li>
   241	            ))}
   242	          </ul>
   243	        )}
   244	      </section>
   245	    </div>
   246	  )
   247	}

   249	/** One lane's result of the combined candidate, fetched through the run's scoped results route like any worker result. */
   250	function LaneResult({ scope, node, lane, launchNodeId, refreshToken, onNavigate }: {
   251	  scope: RunScope
   252	  node: SnapshotNode
   253	  lane: SnapshotNode['lane_results'][number]
   254	  /** The graph node that launched the lane, which shows the files it created; null when the pinned graph has none. */
   255	  launchNodeId: string | null
   256	  refreshToken: number
   257	  onNavigate: (pathname: string) => void
   258	}) {
   259	  const resultPath = scopedResultPath(scope, lane.result_uri)
   260	  const load = useCallback((signal: AbortSignal) => fetchWorkerResult(scope, resultPath!, signal), [scope, resultPath])
   261	  const { state, reload } = useResource(resultPath, load, refreshToken)
   262	  return (
   263	    <section className="lane-result" aria-label={`Lane ${lane.worker}`} data-testid={`lane-result:${lane.worker}`}>
   264	      <h5>Lane {lane.worker} <span className="projects-muted">· candidate attempt {lane.attempt}</span></h5>
   265	      {launchNodeId !== null && (
   266	        <p className="projects-muted" data-testid="lane-files-link">
   267	          Files are captured on the lane's worker snapshot only:{' '}
   268	          <AppLink href={runPathname(scope.projectId, scope.workflowId, scope.runId, launchNodeId)} onNavigate={onNavigate}>open the files the {lane.worker} lane created or changed</AppLink>.
   269	        </p>
   270	      )}
   271	      {resultPath === null && (
   272	        <p className="projects-error-inline" role="alert">The result link <code>{lane.result_uri}</code> is outside this run's results route and was not fetched.</p>
   273	      )}
   274	      {resultPath !== null && state.status === 'loading' && <LoadingPanel>Loading the {lane.worker} lane result…</LoadingPanel>}
   275	      {resultPath !== null && state.status === 'error' && <ErrorPanel error={state.error} what={`The ${lane.worker} lane result`} onRetry={reload} />}
   276	      {resultPath !== null && state.status === 'ready' && <VerifiedEvidence scope={scope} result={state.data} node={node} phase="candidate" testId={`lane-result-evidence:${lane.worker}`} />}
   277	    </section>
   278	  )
   279	}

   292	/**
   293	 * Everything known about one node of a run: state, executor, reuse evidence, result and events. Absent evidence is stated.
   294	 * A launch node shows what the worker produced first (result facts and created files), then its completion signal, launch
   295	 * receipt and the task behind a disclosure; a verify node shows what the verifier proved (gate, checks, screenshots); a
   296	 * review node's Result is the recorded review verdict.
   297	 */
   298	export function NodeDetail({ scope, definition, definitionNodes, snapshotNodes, node, events, onRetryEvents, inputs, onRetryInputs, refreshToken, onNavigate, highlight, onHighlightApplied, onOpenRequirement, fileFocus, onFileFocusApplied, onOpenFile }: Props) {
   299	  const resultPath = node.result_uri === null || definition.kind === 'review' ? null : scopedResultPath(scope, node.result_uri)
   300	  const loadResult = useCallback((signal: AbortSignal) => fetchWorkerResult(scope, resultPath!, signal), [scope, resultPath])
   301	  const { state: result, reload: reloadResult } = useResource(resultPath, loadResult, refreshToken)
   302	  // A review's executor wording depends on the transport its served result records, known once the review panel loads it.
   303	  const [reviewTransport, setReviewTransport] = useState<ReviewTransport | undefined>(undefined)
   304	
   305	  const nodeEvents = events.status === 'ready' ? events.data.filter(event => event.node_id === node.node_id) : []
   306	  const reuse = nodeEvents.filter(event => event.type === 'result_reused')
   307	  const approvals = nodeEvents.filter(event => event.type === 'approval_requested')
   308	
   309	  const isWorker = definition.kind === 'worker'
   310	  const isChallenge = isChallengeNode(definition)
   311	  const isReview = definition.kind === 'review' && !isChallenge
   312	  const recordedInputs = inputs.status === 'ready' ? inputs.data : null
   313	  const worker = isWorker && recordedInputs !== null ? recordedInputs.workers.find(candidate => candidate.launch_node_id === node.node_id) ?? null : null
   314	  const receiptSession = worker?.launch?.session_id ?? null
   315	  // The design challenge is also of kind review; the independent review is the other one.
   316	  const reviewNode = snapshotNodes.find(candidate => candidate.kind === 'review' && !isChallengeNode(candidate)) ?? null
   317	  const hasNode = (nodeId: string) => definitionNodes.some(candidate => candidate.node_id === nodeId)
   318	  const launchNodeOf = (lane: string) => {
   319	    const launched = recordedInputs?.workers.find(candidate => candidate.node_id === lane)?.launch_node_id
   320	    if (launched) return launched
   321	    return hasNode(`launch_${lane}`) ? `launch_${lane}` : null
   322	  }
   323	  const verifyNodeId = worker !== null && hasNode(`verify_${worker.node_id}`) ? `verify_${worker.node_id}` : null
   324	  const checksNode = verifyNodeId === null ? null : { href: runPathname(scope.projectId, scope.workflowId, scope.runId, verifyNodeId), label: definitionNodes.find(candidate => candidate.node_id === verifyNodeId)!.label }
   325	
   326	  const resultBody = (evidence: (data: WorkerResult) => ReactNode) => (
   327	    <>
   328	      {node.result_uri === null && node.lane_results.length === 0 && <p className="projects-muted" data-testid="result-none">No result has been published for this node{node.attempt === 0 ? ' (it has not started)' : ''}.</p>}
   329	      {node.result_uri !== null && resultPath === null && (
   330	        <p className="projects-error-inline" role="alert" data-testid="result-unscoped">
   331	          The result link <code>{node.result_uri}</code> is outside this run's results route and was not fetched.
   332	        </p>
   333	      )}
   334	      {resultPath !== null && result.status === 'loading' && <LoadingPanel>Loading the result…</LoadingPanel>}
   335	      {resultPath !== null && result.status === 'error' && <ErrorPanel error={result.error} what="The node result" onRetry={reloadResult} />}
   336	      {resultPath !== null && result.status === 'ready' && evidence(result.data)}
   337	    </>
   338	  )
   339	
   340	  const reuseSection = (
   341	    <section className="evidence-section" aria-labelledby="node-reuse">
   342	      <h4 id="node-reuse">Reuse evidence</h4>
   343	      {events.status === 'loading' && <LoadingPanel>Loading events…</LoadingPanel>}
   344	      {events.status === 'error' && <p className="projects-muted">Events could not be loaded, so reuse cannot be determined.</p>}
   345	      {events.status === 'ready' && (reuse.length === 0 ? (
   346	        <p className="projects-muted" data-testid="reuse-none">No reuse evidence recorded: the attempts shown are this node's own graph attempts, not reused results.</p>
   347	      ) : (
   348	        <ul className="evidence-list" data-testid="reuse-list">
   349	          {reuse.map(event => (
   350	            <li key={event.event_id}>
   351	              Attempt {event.attempt} reused the result of attempt {event.reused_from_attempt ?? 'unknown'} (event {event.sequence}, {formatTime(event.occurred_at)}): {event.message}
   352	            </li>
   353	          ))}
   354	        </ul>
   355	      ))}
   356	    </section>
   357	  )
   358	
   359	  return (
   360	    <section className="node-detail" aria-labelledby="node-detail-title" data-testid="node-detail" data-node-id={node.node_id}>
   361	      <h3 id="node-detail-title">{definition.label} <span className="projects-muted node-detail-id">({node.node_id})</span></h3>
   362	      <dl className="projects-facts">
   363	        <div><dt>Kind</dt><dd>{KIND_LABEL[definition.kind]}</dd></div>
   364	        <div><dt>Executed by</dt><dd data-testid="node-executor" data-executor={executorCategory(definition.kind)}>{executorOf(definition.kind, reviewTransport, definition.node_id)}</dd></div>
   365	        <div><dt>Status</dt><dd><StatusBadge status={node.status} /> <span data-testid="node-status-meaning">{nodeStatusMeaning(definition.kind, node.status)}</span></dd></div>
   366	        <div><dt>Graph attempt</dt><dd data-testid="node-attempt">{node.attempt === 0 ? '0 (not started)' : node.attempt}</dd></div>
   367	        <div>
   368	          <dt>Session</dt>
   369	          <dd>
   370	            {node.session_id
   371	              ? <code>{node.session_id}</code>
   372	              : receiptSession
   373	                ? <><code>{receiptSession}</code> <span className="projects-muted">(from the launch receipt)</span></>
   374	                : 'No session recorded'}
   375	          </dd>
   376	        </div>
   377	        <div><dt>Depends on</dt><dd>{definition.depends_on.length ? definition.depends_on.join(', ') : 'nothing'}</dd></div>
   378	      </dl>
   379	
   380	      {node.status === 'awaiting_approval' && (
   381	        <div className="projects-notice" role="status" data-testid="awaiting-notice">
   382	          <p><strong>Awaiting approval.</strong> This node is stopped at a decision that has not been recorded. Viewing does not approve it; decisions are made through the workflow CLI.</p>
   383	          {approvals.length > 0 && (
   384	            <ul className="evidence-list">
   385	              {approvals.map(event => <li key={event.event_id}>Requested at {formatTime(event.occurred_at)}: {event.message}</li>)}
   386	            </ul>
   387	          )}
   388	        </div>
   389	      )}
   390	
   391	      {isWorker ? (
   392	        <>
   393	          <section className="evidence-section" aria-labelledby="node-result">
   394	            <h4 id="node-result">Result</h4>
   395	            {resultBody(data => (
   396	              <ProducedEvidence scope={scope} result={data} node={node} reviewNode={reviewNode} refreshToken={refreshToken} focusPath={fileFocus} onFocusApplied={onFileFocusApplied} />
   397	            ))}
   398	          </section>
   399	
   400	          {(inputs.status === 'loading' || inputs.status === 'idle') && <LoadingPanel>Loading the run inputs…</LoadingPanel>}
   401	          {inputs.status === 'error' && <ErrorPanel error={inputs.error} what="The run inputs" onRetry={onRetryInputs} />}
   402	          {inputs.status === 'ready' && recordedInputs === null && (
   403	            <p className="projects-muted" data-testid="worker-inputs-none">
   404	              Inputs not recorded for this run (the export predates run inputs; re-export it with the workflow CLI), so the task, launch receipt and completion signal cannot be shown.
   405	            </p>
   406	          )}
   407	          {recordedInputs !== null && worker === null && (
   408	            <p className="projects-muted" data-testid="worker-inputs-unmatched">The run inputs list no worker launched by this node.</p>
   409	          )}
   410	          {worker !== null && (
   411	            <>
   412	              <WorkerSignals completion={worker.completion} handoff={worker.handoff} worker={worker} result={result} checksNode={checksNode} onNavigate={onNavigate} />
   413	              <WorkerQuestions questions={worker.questions} />
   414	              <LaunchReceipt launch={worker.launch} />
   415	              <TaskDisclosure highlight={highlight}>
   416	                <TaskPanel
   417	                  worker={worker}
   418	                  result={result}
   419	                  highlight={highlight}
   420	                  onHighlightApplied={onHighlightApplied}
   421	                  checksNode={checksNode}
   422	                  onNavigate={onNavigate}
   423	                />
   424	              </TaskDisclosure>
   425	            </>
   426	          )}
   427	          {reuseSection}
   428	        </>
   429	      ) : (
   430	        <>
   431	          {reuseSection}
   432	          <section className="evidence-section" aria-labelledby="node-result">
   433	            <h4 id="node-result">{isChallenge ? 'Design challenge' : isReview ? 'Review result' : 'Result'}</h4>
   434	            {isChallenge ? (
   435	              <ChallengePanel inputs={inputs} onRetry={onRetryInputs} />
   436	            ) : isReview ? (
   437	              <ReviewPanel
   438	                scope={scope}
   439	                node={node}
   440	                definitionNodes={definitionNodes}
   441	                snapshotNodes={snapshotNodes}
   442	                inputs={inputs}
   443	                refreshToken={refreshToken}
   444	                onNavigate={onNavigate}
   445	                onOpenRequirement={onOpenRequirement}
   446	                onOpenFile={onOpenFile}
   447	                onTransport={setReviewTransport}
   448	              />
   449	            ) : (
   450	              <>
   451	                {node.lane_results.length > 0 && (
   452	                  <div className="lane-results" data-testid="lane-results">
   453	                    <p className="projects-muted">The combined candidate verified every lane on one revision; each lane's result, with its checks and screenshots, is shown below.</p>
   454	                    {node.lane_results.map(lane => (
   455	                      <LaneResult key={lane.worker} scope={scope} node={node} lane={lane} launchNodeId={launchNodeOf(lane.worker)} refreshToken={refreshToken} onNavigate={onNavigate} />
   456	                    ))}
   457	                  </div>
   458	                )}
   459	                {resultBody(data => <VerifiedEvidence scope={scope} result={data} node={node} phase="worker" />)}
   460	              </>
   461	            )}
   462	          </section>
   463	        </>
   464	      )}
   465	
   466	      <section className="evidence-section" aria-labelledby="node-timeline">
   467	        <h4 id="node-timeline">Events for this node</h4>
   468	        {events.status === 'loading' && <LoadingPanel>Loading events…</LoadingPanel>}
   469	        {events.status === 'error' && <ErrorPanel error={events.error} what="The run events" onRetry={onRetryEvents} />}
   470	        {events.status === 'ready' && (nodeEvents.length === 0 ? (
   471	          <p className="projects-muted" data-testid="events-none">No events reference this node.</p>
   472	        ) : (
   473	          <ol className="event-list" data-testid="node-events">
   474	            {nodeEvents.map(event => (
   475	              <li key={event.event_id}>
   476	                <span className="event-seq">#{event.sequence}</span> <span className="projects-muted">{formatTime(event.occurred_at)}</span>{' '}
   477	                <span className="event-type">{event.type.replace(/_/g, ' ')}</span>
   478	                {event.status && <> · <StatusBadge status={event.status} /></>}
   479	                {event.attempt > 0 && <> · attempt {event.attempt}</>}
   480	                {event.message && <> — {event.message}</>}
   481	                {event.artifact && <> · artifact <code>{event.artifact.artifact_id}</code> ({event.artifact.kind})</>}
   482	              </li>
   483	            ))}
   484	          </ol>
   485	        ))}
   486	        {worker !== null && <StopLine stop={worker.stop} />}
   487	      </section>
   488	      <p className="projects-muted node-detail-footnote">Status labels here are historical observations; “{STATUS_LABEL.succeeded}” on a worker node is not workflow completion.</p>
   489	    </section>
   490	  )
   491	}
