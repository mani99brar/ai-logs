   675	    """The real command in its own process: a dry run writes nothing, and nothing ever calls claude."""
   676	
   677	    def listing(self) -> dict:
   678	        return {str(path.relative_to(self.directory)): path.read_bytes() if path.is_file() and not path.is_symlink() else None
   679	                for path in sorted(self.directory.rglob("*"))}
   680	
   681	    def test_repair_cli_launches_nothing(self):
   682	        self.block_at_candidate()
   683	        fix = self.commit_on(self.candidate_commit(), {"ui.txt": "after"})
   684	        bin_directory = self.root / "bin"
   685	        bin_directory.mkdir()
   686	        calls = self.root / "claude-calls.log"
   687	        (bin_directory / "claude").write_text(f"#!/bin/sh\necho \"$@\" >> {calls}\nexit 1\n")
   688	        (bin_directory / "claude").chmod(0o700)
   689	        env = {**os.environ, "PATH": f"{bin_directory}{os.pathsep}{os.environ['PATH']}"}
   690	        for lock in ("controller.lock", "automatic-supervisor.lock"):
   691	            (self.directory / lock).touch()  # A real run's controllers made them.
   692	        command = [sys.executable, "-m", "workflow", "repair", str(self.directory), "ui", "--commit", fix, "--reason", REASON]
   693	        cwd = Path(__file__).resolve().parents[1]
   694	        before, launches = self.listing(), (self.directory / "fake-launches.log").read_text()
   695	        result = subprocess.run([*command, "--dry-run"], cwd=cwd, env=env, capture_output=True, text=True, timeout=120)
   696	        self.assertEqual(result.returncode, 0, result.stderr)
   697	        self.assertIn("Dry run", result.stdout)
   698	        self.assertIn("worker:ui 2, candidate:adapter 2, candidate:ui 2", result.stdout)  # One candidate failure: attempt 1 of each.
   699	        self.assertEqual(self.listing(), before)
   700	        self.assertEqual(git(self.repo, "for-each-ref", "refs/workflow-repair"), "")
   701	        result = subprocess.run(command, cwd=cwd, env=env, capture_output=True, text=True, timeout=120)
   702	        self.assertEqual(result.returncode, 0, result.stderr)
   703	        self.assertIn(f"-m workflow automatic {self.directory} --live", result.stdout)
   704	        self.assertFalse(calls.exists())
   705	        self.assertEqual((self.directory / "fake-launches.log").read_text(), launches)
   706	        self.assertEqual(self.entries()[0]["status"], "applied")
