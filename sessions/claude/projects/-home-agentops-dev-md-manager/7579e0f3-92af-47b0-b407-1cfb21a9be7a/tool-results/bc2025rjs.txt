   400	                if (cwd / name).is_symlink():
   401	                    raise ValueError("Symlink changes require manual review before snapshot")
   402	            index = self.directory / f"{node}.snapshot-index"
   403	            if index.exists():
   404	                index.unlink()  # Only our private temporary index, never a user's index.
   405	            env = commit_env()
   406	            env["GIT_INDEX_FILE"] = str(index)
   407	            subprocess.run(["git", "-C", str(cwd), "read-tree", self.plan["base_commit"]], env=env, check=True)
   408	            subprocess.run(["git", "-C", str(cwd), "add", "-A", "--", "."], env=env, check=True)
   409	            tree = subprocess.check_output(["git", "-C", str(cwd), "write-tree"], env=env, text=True).strip()
   410	            # Validate paths from the actual captured tree, including staged renames/deletes.
   411	            captured = git(cwd, "diff-tree", "--no-commit-id", "--no-renames", "--name-only", "-r", self.plan["base_commit"], tree).splitlines()
   412	            if sorted(captured) != changed:
   413	                raise ValueError("Files changed during snapshot; refuse inconsistent evidence")
   414	            commit = self.plan["base_commit"]
   415	            if changed:
   416	                commit = subprocess.check_output(["git", "-C", str(cwd), "-c", "commit.gpgsign=false", "commit-tree", tree,
   417	                                                  "-p", self.plan["base_commit"], "-m", f"Workflow {self.plan['run_id']}: {node}"], env=env, text=True).strip()
   418	            ref = f"refs/workflow/{hashlib.sha256(str(self.directory).encode()).hexdigest()[:16]}/{node}"
   419	            subprocess.run(["git", "-C", str(cwd), "update-ref", ref, commit], check=True)
   420	            receipt = read_json(self.directory / f"{node}.interactive.json")
   421	            snapshots[node] = {"commit": commit, "changed_files": changed, "session_id": receipt["session_id"], **handoffs[node]}
   422	        save_json(record, snapshots)
   423	        self.event("freeze", "succeeded", "Immutable snapshots captured; worker-reported checks are not trusted")
   424	        return snapshots
   425	
   426	    def attempt(self, phase: str, node: str) -> int:
   427	        path = self.directory / "attempts.json"
   428	        value = read_json(path).get(f"{phase}:{node}", 1) if path.exists() else 1
   429	        if type(value) is not int or value < 1 or value > self.policy.get("max_verification_attempts", 3):
   430	            raise ValueError("Verification attempt exceeds the run's hard limit")
   431	        return value
   432	
   433	    def retry_check(self, phase: str, node: str) -> int:
   434	        value = self.attempt(phase, node) + 1
   435	        if value > self.policy.get("max_verification_attempts", 3):
   436	            raise ValueError("Verification attempt limit reached; inspect evidence and create an explicitly revised run")
   437	        path = self.directory / "attempts.json"
   438	        attempts = read_json(path) if path.exists() else {}
   439	        attempts[f"{phase}:{node}"] = value
   440	        save_json(path, attempts)
   441	        return value
   442	
   443	    def native_evidence(self) -> dict:
   444	        return {node: {key: read_json(self.directory / f"{node}.interactive.json").get(key)
   445	                       for key in ("session_id", "background_id", "launcher_invocations", "native_started_at")}
   446	                for node in self.workers}
   447	
   448	    def verify(self, node: str, snapshots: dict) -> str:
   449	        snap = snapshots[node]
   450	        attempt = self.attempt("worker", node)
   451	        self.event(f"verify_{node}", "running", f"Attempt {attempt}; revision {snap['commit']}")
   452	        packet = verify_revision(self.directory, self.plan, self.policy, node, snap["commit"], snap["changed_files"], snap["session_id"], attempt=attempt)
   453	        path = self.directory / "verification" / "worker" / node / str(attempt) / "packet.json"
   454	        drill = self.failure_drill()
   455	        if drill and drill["node_id"] == node and attempt == 1:
   456	            marker = "Intentional lab drill: verification branch failure, not a worker or test failure"
   457	            if marker not in packet["capture_errors"]:
   458	                packet["capture_errors"].append(marker)
   459	                packet["gate"]["reasons"].append(marker)
   460	            packet["gate"]["status"] = "blocked"
   461	            drill_path = self.directory / "failure-drill.json"
   462	            if not drill_path.exists():
   463	                save_json(drill_path, {"node_id": node, "phase": "worker", "attempt": 1,
   464	                                       "injected_at": now(), "native_before": self.native_evidence()})
   465	        packet["result"]["summary"] = snap["summary"]
   466	        packet["result"]["open_assumptions"] = snap["open_assumptions"]
   467	        save_json(path, packet)
   468	        passed = "Required tests and artifacts passed"
   469	        if packet["gate"].get("deferred_checks"):
   470	            passed += "; recorded for the candidate gate: " + ", ".join(packet["gate"]["deferred_checks"])
   471	        self.event(f"verify_{node}", packet["gate"]["status"], "; ".join(packet["gate"]["reasons"]) or passed)
   472	        if packet["gate"]["status"] != "passed":
   473	            raise RuntimeError(f"{node} verification blocked; see {path}. Retry explicitly or start a revised run.")
   474	        return str(path)
   475	
   476	    def candidate(self, state: PipelineState) -> str:
   477	        # Validate branch evidence again before combining anything.
   478	        worker_paths = [Path(state["packets"][node]) for node in self.workers]
   479	        for path in worker_paths:
   480	            packet = recheck_packet(read_json(path), self.policy, self.directory)
   481	            if packet["gate"]["status"] != "passed":
   482	                raise ValueError("Worker evidence no longer passes")
   483	        saved = self.directory / "candidate.json"
   484	        if saved.exists():
   485	            candidate = read_json(saved)
   486	        else:
   487	            cwd = self.directory / "candidate"
   488	            if cwd.exists():
   489	                raise ValueError("Partial candidate worktree exists; inspect before recovery")
   490	            git_worktree(self.plan["repository"], "add", "--detach", str(cwd), self.plan["base_commit"])
   491	            for node in self.workers:  # Declared order, selected lanes only.
   492	                commit = state["snapshots"][node]["commit"]
   493	                if commit != self.plan["base_commit"]:
   494	                    subprocess.run(["git", "-C", str(cwd), "-c", "core.hooksPath=/dev/null", "-c", "commit.gpgsign=false", "cherry-pick", commit], env=commit_env(), check=True, capture_output=True)
   495	            candidate = {"commit": git(cwd, "rev-parse", "HEAD"), "worktree": str(cwd)}
   496	            save_json(saved, candidate)
   497	        candidate_paths = []
   498	        for node in self.workers:
   499	            attempt = self.attempt("candidate", node)
   500	            packet = verify_revision(self.directory, self.plan, self.policy, node, candidate["commit"],
   501	                                     changed_files(Path(candidate["worktree"]), self.plan["base_commit"]),
   502	                                     state["snapshots"][node]["session_id"], phase="candidate", attempt=attempt)
   503	            path = self.directory / "verification" / "candidate" / node / str(attempt) / "packet.json"
   504	            self.event(f"candidate_{node}", packet["gate"]["status"], f"Combined revision {candidate['commit']}")
   505	            if packet["gate"]["status"] != "passed":
   506	                raise RuntimeError(f"Combined candidate failed {node} checks; see {path}")
   507	            candidate_paths.append(path)
   508	        bundle = {"run_id": self.plan["run_id"], "base_commit": self.plan["base_commit"],
   509	                  "candidate_commit": candidate["commit"], "policy_sha256": policy_digest(self.policy),
   510	                  "snapshots": state["snapshots"],
   511	                  "packets": [{"path": str(path), "sha256": digest_file(path)} for path in worker_paths + candidate_paths]}
   512	        path = self.directory / "review-bundle.json"
   513	        drill_path = self.directory / "failure-drill.json"
   514	        if drill_path.exists():
   515	            before = read_json(drill_path)["native_before"]
   516	            after = self.native_evidence()
   517	            audit = {"run_id": self.plan["run_id"], "native_before": before, "native_after": after,
   518	                     "workers_with_changed_launch_evidence": [node for node in self.workers if before.get(node) != after.get(node)],
   519	                     "verification_attempts": {node: sorted(int(item.name) for item in (self.directory / "verification" / "worker" / node).iterdir() if item.is_dir() and item.name.isdigit()) for node in self.workers},
   520	                     "scope": "Pipeline-issued sessions; native launch times/counts are null when unavailable. Out-of-band manual restarts are not certified."}
   521	            save_json(self.directory / "failure-report.json", audit)
   522	            bundle["failure_drill"] = audit
   523	        save_json(path, bundle)
   524	        return str(path)
   525	
   526	    def validate_bundle(self) -> tuple[dict, str]:
   527	        path = self.directory / "review-bundle.json"
   528	        bundle = read_json(path)
   529	        if bundle["run_id"] != self.plan["run_id"] or bundle["policy_sha256"] != policy_digest(self.policy):
   530	            raise ValueError("Bundle identity/policy changed")
   531	        for reference in bundle["packets"]:
   532	            packet_path = Path(reference["path"]).resolve()
   533	            if not packet_path.is_relative_to(self.directory) or digest_file(packet_path) != reference["sha256"]:
   534	                raise ValueError("Review evidence changed")
   535	            packet = recheck_packet(read_json(packet_path), self.policy, self.directory)
   536	            if packet["gate"]["status"] != "passed":
   537	                raise ValueError("Review artifact missing, changed or failing")
   538	        return bundle, digest_file(path)
   539	
   540	    def validate_review(self, review: dict, require_approved: bool = True) -> None:
   541	        """The combined record must name exactly this run's declared reviewers; approval needs every one of them."""
   542	        bundle, digest = self.validate_bundle()
   543	        check_review(review, bundle, digest, require_approved=require_approved, reviewers=reviewer_ids(self.plan))
   544	
   545	    def integrate(self, approved: str) -> str:
   546	        bundle, digest = self.validate_bundle()
   547	        review = read_json(self.directory / "review.json")
   548	        self.validate_review(review)
   549	        if approved != digest:
   550	            raise ValueError("Approval references stale evidence")
   551	        repo = Path(self.plan["repository"])
   552	        candidate = bundle["candidate_commit"]
   553	        intent = self.directory / "integration-intent.json"
   554	        if git(repo, "symbolic-ref", "--short", "HEAD") != self.plan["source_branch"]:
   555	            raise ValueError("Source branch changed")
   556	        if git(repo, "status", "--porcelain"):
   557	            raise ValueError("Source worktree is dirty; refusing integration")
   558	        if intent.exists():
   559	            if read_json(intent) != {"bundle_sha256": digest, "candidate_commit": candidate}:
   560	                raise ValueError("Integration intent changed")
   561	            if git(repo, "rev-parse", "HEAD") == candidate:
   562	                return candidate
   563	        if git(repo, "rev-parse", "HEAD") != self.plan["base_commit"]:
   564	            raise ValueError("Source advanced since preparation; rebase/review explicitly")
   565	        save_json(intent, {"bundle_sha256": digest, "candidate_commit": candidate})
   566	        subprocess.run(["git", "-C", str(repo), "-c", "core.hooksPath=/dev/null", "merge", "--ff-only", candidate], check=True, capture_output=True)
   567	        self.event("integrate", "succeeded", f"Fast-forwarded to {candidate}; no push performed")
   568	        return candidate

   571	class ExportRuntime:
   572	    """Read-only view of a run directory for re-exporting run-state.json.
   573	
   574	    It constructs no sessions and touches no worktree, so a copied or finished
   575	    run whose worktrees are gone still exports. Contradictory files are refused.
   576	    """
   577	
   578	    def __init__(self, directory: Path):
   579	        self.directory = directory.resolve()
   580	        self.plan = read_json(self.directory / "plan.json")
   581	        if "automatic" in self.plan:
   582	            from .automatic import validate_automatic
   583	            validate_automatic(self.plan)
   584	        self.policy = None
   585	        policy_path = self.directory / "policy.json"
   586	        if policy_path.exists():
   587	            self.policy = validate_pipeline_policy(read_json(policy_path))
   588	            if self.plan.get("policy_sha256") != policy_digest(self.policy):
   589	                raise ValueError("Pinned policy changed")
   590	        self.workers, self.excluded = lanes_of(self.plan, self.policy)
   591	        review_path = self.directory / "review.json"
   592	        if review_path.exists():
   593	            bundle_path = self.directory / "review-bundle.json"
   594	            if not bundle_path.exists():
   595	                raise ValueError("review.json exists without review-bundle.json; contradictory run directory")
   596	            # Packet hashes are not rechecked here; the adapter re-verifies packets itself. Reviews recorded
   597	            # before configured lanes may attribute a finding to `both`; the viewer renders it as several lanes.
   598	            check_review(read_json(review_path), read_json(bundle_path), digest_file(bundle_path), require_approved=False,
   599	                         allow_legacy="workers" not in self.plan)
   600	
   601	
   602	def lanes_of(plan: dict, policy: dict | None) -> tuple[list[str], list[str]]:
   603	    """The run's selected and excluded lanes, checked against the pinned policy when it is present."""
   604	    workers, excluded = plan_workers(plan), plan_excluded(plan)
   605	    if policy is not None:
   606	        declared = policy_workers(policy)
   607	        if not set(workers) <= set(declared) or not set(excluded) <= set(declared):
   608	            raise ValueError("Plan names lanes the pinned policy does not declare")
   609	        if "workers" in plan and set(workers) | set(excluded) != set(declared):
   610	            raise ValueError("Plan does not account for every declared lane")
   611	    if set(plan.get("nodes", {})) != set(workers):
   612	        raise ValueError("Plan nodes do not match the selected lanes; inspect retained allocation state")
   613	    return workers, excluded

   616	def carry_legacy_lanes(runtime: ExportRuntime, state, saver: SqliteSaver | None = None):
   617	    """Carry a pre-lane checkpoint's per-lane evidence into the lane state shape.
   618	
   619	    A checkpoint written before configured lanes stored each lane under `<lane>` and
   620	    `<lane>_packet`; the graph no longer has those channels, so `get_state` drops them.
   621	    Read them from the raw checkpoint and carry them under `lanes`/`packets` so the launch
   622	    evidence survives every export, whether from `export` or from the report written at
   623	    each CLI boundary. Runs with configured lanes are returned unchanged.
   624	    """
   625	    from types import SimpleNamespace
   626	    database = runtime.directory / "pipeline.sqlite"
   627	    if "workers" in runtime.plan or not database.exists():
   628	        return state
   629	    config = {"configurable": {"thread_id": runtime.plan["run_id"]}}
   630	
   631	    def carry(saver: SqliteSaver):
   632	        stored = saver.get_tuple(config)
   633	        raw = stored.checkpoint.get("channel_values", {}) if stored else {}
   634	        values = dict(state.values)
   635	        for node in runtime.workers:
   636	            if node in raw:
   637	                values.setdefault("lanes", {}).setdefault(node, raw[node])
   638	            if f"{node}_packet" in raw:
   639	                values.setdefault("packets", {}).setdefault(node, raw[f"{node}_packet"])
   640	        return SimpleNamespace(values=values, next=state.next, tasks=state.tasks)
   641	
   642	    if saver is not None:
   643	        return carry(saver)
   644	    with SqliteSaver.from_conn_string(str(database)) as own:
   645	        return carry(own)

   648	def export_run(runtime: ExportRuntime) -> dict:
   649	    """Re-export from the persisted checkpoint. Reading state never invokes a node or a session."""
   650	    from types import SimpleNamespace
   651	    database = runtime.directory / "pipeline.sqlite"
   652	    if database.exists():
   653	        config = {"configurable": {"thread_id": runtime.plan["run_id"]}}
   654	        with SqliteSaver.from_conn_string(str(database)) as saver:
   655	            state = carry_legacy_lanes(runtime, build_pipeline(saver, runtime).get_state(config), saver)
   656	    else:
   657	        state = SimpleNamespace(values={}, next=tuple(f"launch_{node}" for node in runtime.workers), tasks=[])  # Prepared, never started.
   658	    return export_state(runtime, state)

   661	def build_pipeline(checkpointer, runtime):
   662	    """The graph over the runtime's plan: `launch_<lane>` and `verify_<lane>` per selected lane, then the fixed tail."""
   663	    workers = list(runtime.workers)
   664	    def launcher(node):
   665	        return lambda _state: {"lanes": {node: runtime.launch(node)}}
   666	    def verifier(node):
   667	        return lambda state: {"packets": {node: runtime.verify(node, state["snapshots"])}}
   668	    def handoff(_state):
   669	        message = ("Awaiting explicit completion signals and automatic freeze." if runtime.plan.get("automatic")
   670	                   else f"Type in Claude terminals; provide a handoff for every lane ({', '.join(workers)}), then explicitly freeze.")
   671	        decision = interrupt({"kind": "worker_handoff", "message": message})
   672	        if decision != {"freeze": True}:
   673	            raise ValueError("Explicit freeze required")
   674	        return {"snapshots": runtime.freeze()}
   675	    def candidate(state): return {"bundle": runtime.candidate(state)}
   676	    def review(_state):
   677	        _, digest = runtime.validate_bundle()
   678	        if runtime.plan.get("automatic"):
   679	            from .automatic import review_candidate
   680	            decision = review_candidate(runtime)
   681	        else:
   682	            decision = interrupt({"kind": "independent_review", "bundle_sha256": digest,
   683	                                  "bundle_path": str(runtime.directory / "review-bundle.json")})
   684	        runtime.validate_review(decision)
   685	        save_json(runtime.directory / "review.json", decision)
   686	        runtime.event("review", "approved", decision["reviewer"])
   687	        return {"review": decision}
   688	    def approval(_state):
   689	        _, digest = runtime.validate_bundle()
   690	        if runtime.plan.get("automatic"):
   691	            from .automatic import validate_automatic
   692	            validate_automatic(runtime.plan)
   693	            decision = {"approve": digest}  # Explicit run-level authority: feature branch only.
   694	        else:
   695	            decision = interrupt({"kind": "integration_approval", "bundle_sha256": digest,
   696	                                  "message": "Approve fast-forward of source branch; no push."})
   697	        if decision != {"approve": digest}:
   698	            raise ValueError("Explicit exact-bundle approval required")
   699	        return {"approved_bundle": digest}
   700	    def integrate(state): return {"integrated_commit": runtime.integrate(state["approved_bundle"])}
   701	    graph = StateGraph(PipelineState)
   702	    launches = [f"launch_{node}" for node in workers]
   703	    verifies = [f"verify_{node}" for node in workers]
   704	    for node, name in zip(workers, launches):
   705	        graph.add_node(name, launcher(node))
   706	    for node, name in zip(workers, verifies):
   707	        graph.add_node(name, verifier(node))
   708	    for name, function in (("handoff", handoff), ("candidate", candidate), ("review", review), ("approval", approval), ("integrate", integrate)):
   709	        graph.add_node(name, function)
   710	    for name in launches:
   711	        graph.add_edge(START, name)
   712	    graph.add_edge(launches, "handoff")
   713	    for name in verifies:
   714	        graph.add_edge("handoff", name)
   715	    graph.add_edge(verifies, "candidate")
   716	    graph.add_edge("candidate", "review"); graph.add_edge("review", "approval")
   717	    graph.add_edge("approval", "integrate"); graph.add_edge("integrate", END)
   718	    return graph.compile(checkpointer=checkpointer)

   721	def start_workers(runtime, attach: bool = False) -> None:
   722	    """The first graph step (every launch) after `resume` decided the design challenge; `start` does the same inline."""
   723	    from .guardrails import challenge_gate
   724	    if not challenge_gate(runtime):
   725	        raise RuntimeError("The design challenge has not passed; no worker is launched")
   726	    config = graph_config(runtime)
   727	    with SqliteSaver.from_conn_string(str(runtime.directory / "pipeline.sqlite")) as saver:
   728	        graph = build_pipeline(saver, runtime)
   729	        if graph.get_state(config).values:
   730	            raise RuntimeError("Run already started; use status/explicit controls, never start again")
   731	        try:
   732	            graph.invoke({"run_id": runtime.plan["run_id"]}, config)
   733	        finally:
   734	            print(f"Report: {report(runtime, graph.get_state(config))}")
   735	    if attach:
   736	        print(json.dumps(attach_panels(runtime.sessions), indent=2))

   744	def report(runtime: Pipeline, state) -> Path:
   745	    """Escaped local results viewer, generated on every CLI boundary; no server needed."""
   746	    state = carry_legacy_lanes(runtime, state)  # `status` on a legacy run must export the same evidence as `export`.
   747	    export_state(runtime, state)
   748	    events_path = runtime.directory / "events.jsonl"
   749	    events = [json.loads(line) for line in events_path.read_text().splitlines()] if events_path.exists() else []
   750	    packets = list((runtime.directory / "verification").glob("*/*/*/packet.json"))
   751	    flow = ("Launch workers → completion signals → isolated checks → combined checks → independent review → verified feature branch (no main merge or push)"
   752	            if runtime.plan.get("automatic") else "Launch workers → human handoff → isolated checks → combined checks → independent review → approval → integration")
   753	    parts = ['<!doctype html><meta charset="utf-8"><title>Workflow report</title><style>body{font:16px system-ui;max-width:1100px;margin:40px auto;background:#151820;color:#eee}pre{white-space:pre-wrap}a{color:#8dcaff}img{max-width:100%}section{border:1px solid #555;padding:16px;margin:16px 0}</style>',
   754	             '<h1>Workflow report</h1><p>' + html.escape(flow) + '</p>',
   755	             '<h2>Current state</h2><pre>' + html.escape(json.dumps({"next": state.next, "interrupts": [str(task.interrupts) for task in state.tasks if task.interrupts], "errors": [str(task.error) for task in state.tasks if task.error], "integrated_commit": state.values.get("integrated_commit")}, indent=2)) + '</pre>',
   756	             '<h2>Timeline</h2><pre>' + html.escape(json.dumps(events, indent=2)) + '</pre>']
   757	    positions, edges, width, height = lane_positions(list(runtime.workers))
   758	    svg = [f'<h2>Execution graph</h2><svg role="img" aria-label="Workflow execution graph" viewBox="0 0 {width} {height}">']
   759	    for left, right in edges:
   760	        x1, y1 = positions[left]; x2, y2 = positions[right]
   761	        svg.append(f'<line x1="{x1}" y1="{y1}" x2="{x2}" y2="{y2}" stroke="#9aa"/>')
   762	    for name, (x, y) in positions.items():
   763	        color = '#665000' if name in state.next else '#263747'
   764	        svg.append(f'<rect x="{x-75}" y="{y-23}" width="150" height="46" rx="8" fill="{color}" stroke="#9aa"/><text x="{x}" y="{y+5}" text-anchor="middle" fill="white" font-size="14">{html.escape(name)}</text>')
   765	    lanes = ", ".join(runtime.workers) + (f" (excluded: {', '.join(runtime.excluded)})" if runtime.excluded else "")
   766	    svg.append(f'</svg><p>Lanes: {html.escape(lanes)}. Highlighted nodes are pending. A completed launch is not a completed worker task.</p>')
   767	    parts[2:2] = svg
   768	    from urllib.parse import quote
   769	    for path in packets:
   770	        packet = read_json(path)
   771	        parts.append('<section><h2>' + html.escape(str(path.relative_to(runtime.directory))) + '</h2><pre>' + html.escape(json.dumps({"gate": packet["gate"], "result": packet["result"]}, indent=2)) + '</pre>')
   772	        for artifact in packet["result"]["artifacts"]:
   773	            local = Path(packet["artifact_paths"][artifact["artifact_id"]]).resolve()
   774	            if not local.is_relative_to(runtime.directory):
   775	                continue
   776	            url = quote(str(local.relative_to(runtime.directory)), safe="/")
   777	            parts.append(f'<p><a href="{url}">{html.escape(artifact["artifact_id"])}</a></p>')
