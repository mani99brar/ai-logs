     1	"""`python -m workflow repair`: an operator's fix commit becomes a new lane snapshot after freeze, before review.
     2	
     3	Frozen snapshots never change, so a lane blocked at `verify_<lane>` or at the combined candidate used to need a new
     4	run. `repair` records the fix in `repairs.json`, builds one deterministic snapshot commit per named lane (the lane's
     5	snapshot with the fix's files it owns, parent the run's base), raises the attempt counters past the existing evidence
     6	and forks the run's LangGraph thread at the freeze boundary with the new snapshots. It launches no session, runs no
     7	check and never invokes the graph: `automatic --live` (or `retry` for a manual run) re-verifies the repaired lanes,
     8	rechecks the others, checks a new candidate generation for every lane and only then reaches review. Every apply step
     9	is idempotent, so a crash is completed by rerunning the identical command. `--workspace` makes a detached worktree at
    10	the right base to commit the fix in, never in the source checkout.
    11	"""

    49	def attempt_floor(directory: Path, phase: str, node: str) -> int:
    50	    """The first attempt of the lane's current revision in this phase: the latest applied repair's floor for it, else 1.
    51	
    52	    The attempt limit counts from here, so it is per lane, phase and revision; MAX_REPAIRS bounds the revisions.
    53	    """
    54	    for entry in reversed(applied_repairs(directory)):
    55	        if f"{phase}:{node}" in entry["attempt_floors"]:

    57	    return 1

   101	    """snapshots.json, the generation-0 record that is never rewritten, overlaid by `entries` in order."""
   102	    snapshots = read_json(directory / "snapshots.json")

   119	def parse_lanes(value: str, workers: list[str]) -> list[str]:
   120	    """`<lane>[,<lane>]`: distinct lanes of this run, returned in the run's lane order."""
   121	    named = [item.strip() for item in value.split(",")]
   122	    if any(not item for item in named):
   123	        raise ValueError(f"The lanes are a comma-separated list of lane ids, like ui or ui,adapter; got {value!r}")
   124	    if len(set(named)) != len(named):
   125	        raise ValueError(f"The lane list names a lane twice: {value}")
   126	    unknown = [item for item in named if item not in workers]
   127	    if unknown:
   128	        raise ValueError(f"Not a lane of this run: {', '.join(unknown)} (lanes: {', '.join(workers)})")
   129	    return [node for node in workers if node in named]

   142	def check_before_review(runtime, state) -> None:
   143	    pending = [item.value.get("kind") for task in state.tasks for item in task.interrupts]
   144	    if (any((runtime.directory / name).exists() for name in REVIEW_FILES) or state.values.get("integrated_commit")
   145	            or {"independent_review", "integration_approval"} & set(pending)):
   146	        raise ValueError("Reviewers have seen a candidate; code changes need a new run")

   149	def blocked_step(runtime, state) -> dict:
   150	    """The check verdict the run stopped at: a blocked packet of the candidate, or of a failed verify_<lane>."""
   151	    directory = runtime.directory
   152	    pending = [item.value.get("kind") for task in state.tasks for item in task.interrupts]
   153	    if not state.values:
   154	        raise ValueError("The run was never started; there is no frozen lane to repair")
   155	    if any(step.startswith("launch_") for step in state.next):
   156	        raise ValueError("A launch step is pending or failed; use reconcile")
   157	    if pending == ["worker_handoff"] or not (directory / "snapshots.json").exists():
   158	        raise ValueError("A lane blocked before freeze: repair needs frozen snapshots, and an automatic run blocked there needs a new run "
   159	                         "(RUNBOOK, Blocked after freeze: repair a lane)")
   160	    for phase in ("worker", "candidate"):
   161	        for node in runtime.workers:
   162	            folder = directory / "verification" / phase / node / str(raw_attempt(directory, phase, node))
   163	            if folder.is_dir() and not (folder / "packet.json").exists():
   164	                raise ValueError(f"Interrupted check at {folder}; use retry --phase {phase} --node {node}")
   165	    failed = [task.name for task in state.tasks if task.error and task.name in state.next]
   166	    verifies = {f"verify_{node}" for node in runtime.workers}
   167	    if failed == ["candidate"] and tuple(state.next) == ("candidate",):
   168	        phase, nodes = "candidate", runtime.workers
   169	    elif failed and set(state.next) <= verifies:
   170	        phase, nodes = "worker", [name.removeprefix("verify_") for name in failed]
   171	    else:
   172	        raise ValueError("The run did not stop at a check verdict of the candidate or a verify_<lane> step; inspect, then retry")
   173	    packets = []
   174	    for node in nodes:
   175	        attempt = raw_attempt(directory, phase, node)
   176	        path = directory / "verification" / phase / node / str(attempt) / "packet.json"
   177	        packet = read_json(path) if path.exists() else None
   178	        if packet and packet["gate"]["status"] == "blocked":
   179	            packets.append({"phase": phase, "node_id": node, "path": str(path.relative_to(directory)), "sha256": digest_file(path),
   180	                            "attempt": attempt, "output_commit": packet["expected"]["output_commit"], "reasons": packet["gate"]["reasons"]})
   181	    if not packets:
   182	        raise ValueError("The failed step has no blocked packet at its current attempt, so it is not a check verdict "
   183	                         "(an infrastructure error, a cherry-pick conflict, a partial candidate); inspect, then retry")
   184	    return {"step": "candidate" if phase == "candidate" else ", ".join(failed), "packets": packets}

   187	def check_source(runtime) -> None:
   188	    """integrate()'s own checks, done early: a fix committed on the pinned source branch is outside every gate."""
   189	    plan, repo = runtime.plan, Path(runtime.plan["repository"])
   190	    elsewhere = f"commit fixes in a repair workspace (repair {runtime.directory} <lane> --workspace), never in the source checkout"
   191	    branch = subprocess.run(["git", "-C", str(repo), "symbolic-ref", "-q", "--short", "HEAD"], capture_output=True, text=True).stdout.strip()
   192	    if branch != plan["source_branch"]:
   193	        raise ValueError(f"The source checkout is on {branch or 'a detached HEAD'}, not the run's branch {plan['source_branch']}; {elsewhere}")
   194	    head = git(repo, "rev-parse", "HEAD")
   195	    if head != plan["base_commit"]:
   196	        raise ValueError(f"The source branch {branch} moved from the run's base {plan['base_commit'][:8]} to {head[:8]}, which integration "
   197	                         f"refuses; {elsewhere}")
   198	    if git(repo, "status", "--porcelain"):
   199	        raise ValueError(f"The source checkout has uncommitted changes, which integration refuses; {elsewhere}")

   202	def fork_point(graph, config, runtime, previous: dict, checkpoint_id: str | None = None):
   203	    """The freeze boundary: the newest checkpoint whose next step is exactly the verify fan-out (the handoff's output or the
   204	    previous repair's fork), or the recorded one. Its snapshots must be snapshots.json overlaid by the applied repairs."""
   205	    fan_out = {f"verify_{node}" for node in runtime.workers}
   206	    fork = next((item for item in graph.get_state_history(config)
   207	                 if (item.config["configurable"]["checkpoint_id"] == checkpoint_id if checkpoint_id else set(item.next) == fan_out)), None)
   208	    if fork is None or set(fork.next) != fan_out or fork.values.get("snapshots") != previous:
   209	        raise RuntimeError("Contradictory run state: no freeze-boundary checkpoint holds snapshots.json plus the applied repairs; inspect")
   210	    return fork

   213	def blocked_run(runtime, graph, config, applied: list[dict]) -> tuple:
   214	    """Every state refusal. Returns the head, the blocked step, the fork point and the effective snapshots."""
   215	    state = graph.get_state(config)
   216	    check_before_review(runtime, state)
   217	    if len(applied) >= MAX_REPAIRS:
   218	        raise ValueError(f"{len(applied)} repairs are already applied to this run (at most {MAX_REPAIRS}); start a revised run")
   219	    # A failed verify superstep writes no checkpoint: a continuation that failed a lane's check again leaves the fork as
   220	    # the head with the error pending. Only a fork nothing has failed from is a repair not yet continued.
   221	    if applied and state.config["configurable"]["checkpoint_id"] == applied[-1]["head_after"] and not any(task.error for task in state.tasks):
   222	        raise ValueError(f"Repair {applied[-1]['n']} is applied and the run has not continued from it; an applied repair is not "
   223	                         f"replaced. Continue with: {continuation(runtime)}")
   224	    blocked = blocked_step(runtime, state)
   225	    check_source(runtime)
   226	    previous = effective_snapshots(runtime.directory, applied)
   227	    return state, blocked, fork_point(graph, config, runtime, previous), previous

   232	def resolve_commit(repo: Path, value: str) -> str:
   233	    result = None if value.startswith("-") else subprocess.run(["git", "-C", str(repo), "rev-parse", "--verify", "--quiet", f"{value}^{{commit}}"],
   234	                                                                capture_output=True, text=True)
   235	    if result is None or result.returncode != 0 or not result.stdout.strip():
   236	        raise ValueError(f"--commit {value} is not a commit in {repo}")
   237	    return result.stdout.strip()

   241	    return subprocess.run(["git", "-C", str(repo), "merge-base", "--is-ancestor", ancestor, commit], capture_output=True).returncode == 0

   244	def changed_paths(repo: Path, old: str, new: str) -> list[str]:
   245	    output = subprocess.check_output(["git", "-C", str(repo), "diff-tree", "-r", "--no-renames", "--name-only", "-z", old, new]).decode()
   246	    return sorted(filter(None, output.split("\0")))

   255	def tree_entries(repo: Path, commit: str, paths: list[str]) -> dict:
   256	    """`path -> ls-tree line` for the paths present in `commit`."""
   257	    if not paths:
   258	        return {}
   259	    output = subprocess.check_output(["git", "-C", str(repo), "ls-tree", "-r", "-z", commit, "--", *paths]).decode()
   260	    return {line.split("\t", 1)[1]: line for line in filter(None, output.split("\0"))}

   263	def overlay_tree(repo: Path, snapshot: str, paths: list[str], entries: dict) -> str:
   264	    """The snapshot's tree with `paths` as the fix has them (removed where it deletes them), built in a private index.
   265	
   266	    The index lives in a temporary directory: the source checkout's own index and the run directory are never written.
   267	    """
   268	    with tempfile.TemporaryDirectory(prefix="workflow-repair-") as scratch:
   269	        env = {**os.environ, "GIT_INDEX_FILE": str(Path(scratch) / "index")}
   270	        command = ["git", "-C", str(repo)]
   271	        subprocess.run([*command, "read-tree", snapshot], env=env, check=True, capture_output=True)
   272	        removed = [path for path in paths if path not in entries]
   273	        if removed:
   274	            subprocess.run([*command, "update-index", "--force-remove", "--", *removed], env=env, check=True, capture_output=True)
   275	        present = "".join(entries[path] + "\0" for path in paths if path in entries)
   276	        if present:
   277	            subprocess.run([*command, "update-index", "-z", "--index-info"], input=present.encode(), env=env, check=True, capture_output=True)
   278	        return subprocess.check_output([*command, "write-tree"], env=env, text=True).strip()

   281	def derive(runtime, lanes: list[str], source: str, phase: str, previous: dict) -> dict:
   282	    """The fix's base, the files of each named lane and each lane's new tree, checked with freeze's rules. Read only
   283	    (Git writes the trees' objects, which nothing references)."""
   284	    directory, repo = runtime.directory, Path(runtime.plan["repository"])
   285	    generation = len(applied_repairs(directory))
   286	    saved = [candidate_paths(directory, number)[0] for number in range(generation + 1)]
   287	    commits = [read_json(path)["commit"] if path.exists() else None for path in saved]
   288	    current, older = commits[-1], [commit for commit in commits[:-1] if commit]
   289	    if phase == "candidate" and current and descends(repo, current, source):
   290	        base_kind, base = "candidate", current
   291	    elif any(descends(repo, commit, source) for commit in older):
   292	        raise ValueError(f"--commit {source[:8]} is on the candidate of an earlier generation, not the current one ({(current or 'none')[:8]}); "
   293	                         "commit the fix on the current candidate")
   294	    elif any(descends(repo, previous[lane]["commit"], source) for lane in lanes):
   295	        if len(lanes) > 1:
   296	            raise ValueError("A fix on a lane snapshot repairs one lane only: overlaying base content onto another lane would discard its "
   297	                             "changes. Commit a fix spanning lanes on the failing candidate (repair <run> <lanes> --workspace)")
   298	        base_kind, base = "snapshot", previous[lanes[0]]["commit"]
   299	    else:
   300	        bases = ([f"the current candidate {current[:8]}"] if phase == "candidate" and current else []) + \
   301	                [f"the {lane} snapshot {previous[lane]['commit'][:8]}" for lane in lanes]
   302	        raise ValueError(f"--commit {source[:8]} descends from neither {' nor '.join(bases)}; make it in a repair workspace "
   303	                         f"(repair {directory} <lane> --workspace)")
   304	    fix = changed_paths(repo, base, source)
   305	    for path in fix:
   306	        safe_path(path)
   307	        owner = lane_owner(runtime, path)
   308	        if owner is None:
   309	            raise ValueError(f"The fix changes {path}, which no lane owns")
   310	        if owner in runtime.excluded:
   311	            raise ValueError(f"Ownership violation: the fix changes {path}, owned by excluded lane {owner}")
   312	        if owner not in lanes:
   313	            raise ValueError(f"The fix changes {path}, owned by lane {owner}, which this repair does not name; name {owner} too or leave the file alone")
   314	    entries = tree_entries(repo, source, fix)
   315	    for path, line in entries.items():
   316	        mode = line.split(" ", 1)[0]
   317	        if mode == "120000":
   318	            raise ValueError(f"Symlink changes require manual review before snapshot: {path}")
   319	        if mode == "160000":
   320	            raise ValueError(f"Gitlink (submodule) changes are refused: {path}")
   321	    result = {}
   322	    for lane in lanes:
   323	        snapshot = previous[lane]["commit"]
   324	        paths = [path for path in fix if lane_owner(runtime, path) == lane]
   325	        tree = overlay_tree(repo, snapshot, paths, entries)
   326	        if tree == git(repo, "rev-parse", f"{snapshot}^{{tree}}"):
   327	            raise ValueError(f"The fix changes nothing in the {lane} snapshot {snapshot[:8]}: a repair is not a retry (use retry for a transient failure)")
   328	        changed = changed_paths(repo, runtime.plan["base_commit"], tree)
   329	        runtime.check_ownership(lane, changed)
   330	        result[lane] = {"previous_commit": snapshot, "fix_files": paths, "tree": tree, "changed_files": changed}
   331	    return {"base_kind": base_kind, "base_commit": base, "lanes": result,
   332	            "expected_candidate_tree": git(repo, "rev-parse", f"{source}^{{tree}}") if base_kind == "candidate" else None}

   335	def attempt_targets(directory: Path, lanes: list[str], workers: list[str]) -> dict:
   336	    """The worker phase of each repaired lane and the candidate phase of every lane that has a candidate attempt directory
   337	    start past their counter and past every existing attempt: nothing is moved, and no attempt directory is reused."""
   338	    targets = {}
   339	    for phase, nodes in (("worker", lanes), ("candidate", workers)):
   340	        for node in nodes:
   341	            existing = attempt_directories(directory, phase, node)
   342	            if phase == "worker" or existing:
   343	                targets[f"{phase}:{node}"] = max(raw_attempt(directory, phase, node), max(existing, default=0) + 1)
   344	    return targets

   359	def commit_snapshots(runtime, entry: dict, derived: dict) -> None:
   360	    """S2: one commit per lane, parent the run's base, dated at the recorded time so a rerun makes the same commit; refs keep
   361	    it and the operator's commit. Beside refs/workflow/, where no lane id can name them."""
   362	    repo, plan = runtime.plan["repository"], runtime.plan
   363	    stamp = f"{int(datetime.fromisoformat(entry['recorded_at']).timestamp())} +0000"
   364	    env = {**commit_env(), "GIT_AUTHOR_DATE": stamp, "GIT_COMMITTER_DATE": stamp}
   365	    prefix = f"refs/workflow-repair/{hashlib.sha256(str(runtime.directory).encode()).hexdigest()[:16]}/{entry['n']}"
   366	    for lane, item in derived["lanes"].items():
   367	        commit = plan["base_commit"]  # As freeze records a lane that changed nothing.
   368	        if item["changed_files"]:
   369	            commit = subprocess.check_output(["git", "-C", repo, "-c", "commit.gpgsign=false", "commit-tree", item["tree"], "-p", plan["base_commit"],
   370	                                              "-m", f"Workflow {plan['run_id']}: {lane} repair {entry['n']}"], env=env, text=True).strip()
   371	        item["commit"] = commit
   372	        subprocess.run(["git", "-C", repo, "update-ref", f"{prefix}/{lane}", commit], check=True)
   373	    subprocess.run(["git", "-C", repo, "update-ref", f"{prefix}/source_commit", entry["source_commit"]], check=True)

   376	def save_lanes(runtime, entry: dict, derived: dict) -> None:
   377	    """S3: each lane's commit and files into the journal, and the operator's change per lane as repair-<n>.diff."""
   378	    for lane, item in derived["lanes"].items():
   379	        entry["lanes"][lane].update(commit=item["commit"], changed_files=item["changed_files"])
   380	    save_entry(runtime.directory, entry)
   381	    with (runtime.directory / f"repair-{entry['n']}.diff").open("wb") as handle:
   382	        for item in entry["lanes"].values():
   383	            handle.write(subprocess.check_output(["git", "-C", runtime.plan["repository"], "diff", "--binary", item["previous_commit"], item["commit"]]))

   386	def raise_attempts(directory: Path, entry: dict) -> None:
   387	    """S4: each target counter raised, never lowered."""
   388	    path = directory / "attempts.json"
   389	    attempts = read_json(path) if path.exists() else {}
   390	    for key, target in entry["attempt_targets"].items():
   391	        attempts[key] = max(attempts.get(key, 1), target)
   392	    save_json(path, attempts)

   395	def fork_run(graph, config, runtime, entry: dict, fork, snapshots: dict) -> str:
   396	    """S5: exactly one new checkpoint, as if `handoff` had written the repaired snapshots, forked from the freeze boundary.
   397	
   398	    `fork.config` carries its checkpoint_id, and pending writes are not carried into a fork: the failed step's trigger
   399	    and the old packets stay on the failed branch. The bare thread config would update the failed head instead, which
   400	    runs the candidate beside the verifies with the new snapshots and the old packets. Neither the handoff's interrupt,
   401	    nor freeze, nor the worker stop runs; only its edges fire.
   402	    """
   403	    head = graph.get_state(config)
   404	    if head.config["configurable"]["checkpoint_id"] == entry["head_before"]:
   405	        graph.update_state(fork.config, {"snapshots": snapshots}, as_node="handoff")
   406	        head = graph.get_state(config)
   407	    elif not (head.parent_config and head.parent_config["configurable"]["checkpoint_id"] == entry["fork_from"] and head.metadata.get("source") == "update"):
   408	        raise RuntimeError(f"Checkpoint moved since repair {entry['n']} was recorded (head {head.config['configurable']['checkpoint_id']}); inspect")
   409	    if set(head.next) != {f"verify_{node}" for node in runtime.workers} or any(task.error for task in head.tasks) or head.values.get("snapshots") != snapshots:
   410	        raise RuntimeError(f"Contradictory run state after the fork of repair {entry['n']}: the next step is not the verify fan-out; inspect")
   411	    return head.config["configurable"]["checkpoint_id"]

   414	def finish_repair(runtime, graph, config, entry: dict, head_after: str) -> None:
   415	    """S6: the timeline, the entry `applied`, then the report and run-state.json. Events a crash duplicated are harmless."""
   416	    directory, n = runtime.directory, entry["n"]
   417	    answers = " ".join(f"Answers {packet['phase']}/{packet['node_id']} attempt {packet['attempt']}: {'; '.join(packet['reasons'])}."
   418	                       for packet in entry["blocked"]["packets"])
   419	    for lane, item in entry["lanes"].items():
   420	        runtime.event(f"verify_{lane}", "paused", f"Repair {n} by the operator: snapshot {item['commit'][:8]} = {item['previous_commit'][:8]} + "
   421	                                                  f"{entry['source_commit'][:8]} on {entry['base_kind']} {entry['base_commit'][:8]} "
   422	                                                  f"({', '.join(item['fix_files'])}). Reason: {entry['reason']}. {answers} "
   423	                                                  f"Continue with {continuation(runtime, 'python')}")
   424	    superseded = candidate_paths(directory, n - 1)[0]
   425	    if superseded.exists():
   426	        runtime.event("candidate", "paused", f"Repair {n} supersedes combined revision {read_json(superseded)['commit'][:8]}; "
   427	                                             f"candidate-{n} is built after the lanes re-verify")
   428	    runtime.event("controller", "running", f"Repair {n} applied: checkpoint forked from {entry['fork_from'][:8]} (after handoff); attempts "
   429	                                           + ", ".join(f"{key} {value}" for key, value in entry["attempt_targets"].items()))
   430	    entry.update(status="applied", applied_at=now(), head_after=head_after)
   431	    save_entry(directory, entry)
   432	    report(runtime, graph.get_state(config))

   435	def apply_repair(runtime, graph, config, lanes: list[str], commit: str, reason: str, dry_run: bool) -> None:
   436	    directory, repo = runtime.directory, Path(runtime.plan["repository"])
   437	    source = resolve_commit(repo, commit)
   438	    entries = load_repairs(directory)
   439	    applied = [entry for entry in entries if entry["status"] == "applied"]
   440	    recorded = next((entry for entry in entries if entry["status"] == "recorded"), None)
   441	    if recorded is None:
   442	        state, blocked, fork, previous = blocked_run(runtime, graph, config, applied)
   443	    elif list(recorded["lanes"]) != lanes or recorded["source_commit"] != source or recorded["reason"] != reason:
   444	        raise RuntimeError(f"Repair {recorded['n']} is recorded but not applied; only the identical command completes it: "
   445	                           f"{repair_command(directory, recorded)}")
   446	    else:
   447	        # Completing a crashed repair: the head may already be its fork, so the recorded step and fork point stand.
   448	        check_before_review(runtime, graph.get_state(config))
   449	        check_source(runtime)
   450	        blocked, previous = recorded["blocked"], effective_snapshots(directory, applied)
   451	        fork = fork_point(graph, config, runtime, previous, recorded["fork_from"])
   452	    derived = derive(runtime, lanes, source, blocked["packets"][0]["phase"], previous)
   453	    if recorded is None:
   454	        targets = attempt_targets(directory, lanes, runtime.workers)
   455	        entry = {"n": len(entries) + 1, "status": "recorded", "mode": "commit", "reason": reason, "recorded_at": now(), "blocked": blocked,
   456	                 "source_commit": source, "base_kind": derived["base_kind"], "base_commit": derived["base_commit"],
   457	                 "expected_candidate_tree": derived["expected_candidate_tree"],
   458	                 "lanes": {lane: {"previous_commit": item["previous_commit"], "fix_files": item["fix_files"]} for lane, item in derived["lanes"].items()},
   459	                 "attempt_targets": targets, "attempt_floors": dict(targets),
   460	                 "fork_from": fork.config["configurable"]["checkpoint_id"], "head_before": state.config["configurable"]["checkpoint_id"]}
   461	    else:
   462	        entry = recorded
   463	        if any(entry[key] != derived[key] for key in ("base_kind", "base_commit", "expected_candidate_tree")) or \
   464	                any(entry["lanes"][lane]["fix_files"] != item["fix_files"] for lane, item in derived["lanes"].items()):
   465	            raise RuntimeError(f"Contradictory run state: repair {entry['n']} no longer derives the snapshots it recorded; inspect")
   466	    if dry_run:
   467	        print(f"Dry run of repair {entry['n']} (nothing written): {reason}\n{describe(entry, derived, dry_run)}\n"
   468	              f"  fork from checkpoint {entry['fork_from']} (after handoff)\nApply: the same command without --dry-run. Then continue with: {continuation(runtime)}")
   469	        return
   470	    if recorded is None:
   471	        record_repair(directory, entry)                                                   # S1
   472	    commit_snapshots(runtime, entry, derived)                                             # S2
   473	    save_lanes(runtime, entry, derived)                                                   # S3
   474	    raise_attempts(directory, entry)                                                      # S4
   475	    snapshots = effective_snapshots(directory, applied + [entry])
   476	    finish_repair(runtime, graph, config, entry, fork_run(graph, config, runtime, entry, fork, snapshots))  # S5, S6
   477	    print(f"Repair {entry['n']} applied: {reason}\n{describe(entry, derived, dry_run)}\n  checkpoint forked from {entry['fork_from']} "
   478	          f"(after handoff); nothing launched, no check run\nContinue with: {continuation(runtime)}")
   479	
   480	
   481	def describe(entry: dict, derived: dict, dry_run: bool) -> str:
   482	    lines = [f"  base: {entry['base_kind']} {entry['base_commit']}"]
   483	    for lane, item in derived["lanes"].items():
   484	        new = f"a commit of tree {item['tree']} on the base" if dry_run else item["commit"]
   485	        lines.append(f"  {lane}: {item['previous_commit']} -> {new}; fix files: {', '.join(item['fix_files'])}; "
   486	                     f"changed files: {', '.join(item['changed_files'])}")
   487	    lines.append("  attempts (each also the floor of its limit): " + ", ".join(f"{key} {value}" for key, value in entry["attempt_targets"].items()))
   488	    return "\n".join(lines)

   497	def make_workspace(runtime, graph, config, lanes: list[str], reason: str | None) -> None:
   498	    """A detached worktree at the base the fix belongs on, with a brief of what blocked the run. No journal, no graph state."""
   499	    directory, repo = runtime.directory, Path(runtime.plan["repository"])
   500	    refuse_recorded(directory)
   501	    applied = applied_repairs(directory)
   502	    _, blocked, _, previous = blocked_run(runtime, graph, config, applied)
   503	    if blocked["packets"][0]["phase"] == "candidate":
   504	        base = read_json(candidate_paths(directory, len(applied))[0])["commit"]
   505	        what = "the combined candidate that failed"
   506	    elif len(lanes) == 1:
   507	        base, what = previous[lanes[0]]["commit"], f"the {lanes[0]} snapshot"
   508	    else:
   509	        raise ValueError("A worker-phase block has no combined candidate: a fix on a lane snapshot repairs one lane only; name one lane")
   510	    number = 1
   511	    while (directory / f"repair-workspace-{number}").exists() or (directory / f"repair-workspace-{number}.brief.md").exists():
   512	        number += 1
   513	    path = directory / f"repair-workspace-{number}"
   514	    git_worktree(repo, "add", "--detach", str(path), base)
   515	    command = (f"{sys.executable} -m workflow repair {shlex.quote(str(directory))} {','.join(lanes)} --commit $(git -C {shlex.quote(str(path))} rev-parse HEAD) "
   516	               f"--reason {shlex.quote(reason or '<why the fix is needed>')}")
   517	    brief = [f"# Repair workspace {number} of run {runtime.plan['run_id']}", "",
   518	             f"Detached at {base}, {what}. Commit the fix here, never on the source branch {runtime.plan['source_branch']}: "
   519	             f"integration needs it at the run's base {runtime.plan['base_commit']}.", "", f"## Blocked: {blocked['step']}"]
   520	    for packet in blocked["packets"]:
   521	        folder = (directory / packet["path"]).parent
   522	        evidence = sorted(folder.glob("setup-*.log")) + sorted(folder.glob("check-*.log")) + sorted(folder.glob("browser-report-*.json"))
   523	        brief += ["", f"{packet['phase']}/{packet['node_id']} attempt {packet['attempt']} at {packet['output_commit']}, gate reasons verbatim:",
   524	                  *(f"- {item}" for item in packet["reasons"]), "", "Evidence:", f"- {directory / packet['path']}", *(f"- {item}" for item in evidence)]
   525	    brief += ["", "## Lanes to repair"]
   526	    for lane in lanes:
   527	        worker = runtime.worker_policy(lane)
   528	        brief += ["", f"{lane} owns: {', '.join(worker['owned_paths'])}. The fix may change only paths the named lanes own.", "Checks:",
   529	                  *(f"- {check['id']} ({check['kind']}): {shlex.join(check['argv'])}" for check in worker["checks"])]
   530	    if any(check["kind"] == "browser" for lane in lanes for check in runtime.worker_policy(lane)["checks"]):
   531	        brief += ["", "## Browser evidence", "", BROWSER_RULES]
   532	    brief += ["", "## Then", "", f"git -C {shlex.quote(str(path))} commit -am '<what the fix does>'", command, ""]
   533	    (directory / f"repair-workspace-{number}.brief.md").write_text("\n".join(brief))
   534	    print(f"Repair workspace {number}: {path}, detached at {base} ({what}).\nBrief: {directory / f'repair-workspace-{number}.brief.md'}\n"
   535	          f"Commit the fix there, never on the source branch, then:\n  {command}")

   538	def repair_main(argv=None):
   539	    parser = argparse.ArgumentParser(prog="python -m workflow repair", description="Turn an operator's fix commit into new snapshots of "
   540	                                     "frozen lanes blocked at their checks or at the combined candidate, before review. Launches nothing and "
   541	                                     "runs no check; automatic --live (or retry for a manual run) re-verifies.")
   542	    parser.add_argument("directory", type=Path)
   543	    parser.add_argument("lanes", help="The lane the fix belongs to, or comma-separated lanes for a fix on the combined candidate")
   544	    parser.add_argument("--commit", metavar="SHA", help="The fix commit, on the failing candidate or on the lane's snapshot")
   545	    parser.add_argument("--reason", help="Why: recorded in the journal, the timeline, the snapshot summary and the reviewer prompt")
   546	    parser.add_argument("--workspace", action="store_true", help="Create a detached worktree at the right base, with a brief, to commit the fix in")
   547	    parser.add_argument("--dry-run", action="store_true", help="Validate and print what --commit would do; write nothing")
   548	    args = parser.parse_args(argv)
   549	    directory = args.directory.resolve()
   550	    try:
   551	        if bool(args.commit) == args.workspace:
   552	            raise ValueError("Give exactly one of --commit <sha> (apply a fix) and --workspace (make a worktree to commit one in)")
   553	        if args.commit and not (args.reason or "").strip():
   554	            raise ValueError("--reason is required with --commit and must not be empty")
   555	        if args.dry_run and args.workspace:
   556	            raise ValueError("--dry-run applies to --commit")
   557	        with run_lock(directory, "automatic-supervisor.lock"), run_lock(directory):
   558	            runtime = Pipeline(directory)
   559	            if "workers" not in runtime.plan:
   560	                raise ValueError("A plan pinned before configured lanes cannot be repaired; start a revised run")
   561	            lanes = parse_lanes(args.lanes, runtime.workers)
   562	            if not (directory / "pipeline.sqlite").exists():
   563	                raise ValueError("The run was never started; there is no frozen lane to repair")
   564	            with SqliteSaver.from_conn_string(str(directory / "pipeline.sqlite")) as saver:
   565	                graph, config = build_pipeline(saver, runtime), graph_config(runtime)
   566	                if args.workspace:
   567	                    make_workspace(runtime, graph, config, lanes, args.reason)
   568	                else:
   569	                    apply_repair(runtime, graph, config, lanes, args.commit, args.reason.strip(), args.dry_run)
   570	    except (ValueError, RuntimeError, OSError, subprocess.SubprocessError) as error:
   571	        parser.exit(1, f"Blocked: {error}\nAll work/evidence retained at {directory}. Nothing launched.\n")
