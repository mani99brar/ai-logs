   110	        from .automatic import advance_failed_checks
   111	        policy = {"max_verification_attempts": 3}
   112	        attempts = {"worker:adapter": 2}
   113	        bumped = []
   114	        runtime = SimpleNamespace(directory=self.root, policy=policy,
   115	                                  attempt=lambda phase, node: attempts.get(f"{phase}:{node}", 1),
   116	                                  retry_check=lambda phase, node: bumped.append((phase, node)))
   117	        state = SimpleNamespace(next=("verify_adapter",), tasks=[SimpleNamespace(name="verify_adapter", error="blocked")])

   128	                  {"gate": {"status": "blocked", "reasons": ["browser: Playwright reported global errors"]}})

   263	    def test_only_an_interrupted_review_of_a_live_session_resumes(self):
   264	        """An error recorded against the review node is not a licence to review again."""
   265	        from .automatic import advance_failed_checks
   266	        live = {"row": None}
   267	        runtime = SimpleNamespace(directory=self.root, plan=self.plan, policy={},
   268	                                  sessions=SimpleNamespace(inventory=lambda: [], locate=lambda node, rows: live["row"]))
   269	        state = SimpleNamespace(next=("review",), tasks=[SimpleNamespace(name="review", error="interrupted")])
   270	        self.assertFalse(advance_failed_checks(runtime, state))  # No reviewer was ever launched.
   271	        save_json(self.root / "automatic-review.json", {"transport": "native", "status": "running"})
   272	        self.assertFalse(advance_failed_checks(runtime, state))  # The session is gone; reconcile instead.

   274	        self.assertTrue(advance_failed_checks(runtime, state))

   276	                        {"transport": "print", "status": "running"}):

   281	        self.assertFalse(advance_failed_checks(runtime, state))

   393	        f = self.fixture
   394	        script = '''import sys
   395	from pathlib import Path
   396	from workflow import automatic
   397	from workflow.sessions import read_json
   398	from workflow.test_pipeline import OfflinePipeline, FakeSessions
   399	directory = Path(sys.argv[1])

   401	sessions.executable = sys.argv[2]

   422	                                    cwd=Path(__file__).resolve().parents[1], capture_output=True, text=True, timeout=90)
   423	            codes.append(result.returncode)

   441	        self.assertEqual(git(f.repo, "rev-parse", self.original_branch), f.plan["base_commit"])
