    22	DEFAULTS = {"finish": "verified-feature-branch", "permission_mode": "bypassPermissions",
    23	            "worker_timeout_seconds": 4 * 3600, "review_timeout_seconds": 1800,
    24	            "reviewer_transport": "native"}
    25	TIMEOUT_KEYS = ("worker_timeout_seconds", "review_timeout_seconds")
    26	TRANSPORTS = ("native", "print")

    31	def automatic_settings(worker_timeout_seconds: int | None = None, review_timeout_seconds: int | None = None,
    32	                       reviewer_transport: str | None = None) -> dict:
    33	    """Run-scoped automatic configuration; deadlines are pinned into plan.json at prepare."""
    34	    settings = dict(DEFAULTS)
    35	    for key, value in (("worker_timeout_seconds", worker_timeout_seconds), ("review_timeout_seconds", review_timeout_seconds),
    36	                       ("reviewer_transport", reviewer_transport)):
    37	        if value is not None:
    38	            settings[key] = value
    39	    validate_automatic({"automatic": settings, "source_branch": "feature/validation-only"})
    40	    return settings

    43	def transport(plan: dict) -> str:
    44	    return plan.get("automatic", {}).get("reviewer_transport", DEFAULTS["reviewer_transport"])

    47	def validate_automatic(plan: dict) -> None:
    48	    settings = plan.get("automatic")
    49	    if not isinstance(settings, dict) or not REQUIRED_SETTINGS <= set(settings) <= set(DEFAULTS):
    50	        raise ValueError("Malformed automatic run configuration")
    51	    if settings["finish"] != DEFAULTS["finish"] or settings["permission_mode"] != "bypassPermissions":
    52	        raise ValueError("Unsupported automatic authority")
    53	    for key in TIMEOUT_KEYS:
    54	        if type(settings[key]) is not int or not 1 <= settings[key] <= 86400:
    55	            raise ValueError("Automatic timeouts must be bounded positive seconds (at most 86400)")
    56	    if transport(plan) not in TRANSPORTS:
    57	        raise ValueError(f"Reviewer transport must be one of {TRANSPORTS}")
    58	    if not plan.get("source_branch", "").startswith("feature/"):
    59	        raise ValueError("Automatic completion is restricted to a feature/ branch")

    62	def completion_prompt(directory: Path, plan: dict, node: str) -> str:
    63	    example = {"version": "1.0.0", "run_id": plan["run_id"], "node_id": node,
    64	               "launch_token": plan["nodes"][node]["session_id"], "status": "completed",
    65	               "summary": "Describe actual work and checks executed", "open_assumptions": []}
    66	    return ("\n\nAUTOMATIC MODE: permission checks are bypassed and Bash is available. "
    67	            "Do not wait for a human handoff. Stay within assigned ownership; do not commit, merge, push, "
    68	            "launch agents, change runtime evidence or switch billing/provider. "
    69	            "On completion write the following JSON shape atomically (temporary file then rename) to "
    70	            f"{directory / (node + '.completion.json')}. This one output file is allowed outside your worktree. "
    71	            "Use status blocked if you cannot finish; never manufacture checks. Write it as your last action, "
    72	            "then finish your turn and do not modify more files. Controller checks and independent review "
    73	            "still determine acceptance.\n" + json.dumps(example))

    76	def read_completion(runtime, node: str) -> dict:
    77	    path = runtime.directory / f"{node}.completion.json"
    78	    if path.is_symlink() or not path.is_file() or path.stat().st_size > 65536:
    79	        raise ValueError(f"Invalid completion file for {node}")
    80	    item = read_json(path)
    81	    expected = {"version", "run_id", "node_id", "launch_token", "status", "summary", "open_assumptions"}
    82	    if not isinstance(item, dict) or set(item) != expected:
    83	        raise ValueError("Malformed completion signal")
    84	    if (item["version"] != "1.0.0" or item["run_id"] != runtime.plan["run_id"] or item["node_id"] != node
    85	            or item["launch_token"] != runtime.plan["nodes"][node]["session_id"]):
    86	        raise ValueError("Stale or foreign worker completion signal")
    87	    if item["status"] not in {"completed", "blocked"} or not isinstance(item["summary"], str) or not item["summary"].strip():
    88	        raise ValueError("Invalid completion status/summary")
    89	    if not isinstance(item["open_assumptions"], list) or any(not isinstance(x, str) for x in item["open_assumptions"]):
    90	        raise ValueError("Invalid completion assumptions")
    91	    if item["status"] == "blocked":
    92	        raise RuntimeError(f"Worker {node} explicitly blocked: {item['summary']}")
    93	    return {"summary": item["summary"], "open_assumptions": item["open_assumptions"]}

    96	def wait_handoffs(runtime, *, clock=time.time, sleep=time.sleep) -> None:
    97	    """Idle alone never means completion. Deadlines survive controller restart."""
    98	    validate_automatic(runtime.plan)
    99	    if any((runtime.directory / f"{node}.stop.json").exists() for node in NODES):
   100	        # Recover a controller crash after durable handoffs/stop intent, before snapshot.
   101	        for node in NODES:
   102	            if read_completion(runtime, node) != read_json(runtime.directory / f"{node}.handoff.json"):
   103	                raise RuntimeError("Handoff changed after stop intent")
   104	        return
   105	    while True:
   106	        rows = runtime.sessions.inventory()
   107	        handoffs = {}
   108	        for node in NODES:
   109	            receipt = read_json(runtime.directory / f"{node}.interactive.json")
   110	            started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   111	            if clock() >= started + runtime.plan["automatic"]["worker_timeout_seconds"]:
   112	                raise RuntimeError(f"Worker {node} deadline exhausted; no automatic relaunch")
   113	            row = runtime.sessions.locate(node, rows)
   114	            if row is None:
   115	                raise RuntimeError("Native worker missing; reconciliation required")
   116	            if row["state"] == "blocked":
   117	                raise RuntimeError(f"Worker {node} blocked; inspect quota or native error. No billing/provider fallback.")
   118	            path = runtime.directory / f"{node}.completion.json"
   119	            if row["state"] in {"idle", "done"} and path.exists():
   120	                handoffs[node] = read_completion(runtime, node)
   121	        if set(handoffs) == set(NODES):
   122	            for node, value in handoffs.items():
   123	                save_json(runtime.directory / f"{node}.handoff.json", value)
   124	            return
   125	        sleep(2)

   176	def read_review_completion(runtime, bundle: dict, digest: str, token: str) -> dict:
   177	    """Schema plus the bindings the schema cannot express. Validity is not acceptance."""
   178	    from .verification import validate_schema
   179	    path = runtime.directory / "review.completion.json"
   180	    if path.is_symlink() or not path.is_file() or path.stat().st_size > 65536:
   181	        raise ValueError("Invalid reviewer completion file")
   182	    item = read_json(path)
   183	    validate_schema("reviewCompletion", item)
   184	    if (item["run_id"] != runtime.plan["run_id"] or item["launch_token"] != token
   185	            or item["bundle_sha256"] != digest or item["candidate_commit"] != bundle["candidate_commit"]):
   186	        raise ValueError("Stale or foreign reviewer completion signal")
   187	    # Cross-field rules mirrored from contracts/workflow/v1.ts.
   188	    if item["verdict"] == "blocked" and not item["findings"]:
   189	        raise ValueError("A blocked verdict must name the findings that caused it")
   190	    if item["verdict"] == "approved" and any(finding["severity"] != "P2" and finding["disposition"] != "resolved"
   191	                                             for finding in item["findings"]):
   192	        raise ValueError("Approval cannot leave an unresolved P0/P1 finding")
   193	    return {"verdict": item["verdict"], "findings": item["findings"]}

   196	def wait_review(runtime, bundle: dict, digest: str, token: str, *, clock=time.time, sleep=time.sleep) -> dict:
   197	    """Idle is not a verdict, for the reviewer as for the workers."""
   198	    validate_automatic(runtime.plan)
   199	    receipt = read_json(runtime.sessions.receipt_path(REVIEWER))
   200	    started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   201	    while True:
   202	        if clock() >= started + runtime.plan["automatic"]["review_timeout_seconds"]:
   203	            raise RuntimeError("Reviewer deadline exhausted; no automatic relaunch and no second reviewer")
   204	        row = runtime.sessions.locate(REVIEWER, runtime.sessions.inventory())
   205	        if row is None:
   206	            raise RuntimeError("Native reviewer missing; reconciliation required")
   207	        if row["state"] == "blocked":
   208	            raise RuntimeError("Reviewer blocked; inspect quota or native error. No billing/provider fallback.")
   209	        if row["state"] in {"idle", "done"} and (runtime.directory / "review.completion.json").exists():
   210	            return read_review_completion(runtime, bundle, digest, token)
   211	        sleep(2)

   214	def review_worktree(runtime, bundle: dict, receipt: dict | None) -> tuple[Path, Path]:
   215	    """Read-only candidate checkout and its diff. A partial allocation is never overwritten."""
   216	    cwd = runtime.directory / "review-worktree"
   217	    patch = runtime.directory / "review.diff"
   218	    if cwd.exists():
   219	        from .pipeline import digest_file
   220	        if receipt is None:
   221	            raise RuntimeError("Partial review worktree exists; reconcile rather than overwrite")
   222	        if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   223	            raise RuntimeError("Reviewer worktree changed")
   224	        if digest_file(patch) != receipt["patch_sha256"]:
   225	            raise RuntimeError("Evidence changed during review")
   226	        return cwd, patch
   227	    subprocess.run(["git", "-C", runtime.plan["repository"], "worktree", "add", "--detach", str(cwd), bundle["candidate_commit"]],
   228	                   check=True, capture_output=True)
   229	    with patch.open("w") as handle:
   230	        subprocess.run(["git", "-C", str(cwd), "diff", "--binary", runtime.plan["base_commit"], bundle["candidate_commit"]], stdout=handle, check=True)
   231	    return cwd, patch

   234	def review_candidate(runtime) -> dict:
   235	    """Called only by the LangGraph review node. Ambiguous invocations never replay."""
   236	    validate_automatic(runtime.plan)
   237	    bundle, digest = runtime.validate_bundle()
   238	    receipt_path = runtime.directory / "automatic-review.json"
   239	    receipt = read_json(receipt_path) if receipt_path.exists() else None
   240	    if receipt is not None:
   241	        if receipt.get("bundle_sha256") != digest:
   242	            raise RuntimeError("Prior reviewer invocation needs reconciliation; no automatic relaunch")
   243	        if receipt.get("status") == "succeeded":
   244	            runtime.validate_review(receipt["review"])
   245	            return receipt["review"]
   246	        # Only an interrupted native review resumes, and only into its own live session.
   247	        if receipt.get("transport") != "native" or receipt.get("status") not in {"launching", "running"}:
   248	            raise RuntimeError("Prior reviewer invocation needs reconciliation; no automatic relaunch")
   249	    if transport(runtime.plan) == "print":
   250	        return print_review(runtime, bundle, digest, receipt_path)
   251	    return native_review(runtime, bundle, digest, receipt_path, receipt)

   254	def native_review(runtime, bundle: dict, digest: str, receipt_path: Path, receipt: dict | None) -> dict:
   255	    """A reviewer session with a worker's lifecycle: pane, human input, completion file."""
   256	    from .pipeline import digest_file
   257	    cwd, patch = review_worktree(runtime, bundle, receipt)
   258	    if receipt is None:
   259	        receipt = {"transport": "native", "launch_token": str(uuid.uuid4()), "session_id": None,
   260	                   "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   261	                   "status": "launching", "patch_sha256": digest_file(patch)}
   262	        save_json(receipt_path, receipt)
   263	    token = receipt["launch_token"]
   264	    prompt = review_instructions(runtime, patch) + review_completion_prompt(runtime, bundle, digest, token)
   265	    try:
   266	        launch = runtime.launch_reviewer(prompt, token, bundle["candidate_commit"])
   267	        receipt.update(status="running", session_id=launch["session_id"], background_id=launch["background_id"])
   268	        save_json(receipt_path, receipt)
   269	        if launch["session_id"] in {item["session_id"] for item in bundle["snapshots"].values()}:
   270	            raise RuntimeError("Reviewer session is not independent of the workers")
   271	        decision = wait_review(runtime, bundle, digest, token)
   272	        if decision["verdict"] != "approved" or any(item["severity"] in {"P0", "P1"} for item in decision["findings"]):
   273	            raise RuntimeError("Independent reviewer blocked the candidate")
   274	        if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   275	            raise RuntimeError("Reviewer worktree changed")
   276	        if runtime.validate_bundle()[1] != digest or digest_file(patch) != receipt["patch_sha256"]:
   277	            raise RuntimeError("Evidence changed during review")
   278	        review = {"run_id": bundle["run_id"], "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   279	                  "reviewer": launch["session_id"], "independent": True, **decision}
   280	        runtime.validate_review(review)
   281	        runtime.stop_reviewer()  # Identity re-checked; the transcript stays resumable.
   282	        receipt.update(status="succeeded", review=review)
   283	        return review
   284	    except KeyboardInterrupt:
   285	        # Operator interruption is not a reviewer failure: the session keeps running
   286	        # and the next controller process resumes this same review.
   287	        raise
   288	    except BaseException as error:
   289	        receipt.update(status="blocked", error=str(error))
   290	        raise
   291	    finally:
   292	        save_json(receipt_path, receipt)

   295	def print_review(runtime, bundle: dict, digest: str, receipt_path: Path) -> dict:
   296	    """Retained non-Herdr transport: one `claude --print` invocation, no pane, no human input."""
   297	    from jsonschema import validate
   298	    from .pipeline import digest_file
   299	    cwd, patch = review_worktree(runtime, bundle, None)
   300	    session_id = str(uuid.uuid4())
   301	    receipt = {"transport": "print", "session_id": session_id, "bundle_sha256": digest,
   302	               "candidate_commit": bundle["candidate_commit"], "status": "launching", "patch_sha256": digest_file(patch)}
   303	    save_json(receipt_path, receipt)
   304	    prompt = review_instructions(runtime, patch) + " Return the requested JSON schema."
   305	    command = [runtime.sessions.executable, "--print", "--output-format", "json", "--session-id", session_id,
   306	               "--safe-mode", "--strict-mcp-config", "--mcp-config", '{"mcpServers":{}}',
   307	               "--tools", "Read,Glob,Grep", "--permission-mode", "dontAsk", "--permission-prompts", "none",
   308	               "--add-dir", str(runtime.directory), "--json-schema", json.dumps(REVIEW_SCHEMA)]
   309	    env = {key: value for key, value in os.environ.items() if not key.startswith("HERDR_")}
   310	    process = None
   311	    try:
   312	        with (runtime.directory / "review.stdout.json").open("w") as output, (runtime.directory / "review.stderr.log").open("w") as errors:
   313	            process = subprocess.Popen(command, cwd=cwd, env=env, stdin=subprocess.PIPE, stdout=output, stderr=errors,
   314	                                       text=True, start_new_session=True)
   315	            receipt.update(status="running", pid=process.pid)
   316	            save_json(receipt_path, receipt)
   317	            process.communicate(prompt, timeout=runtime.plan["automatic"]["review_timeout_seconds"])
   318	        result = read_json(runtime.directory / "review.stdout.json")
   319	        if process.returncode != 0 or result.get("session_id") != session_id or result.get("is_error") is not False or result.get("subtype") != "success":
   320	            raise RuntimeError("Reviewer did not succeed; inspect retained output. No automatic retry/provider switch.")
   321	        decision = result.get("structured_output")
   322	        validate(decision, REVIEW_SCHEMA)
   323	        if decision["verdict"] != "approved" or any(x["severity"] in {"P0", "P1"} for x in decision["findings"]):
   324	            raise RuntimeError("Independent reviewer blocked the candidate")
   325	        if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   326	            raise RuntimeError("Reviewer worktree changed")
   327	        if runtime.validate_bundle()[1] != digest or digest_file(patch) != receipt["patch_sha256"]:
   328	            raise RuntimeError("Evidence changed during review")
   329	        review = {"run_id": bundle["run_id"], "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   330	                  "reviewer": session_id, "independent": True, **decision}
   331	        runtime.validate_review(review)
   332	        receipt.update(status="succeeded", review=review)
   333	        return review
   334	    except BaseException as error:
   335	        if process is not None and process.poll() is None:
   336	            terminate(process)
   337	        receipt.update(status="blocked", error=str(error))
   338	        raise
   339	    finally:
   340	        save_json(receipt_path, receipt)

   343	def resumable_review(runtime) -> bool:
   344	    """An interrupted review resumes into its own live session; a blocked one never does."""
   345	    path = runtime.directory / "automatic-review.json"
   346	    if transport(runtime.plan) != "native" or not path.exists():
   347	        return False
   348	    receipt = read_json(path)
   349	    if receipt.get("transport") != "native" or receipt.get("status") not in {"launching", "running"}:
   350	        return False
   351	    return runtime.sessions.locate(REVIEWER, runtime.sessions.inventory()) is not None

   354	def advance_failed_checks(runtime, state) -> bool:
   355	    """Retry only recorded failing verification packets, never launches or review."""
   356	    # A checkpoint can carry an error from an earlier attempt of a task that has since
   357	    # succeeded (its writes are applied and it is no longer pending). Only pending
   358	    # tasks with errors are failures to classify.
   359	    failures = [task.name for task in state.tasks if task.error and task.name in state.next]
   360	    if not failures:
   361	        return False
   362	    if failures == ["review"]:
   363	        # Nothing is re-run and no attempt is spent: the review node re-enters its
   364	        # own still-running reviewer session and keeps waiting for the same file.
   365	        return resumable_review(runtime)
   366	    if any(name not in {"verify_ui", "verify_adapter", "candidate"} for name in failures):
   367	        return False
   368	    targets = []
   369	    for name in failures:
   370	        phase = "candidate" if name == "candidate" else "worker"
   371	        stage_targets = []
   372	        for node in NODES if phase == "candidate" else (name.removeprefix("verify_"),):
   373	            attempt = runtime.attempt(phase, node)
   374	            path = runtime.directory / "verification" / phase / node / str(attempt) / "packet.json"
   375	            if path.exists() and read_json(path)["gate"]["status"] != "passed":
   376	                previous = runtime.directory / "verification" / phase / node / str(attempt - 1) / "packet.json"
   377	                if attempt > 1 and previous.exists() and read_json(previous)["gate"]["reasons"] == read_json(path)["gate"]["reasons"]:
   378	                    # Retries rerun immutable code; two identical failures mean the cause is
   379	                    # deterministic (code or environment), and more attempts only burn time.
   380	                    raise RuntimeError(f"{phase}/{node} failed identically on attempts {attempt - 1} and {attempt}; "
   381	                                       f"not transient, inspect {path}")
   382	                stage_targets.append((phase, node))
   383	        if not stage_targets:
   384	            return False
   385	        targets.extend(stage_targets)
   386	    if not targets:
   387	        return False
   388	    # Check all bounds before changing any counters.
   389	    if any(runtime.attempt(p, n) >= runtime.policy.get("max_verification_attempts", 3) for p, n in targets):
   390	        raise RuntimeError("Verification retry limit exhausted; work and evidence retained")
   391	    for phase, node in targets:
   392	        runtime.retry_check(phase, node)
   393	    return True

   396	def supervise(directory: Path) -> None:
   397	    """Each recovery uses a new controller process, not just an in-memory replay."""
   398	    validate_automatic(read_json(directory / "plan.json"))
   399	    with run_lock(directory, "automatic-supervisor.lock"):
   400	        for _ in range(45):
   401	            try:
   402	                result = subprocess.run([sys.executable, "-m", "workflow", "automatic-step", str(directory), "--live"],
   403	                                        cwd=Path(__file__).resolve().parents[1])
   404	            except KeyboardInterrupt:
   405	                raise RuntimeError(RESUME_NOTE.format(directory=directory)) from None
   406	            if result.returncode == 0:
   407	                return
   408	            if result.returncode != 75:
   409	                raise RuntimeError(f"Automatic controller blocked (exit {result.returncode}); inspect retained run")
   410	        raise RuntimeError("Automatic controller restart limit exhausted")
   411	
   412	
   413	RESUME_NOTE = ("Supervisor interrupted. Native sessions (workers and any reviewer) were NOT stopped "
   414	               "and keep running; resume with: python -m workflow automatic {directory} --live")
   415	
   416	
   417	def drive(runtime, *, single_step=False) -> str | None:
   418	    """Advance persisted graph state; CLI supervision restarts this process at joins."""
   419	    from .pipeline import build_pipeline, report
   420	    validate_automatic(runtime.plan)
   421	    if git(Path(runtime.plan["repository"]), "symbolic-ref", "--short", "HEAD") != runtime.plan["source_branch"]:
   422	        raise RuntimeError("Source feature branch changed; no automatic continuation")
   423	    runtime.event("controller", "running", f"Automatic checkpoint controller PID {os.getpid()}")
   424	    config = {"configurable": {"thread_id": runtime.plan["run_id"]}, "max_concurrency": 2}
   425	    while True:
   426	        with SqliteSaver.from_conn_string(str(runtime.directory / "pipeline.sqlite")) as saver:
   427	            graph = build_pipeline(saver, runtime)
   428	            state = graph.get_state(config)
   429	            if not state.values or any(name.startswith("launch_") for name in state.next):
   430	                raise RuntimeError("Automatic supervision requires a completed start; reconcile uncertain launches explicitly")
   431	            if not state.next:
   432	                commit = state.values.get("integrated_commit")
   433	                if (not commit or git(Path(runtime.plan["repository"]), "rev-parse", "HEAD") != commit
   434	                        or git(Path(runtime.plan["repository"]), "status", "--porcelain")):
   435	                    raise RuntimeError("No verified feature-branch completion")
   436	                runtime.validate_review(read_json(runtime.directory / "review.json"))
   437	                report(runtime, state)
   438	                return commit
   439	            pending = [item.value.get("kind") for task in state.tasks for item in task.interrupts]
   440	            value = None
   441	            if pending == ["worker_handoff"]:
   442	                try:
   443	                    wait_handoffs(runtime)
   444	                except KeyboardInterrupt:
   445	                    # Operator/terminal interruption is not a worker failure: leave the
   446	                    # native sessions running so `automatic --live` can resume polling.
   447	                    runtime.event("controller", "interrupted", RESUME_NOTE.format(directory=runtime.directory))
   448	                    report(runtime, state)
   449	                    raise
   450	                except BaseException as error:
   451	                    # Deadline, quota block, missing/blocked completion: stop the workers so
   452	                    # no session keeps consuming usage for a run that cannot continue.
   453	                    runtime.event("controller", "blocked", str(error))
   454	                    try:
   455	                        runtime.stop_workers()
   456	                    except Exception as cleanup_error:
   457	                        runtime.event("freeze", "blocked", f"Could not confirm worker stop: {cleanup_error}")
   458	                    report(runtime, state)
   459	                    raise
   460	                value = Command(resume={"freeze": True})
   461	            elif pending:
   462	                raise RuntimeError("Unexpected manual gate in automatic run; inspect state")
   463	            elif any(task.error for task in state.tasks) and not advance_failed_checks(runtime, state):
   464	                raise RuntimeError("Non-retryable graph failure; inspect retained evidence")
   465	            try:
   466	                graph.invoke(value, config)
   467	            except KeyboardInterrupt:
   468	                # Same rule as the worker wait: an interrupted controller never stops a
   469	                # native session. A running reviewer is resumed by the next process.
   470	                runtime.event("controller", "interrupted", RESUME_NOTE.format(directory=runtime.directory))
   471	                raise
   472	            except Exception:
   473	                failed = graph.get_state(config)
   474	                if not any(task.error for task in failed.tasks):
   475	                    raise
   476	                # Next loop reopens the checkpointer and classifies the exact failure.
   477	            finally:
   478	                report(runtime, graph.get_state(config))
   479	            if single_step:
   480	                return None  # Exit 75: supervisor reopens this checkpoint in a fresh process.
