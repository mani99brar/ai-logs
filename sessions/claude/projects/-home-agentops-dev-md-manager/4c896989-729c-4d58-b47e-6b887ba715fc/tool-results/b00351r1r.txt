    22	class FakeSessions:
    23	    """Offline stand-in for InteractiveSessions. Receipts on disk are its only cross-process state."""
    24	
    25	    REVIEWER_UUID = "33333333-3333-4333-8333-333333333333"
    26	
    27	    def __init__(self, directory, plan):
    28	        self.directory, self.plan = directory, plan
    29	        self.starts = []
    30	        self.reviewer_verdict_file = None  # A file whose text is the fake reviewer's verdict (default approved).
    31	        self.reviewer_findings = []        # Findings the fake reviewer reports.
    32	        self.reviewer_mutate = None        # Callable applied to the completion payload before it is written.
    33	        self.reviewer_writes_file = True   # False: the reviewer idles without ever writing a completion file.
    34	        self.reviewer_session_id = self.REVIEWER_UUID  # The native UUID the reviewer session reports (a worker's: not independent).
    35	        self.reviewer_after_file = None    # Callable run right after the completion file is written (dirty the worktree, rewrite the diff).
    36	        self.reviewer_row_after_file = None  # Once the completion file exists: a dict merged into the located reviewer row, or "missing" for None.
    37	
    38	    def native_id(self, node):
    39	        """The UUID the native registry reports for a session; receipts record it, they do not define it."""
    40	        return self.reviewer_session_id if node == "review" else self.plan["nodes"][node]["session_id"]
    41	
    42	    def background_id(self, node):
    43	        return self.REVIEWER_UUID[:8] if node == "review" else f"fake-{node}"
    44	
    45	    def record(self, node):
    46	        self.starts.append(node)
    47	        with (self.directory / "fake-launches.log").open("a") as log:
    48	            log.write(node + "\n")
    49	
    50	    def run(self, node):
    51	        self.record(node)
    52	        cwd = Path(self.plan["nodes"][node]["worktree"])
    53	        (cwd / ("ui.txt" if node == "ui" else "backend.py")).write_text("after" if node == "ui" else "VALUE = 2\n")
    54	        receipt = {"node_id": node, "session_id": self.native_id(node), "launch_token": self.plan["nodes"][node]["session_id"],
    55	                   "background_id": self.background_id(node), "status": "fake-worker-completed", "attempt": 1, "launcher_invocations": 1,
    56	                   "launch_requested_at": now(), "observed_state": "idle", "native_started_at": None}
    57	        save_json(self.directory / f"{node}.interactive.json", receipt)
    58	        save_json(self.directory / f"{node}.handoff.json", {"summary": "Synthetic implementation for offline test", "open_assumptions": []})
    59	        return receipt
    60	
    61	    def run_reviewer(self, prompt, launch_token, candidate_commit):
    62	        self.record("review")
    63	        receipt = {"node_id": "review", "session_id": self.native_id("review"), "launch_token": launch_token, "background_id": self.background_id("review"),
    64	                   "worktree": str(self.directory / "review-worktree"), "candidate_commit": candidate_commit,
    65	                   "status": "attached_session_available", "attempt": 1, "launcher_invocations": 1,
    66	                   "launch_requested_at": now(), "observed_state": "idle", "native_started_at": None}
    67	        save_json(self.directory / "review.interactive.json", receipt)
    68	        (self.directory / "review.prompt.txt").write_text(prompt)
    69	        if self.reviewer_writes_file:
    70	            verdict = Path(self.reviewer_verdict_file).read_text().strip() if self.reviewer_verdict_file else "approved"
    71	            completion = {"version": "1.0.0", "run_id": self.plan["run_id"], "node_id": "review", "launch_token": launch_token,
    72	                          "bundle_sha256": digest_file(self.directory / "review-bundle.json"), "candidate_commit": candidate_commit,
    73	                          "verdict": verdict, "findings": copy.deepcopy(self.reviewer_findings)}
    74	            if self.reviewer_mutate:
    75	                completion = self.reviewer_mutate(completion)
    76	            save_json(self.directory / "review.completion.json", completion)
    77	            if self.reviewer_after_file:
    78	                self.reviewer_after_file()
    79	        return receipt
    80	
    81	    def reconcile(self, node, path, receipt):
    82	        """Mirror of InteractiveSessions.reconcile: bind the one surviving session; never launch."""
    83	        row = self.locate(node, self.inventory())
    84	        if row is None:
    85	            raise RuntimeError("Existing launch cannot be reconciled; no automatic relaunch")
    86	        receipt.update(status="attached_session_available", background_id=row["id"], session_id=row["sessionId"], observed_state=row["state"])
    87	        receipt.pop("error", None)
    88	        save_json(path, receipt)
    89	        return receipt
    90	
    91	    def inventory(self):
    92	        """The native registry: one row per session this fake launched, whatever its receipt recorded so far."""
    93	        return [{"id": self.background_id(node), "sessionId": self.native_id(node), "state": "idle", "pid": os.getpid(), "kind": "background"}
    94	                for node in ("ui", "adapter", "review") if (self.directory / f"{node}.interactive.json").exists()]
    95	
    96	    def locate(self, node, rows):
    97	        if not (self.directory / f"{node}.interactive.json").exists():
    98	            return None
    99	        row = next((row for row in rows if row["id"] == self.background_id(node)), None)
   100	        if node == "review" and row is not None and self.reviewer_row_after_file is not None and (self.directory / "review.completion.json").exists():
   101	            return None if self.reviewer_row_after_file == "missing" else {**row, **self.reviewer_row_after_file}
   102	        return row

   105	class OfflinePipeline(Pipeline):
   106	    def stop_workers(self):
   107	        self.event("freeze", "stopped", "Fake workers have no background processes")
   108	
   109	    def stop_reviewer(self):
   110	        # Fake sessions have no process to stop; record the intent the real path would persist.
   111	        receipt = read_json(self.directory / "review.interactive.json")
   112	        save_json(self.directory / "review.stop.json", {"background_id": receipt["background_id"], "session_id": receipt["session_id"],
   113	                                                        "pid": None, "stopped": True, "synthetic": True})
   114	        self.event("review", "stopped", "Fake reviewer has no background process")

   397	    def test_launch_reviewer_pane_is_best_effort_and_launch_failure_is_recorded(self):
   398	        receipt = {"session_id": "33333333-3333-4333-8333-333333333333", "background_id": "33333333", "status": "attached_session_available"}
   399	        launches = []
   400	        def run_reviewer(prompt, launch_token, candidate_commit):
   401	            launches.append((prompt, launch_token, candidate_commit))
   402	            return dict(receipt)
   403	        self.runtime.sessions = SimpleNamespace(run_reviewer=run_reviewer)
   404	        def review_events():
   405	            return [(event["status"], event["message"]) for event in
   406	                    (json.loads(line) for line in (self.directory / "events.jsonl").read_text().splitlines()) if event["node"] == "review"]
   407	        # No terminals.json: nothing to attach to, no attach attempted.
   408	        with patch.dict(os.environ, {"HERDR_ENV": "1"}), patch("workflow.pipeline.attach_reviewer_panel") as attach:
   409	            self.assertEqual(self.runtime.launch_reviewer("prompt", "token", "c" * 40), receipt)
   410	        attach.assert_not_called()
   411	        self.assertEqual([status for status, _ in review_events()], ["running", "interactive"])
   412	        self.assertIn("33333333-3333-4333-8333-333333333333", review_events()[-1][1])
   413	        # The tab exists but the pane cannot be attached: the launch still succeeds, the reason is on the timeline.
   414	        save_json(self.directory / "terminals.json", {"ui": {"pane_id": "w1:p2"}, "adapter": {"pane_id": "w1:p3"}})
   415	        with patch.dict(os.environ, {"HERDR_ENV": "1"}), patch("workflow.pipeline.attach_reviewer_panel", side_effect=RuntimeError("Pane w1:p3 is occupied")) as attach:
   416	            self.assertEqual(self.runtime.launch_reviewer("prompt", "token", "c" * 40), receipt)
   417	        attach.assert_called_once_with(self.runtime.sessions)
   418	        self.assertIn(("running", "Reviewer pane not attached: Pane w1:p3 is occupied"), review_events())
   419	        # Ctrl-C during the attach: the session was launched; the interruption is recorded and propagated as such.
   420	        with patch.dict(os.environ, {"HERDR_ENV": "1"}), patch("workflow.pipeline.attach_reviewer_panel", side_effect=KeyboardInterrupt):
   421	            with self.assertRaises(KeyboardInterrupt):
   422	                self.runtime.launch_reviewer("prompt", "token", "c" * 40)
   423	        self.assertIn(("running", "Reviewer pane not attached: KeyboardInterrupt"), review_events())
   424	        # Outside a managed Herdr pane no attach is attempted even with the tab mapping present.
   425	        with patch.dict(os.environ, {"HERDR_ENV": "0"}), patch("workflow.pipeline.attach_reviewer_panel") as attach:
   426	            self.runtime.launch_reviewer("prompt", "token", "c" * 40)
   427	        attach.assert_not_called()
   428	        self.assertEqual(len(launches), 4)
   429	        # A launch that fails is a blocked review event; the pane is never touched.
   430	        self.runtime.sessions = SimpleNamespace(run_reviewer=lambda *args: (_ for _ in ()).throw(RuntimeError("Claude background launch exited 1; inspect launch log")))
   431	        with patch.dict(os.environ, {"HERDR_ENV": "1"}), patch("workflow.pipeline.attach_reviewer_panel") as attach:
   432	            with self.assertRaisesRegex(RuntimeError, "exited 1"):
   433	                self.runtime.launch_reviewer("prompt", "token", "c" * 40)
   434	        attach.assert_not_called()
   435	        self.assertEqual(review_events()[-1], ("blocked", "Claude background launch exited 1; inspect launch log"))
