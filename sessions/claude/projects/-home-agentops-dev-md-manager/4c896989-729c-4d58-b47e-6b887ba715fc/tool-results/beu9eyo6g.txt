   110	        from .automatic import advance_failed_checks
   111	        policy = {"max_verification_attempts": 3}
   112	        attempts = {"worker:adapter": 2}
   113	        bumped = []
   114	        runtime = SimpleNamespace(directory=self.root, policy=policy,
   115	                                  attempt=lambda phase, node: attempts.get(f"{phase}:{node}", 1),
   116	                                  retry_check=lambda phase, node: bumped.append((phase, node)))
   117	        state = SimpleNamespace(next=("verify_adapter",), tasks=[SimpleNamespace(name="verify_adapter", error="blocked")])
   118	        reasons = ["backend-unit: exit 1", "backend-unit: no passing test evidence or failed tests"]
   119	        for attempt in (1, 2):
   120	            (self.root / "verification" / "worker" / "adapter" / str(attempt)).mkdir(parents=True)

   124	            advance_failed_checks(runtime, state)
   125	        self.assertEqual(bumped, [])
   126	        # A different failure on the latest attempt is still retried within the cap.
   127	        save_json(self.root / "verification" / "worker" / "adapter" / "2" / "packet.json",
   128	                  {"gate": {"status": "blocked", "reasons": ["browser: Playwright reported global errors"]}})

   130	        self.assertEqual(bumped, [("worker", "adapter")])

   251	    def test_wait_accepts_only_an_idle_reviewer_with_a_file_and_never_relaunches(self):
   252	        from .automatic import wait_review
   253	        ticks = iter([1, 1, 1, DEFAULTS["review_timeout_seconds"] + 1])
   254	        with self.assertRaisesRegex(RuntimeError, "deadline exhausted; no second reviewer"):
   255	            wait_review(self.runtime, clock=lambda: next(ticks), sleep=lambda _: None)
   256	        save_json(self.root / "review.completion.json", self.completion())
   257	        self.rows[0]["state"] = "working"
   258	        ticks = iter([1, 1, DEFAULTS["review_timeout_seconds"] + 1])
   259	        with self.assertRaisesRegex(RuntimeError, "deadline exhausted"):
   260	            wait_review(self.runtime, clock=lambda: next(ticks), sleep=lambda _: None)
   261	        # `blocked` means the session needs a human (a question in its pane): the wait continues
   262	        # until the deadline, recorded once, and never treated as a verdict or a failure.
   263	        self.rows[0]["state"] = "blocked"
   264	        ticks = iter([1, 1, 1, DEFAULTS["review_timeout_seconds"] + 1])
   265	        with self.assertRaisesRegex(RuntimeError, "deadline exhausted"):
   266	            wait_review(self.runtime, clock=lambda: next(ticks), sleep=lambda _: None)
   267	        self.assertEqual([event[1] for event in self.events], ["interactive"])
   268	        self.assertIn("needs attention", self.events[0][2])
   269	        # Answered in the pane, the reviewer finishes: the file is accepted once the session is idle.
   270	        states = iter(["blocked", "blocked", "idle"])
   271	        self.rows[0]["state"] = "blocked"
   272	        def settle(_seconds):
   273	            self.rows[0]["state"] = next(states)
   274	        self.assertEqual(wait_review(self.runtime, clock=lambda: 1, sleep=settle)["verdict"], "approved")
   275	        self.rows[0]["state"] = "done"
   276	        self.assertEqual(wait_review(self.runtime, clock=lambda: 1, sleep=lambda _: self.fail("Unexpected wait"))["verdict"], "approved")
   277	        self.rows.clear()
   278	        with self.assertRaisesRegex(RuntimeError, "missing; reconciliation"):
   279	            wait_review(self.runtime, clock=lambda: 1, sleep=lambda _: None)
