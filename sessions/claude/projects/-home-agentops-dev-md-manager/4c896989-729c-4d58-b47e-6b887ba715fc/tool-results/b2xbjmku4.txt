    22	class FakeReviewerSessions(fixtures.FakeSessions):
    23	    """Offline native reviewer: journals the launch receipt and writes a completion file at once.
    24	
    25	    The completion template lives in a file next to the run so that separate controller processes
    26	    (recovery tests) share it. A missing template means the reviewer idles without a verdict.
    27	    """
    28	
    29	    def __init__(self, directory, plan):
    30	        super().__init__(directory, plan)
    31	        self.reviewer_launches = directory.parent / "reviewer-launches"
    32	        self.template = directory.parent / "review-completion.json"
    33	        self.state = "idle"
    34	
    35	    def run_reviewer(self, prompt, completion_path):
    36	        receipt_path = self.directory / "review.interactive.json"
    37	        if receipt_path.exists():
    38	            receipt = read_json(receipt_path)
    39	            receipt["status"] = "attached_session_available"
    40	            save_json(receipt_path, receipt)
    41	            return receipt
    42	        count = int(self.reviewer_launches.read_text()) + 1 if self.reviewer_launches.exists() else 1
    43	        self.reviewer_launches.write_text(str(count))
    44	        assert "review.completion.json" in prompt and "Schema:" in prompt
    45	        from datetime import datetime, timezone
    46	        from .sessions import plan_digest
    47	        receipt = {"node_id": "review", "session_id": REVIEWER_UUID, "background_id": REVIEWER_UUID[:8], "launch_token": None,
    48	                   "plan_digest": plan_digest(self.plan), "worktree": str(self.directory / "review-worktree"), "role": "reviewer",
    49	                   "status": "attached_session_available", "attempt": 1, "launcher_invocations": 1,
    50	                   "launch_requested_at": datetime.now(timezone.utc).isoformat()}
    51	        save_json(receipt_path, receipt)
    52	        if self.template.exists():
    53	            bundle = read_json(self.directory / "review-bundle.json")
    54	            from .pipeline import digest_file
    55	            completion = {"version": "1.0.0", "run_id": self.plan["run_id"], "node_id": "review",
    56	                          "bundle_sha256": digest_file(self.directory / "review-bundle.json"),
    57	                          "candidate_commit": bundle["candidate_commit"], "reviewer_session": REVIEWER_UUID,
    58	                          **read_json(self.template)}
    59	            save_json(completion_path, completion)
    60	        return receipt
    61	
    62	    def inventory(self):
    63	        if not (self.directory / "review.interactive.json").exists() or (self.directory / "review.stop.json").exists():
    64	            return []
    65	        return [{"id": REVIEWER_UUID[:8], "sessionId": REVIEWER_UUID, "state": self.state, "pid": 4242,
    66	                 "name": f"workflow-{self.plan['run_id']}-reviewer", "kind": "background", "cwd": str(self.directory / "review-worktree")}]
    67	
    68	    def locate(self, node, rows):
    69	        if node != "review":
    70	            return None
    71	        return next((row for row in rows if row["sessionId"] == REVIEWER_UUID), None)

    74	class OfflineNativePipeline(fixtures.OfflinePipeline):
    75	    def stop_session(self, node):
    76	        marker = self.directory / f"{node}.stop.json"
    77	        if marker.exists():
    78	            return read_json(marker)
    79	        row = self.sessions.locate(node, self.sessions.inventory())
    80	        if row is None:
    81	            raise RuntimeError(f"{node} session missing before stop; reconcile before continuing")
    82	        intent = {"background_id": row["id"], "session_id": row["sessionId"], "pid": row["pid"], "stopped": True}
    83	        save_json(marker, intent)
    84	        return intent

   457	    def test_recovery_in_new_controller_processes_including_a_reviewer_interruption(self):
   458	        f = self.fixture
   459	        interrupt_marker = f.root / "review-interrupted-once"
   460	        script = '''import sys
   461	from pathlib import Path
   462	from workflow import automatic
   463	from workflow.sessions import read_json
   464	from workflow.test_automatic import FakeReviewerSessions, OfflineNativePipeline
   465	directory = Path(sys.argv[1])
   466	sessions = FakeReviewerSessions(directory, read_json(directory / 'plan.json'))
   467	runtime = OfflineNativePipeline(directory, sessions)
   468	automatic.wait_handoffs = lambda runtime: None
   469	marker = Path(sys.argv[2])
   470	real_wait = automatic.wait_review
   471	def wait_once(runtime, digest, bundle, **kwargs):
   472	    if not marker.exists():
   473	        marker.write_text('interrupted')
   474	        raise KeyboardInterrupt
   475	    return real_wait(runtime, digest, bundle, **kwargs)
   476	automatic.wait_review = wait_once
   477	try:
   478	    commit = automatic.drive(runtime, single_step=True)
   479	except KeyboardInterrupt:
   480	    sys.exit(130)
   481	assert not sessions.starts, 'Restart launched workers again'
   482	sys.exit(0 if commit else 75)
   483	'''
   484	        codes = []
   485	        for _ in range(6):
   486	            result = subprocess.run([sys.executable, "-c", script, str(f.directory), str(interrupt_marker)],
   487	                                    cwd=Path(__file__).resolve().parents[1], capture_output=True, text=True, timeout=90)
   488	            codes.append(result.returncode)
   489	            self.assertIn(result.returncode, (0, 75, 130), result.stderr)
   490	            if result.returncode == 130:
   491	                self.assertEqual(read_json(f.directory / "automatic-review.json")["status"], "running")
   492	                self.assertFalse((f.directory / "review.stop.json").exists(), "interruption must not stop the reviewer")
   493	            if result.returncode == 0:
   494	                break
   495	        # Persisted steps continue in fresh processes; exactly one of them was interrupted mid-review.
   496	        self.assertEqual((len(codes), codes.count(130), codes[-1]), (4, 1, 0), codes)
   497	        self.assertEqual(self.sessions.reviewer_launches.read_text(), "1")
   498	        self.assertEqual(read_json(f.directory / "review.json")["reviewer"], REVIEWER_UUID)
   499	        self.assertTrue(read_json(f.directory / "review.stop.json")["stopped"])
   500	        self.assertEqual(git(f.repo, "rev-parse", self.original_branch), f.plan["base_commit"])
