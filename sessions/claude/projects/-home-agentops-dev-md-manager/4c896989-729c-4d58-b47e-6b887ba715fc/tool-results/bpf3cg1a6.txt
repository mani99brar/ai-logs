    31	def automatic_settings(worker_timeout_seconds: int | None = None, review_timeout_seconds: int | None = None,
    32	                       reviewer_transport: str | None = None) -> dict:
    33	    """Run-scoped automatic configuration; deadlines are pinned into plan.json at prepare."""
    34	    settings = dict(DEFAULTS)
    35	    for key, value in (("worker_timeout_seconds", worker_timeout_seconds), ("review_timeout_seconds", review_timeout_seconds),
    36	                       ("reviewer_transport", reviewer_transport)):
    37	        if value is not None:
    38	            settings[key] = value
    39	    validate_automatic({"automatic": settings, "source_branch": "feature/validation-only"})
    40	    return settings

    43	def reviewer_transport(plan: dict) -> str:
    44	    # Runs prepared before the native reviewer existed carry no transport key; they were reviewed in print mode.
    45	    return plan["automatic"].get("reviewer_transport", "print")

    48	def validate_automatic(plan: dict) -> None:
    49	    settings = plan.get("automatic")
    50	    if not isinstance(settings, dict) or set(settings) - {"reviewer_transport"} != set(DEFAULTS) - {"reviewer_transport"}:
    51	        raise ValueError("Malformed automatic run configuration")
    52	    if settings["finish"] != DEFAULTS["finish"] or settings["permission_mode"] != "bypassPermissions":
    53	        raise ValueError("Unsupported automatic authority")
    54	    if settings.get("reviewer_transport", "print") not in TRANSPORTS:
    55	        raise ValueError("Unsupported reviewer transport")
    56	    for key in TIMEOUT_KEYS:
    57	        if type(settings[key]) is not int or not 1 <= settings[key] <= 86400:
    58	            raise ValueError("Automatic timeouts must be bounded positive seconds (at most 86400)")
    59	    if not plan.get("source_branch", "").startswith("feature/"):
    60	        raise ValueError("Automatic completion is restricted to a feature/ branch")

    63	def completion_prompt(directory: Path, plan: dict, node: str) -> str:
    64	    example = {"version": "1.0.0", "run_id": plan["run_id"], "node_id": node,
    65	               "launch_token": plan["nodes"][node]["session_id"], "status": "completed",
    66	               "summary": "Describe actual work and checks executed", "open_assumptions": []}
    67	    return ("\n\nAUTOMATIC MODE: permission checks are bypassed and Bash is available. "
    68	            "Do not wait for a human handoff. Stay within assigned ownership; do not commit, merge, push, "
    69	            "launch agents, change runtime evidence or switch billing/provider. "
    70	            "On completion write the following JSON shape atomically (temporary file then rename) to "
    71	            f"{directory / (node + '.completion.json')}. This one output file is allowed outside your worktree. "
    72	            "Use status blocked if you cannot finish; never manufacture checks. Write it as your last action, "
    73	            "then finish your turn and do not modify more files. Controller checks and independent review "
    74	            "still determine acceptance.\n" + json.dumps(example))

    77	def read_completion(runtime, node: str) -> dict:
    78	    path = runtime.directory / f"{node}.completion.json"
    79	    if path.is_symlink() or not path.is_file() or path.stat().st_size > 65536:
    80	        raise ValueError(f"Invalid completion file for {node}")
    81	    item = read_json(path)
    82	    expected = {"version", "run_id", "node_id", "launch_token", "status", "summary", "open_assumptions"}
    83	    if not isinstance(item, dict) or set(item) != expected:
    84	        raise ValueError("Malformed completion signal")
    85	    if (item["version"] != "1.0.0" or item["run_id"] != runtime.plan["run_id"] or item["node_id"] != node
    86	            or item["launch_token"] != runtime.plan["nodes"][node]["session_id"]):
    87	        raise ValueError("Stale or foreign worker completion signal")
    88	    if item["status"] not in {"completed", "blocked"} or not isinstance(item["summary"], str) or not item["summary"].strip():
    89	        raise ValueError("Invalid completion status/summary")
    90	    if not isinstance(item["open_assumptions"], list) or any(not isinstance(x, str) for x in item["open_assumptions"]):
    91	        raise ValueError("Invalid completion assumptions")
    92	    if item["status"] == "blocked":
    93	        raise RuntimeError(f"Worker {node} explicitly blocked: {item['summary']}")
    94	    return {"summary": item["summary"], "open_assumptions": item["open_assumptions"]}

    97	def wait_handoffs(runtime, *, clock=time.time, sleep=time.sleep) -> None:
    98	    """Idle alone never means completion. Deadlines survive controller restart."""
    99	    validate_automatic(runtime.plan)
   100	    if any((runtime.directory / f"{node}.stop.json").exists() for node in NODES):
   101	        # Recover a controller crash after durable handoffs/stop intent, before snapshot.
   102	        for node in NODES:
   103	            if read_completion(runtime, node) != read_json(runtime.directory / f"{node}.handoff.json"):
   104	                raise RuntimeError("Handoff changed after stop intent")
   105	        return
   106	    while True:
   107	        rows = runtime.sessions.inventory()
   108	        handoffs = {}
   109	        for node in NODES:
   110	            receipt = read_json(runtime.directory / f"{node}.interactive.json")
   111	            started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   112	            if clock() >= started + runtime.plan["automatic"]["worker_timeout_seconds"]:
   113	                raise RuntimeError(f"Worker {node} deadline exhausted; no automatic relaunch")
   114	            row = runtime.sessions.locate(node, rows)
   115	            if row is None:
   116	                raise RuntimeError("Native worker missing; reconciliation required")
   117	            if row["state"] == "blocked":
   118	                raise RuntimeError(f"Worker {node} blocked; inspect quota or native error. No billing/provider fallback.")
   119	            path = runtime.directory / f"{node}.completion.json"
   120	            if row["state"] in {"idle", "done"} and path.exists():
   121	                handoffs[node] = read_completion(runtime, node)
   122	        if set(handoffs) == set(NODES):
   123	            for node, value in handoffs.items():
   124	                save_json(runtime.directory / f"{node}.handoff.json", value)
   125	            return
   126	        sleep(2)

   129	def completion_schema() -> dict:
   130	    """The committed reviewer completion-file schema, shared with the reviewer prompt."""
   131	    return json.loads((CONTRACTS / "reviewCompletion.schema.json").read_text())

   134	def structured_review_schema() -> dict:
   135	    """Print-mode structured output: the verdict and findings exactly as the completion file defines them."""
   136	    schema = completion_schema()
   137	    return {"type": "object", "additionalProperties": False, "required": ["verdict", "findings"],
   138	            "properties": {key: schema["properties"][key] for key in ("verdict", "findings")}}

   156	def native_review_prompt(runtime, patch: Path, bundle: dict, digest: str, completion_path: Path) -> str:
   157	    example = {"version": "1.0.0", "run_id": runtime.plan["run_id"], "node_id": REVIEW_NODE,
   158	               "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   159	               "reviewer_session": "<your own session UUID as shown by `claude agents`; the controller verifies it>",
   160	               "verdict": "approved | blocked",
   161	               "findings": [{"severity": "P2", "message": "...", "disposition": "open", "worker": "adapter", "requirement": "verbatim quote or null"}]}
   162	    return (review_prompt(runtime, patch) +
   163	            "\n\nREVIEW SESSION: a human may type into this terminal to answer your questions; the transcript is the record, "
   164	            "the verdict is only the completion file. Your tools are Read, Glob and Grep, plus Write permitted for exactly one file. "
   165	            f"When your review is complete, write your verdict as JSON matching this schema to {completion_path} "
   166	            "(that path is the only file you may write), then finish your turn. The controller accepts the file only if it "
   167	            "validates, names this exact bundle hash and candidate commit, and carries your session UUID; it stops this session "
   168	            "afterwards. If you finish a turn without the file the controller keeps waiting until its deadline, then stops "
   169	            "without a verdict; no second reviewer is launched.\n"
   170	            f"Schema: {json.dumps(completion_schema())}\nExample: {json.dumps(example)}")

   173	def read_review_completion(runtime, digest: str, bundle: dict, session_id: str) -> dict:
   174	    """Fail closed on anything but a schema-valid file bound to this bundle, candidate and reviewer."""
   175	    from jsonschema import validate
   176	    path = runtime.directory / f"{REVIEW_NODE}.completion.json"
   177	    if path.is_symlink() or not path.is_file() or path.stat().st_size > 262144:
   178	        raise ValueError("Invalid review completion file")
   179	    item = read_json(path)
   180	    validate(item, completion_schema())
   181	    if item["run_id"] != runtime.plan["run_id"] or item["bundle_sha256"] != digest or item["candidate_commit"] != bundle["candidate_commit"]:
   182	        raise ValueError("Review completion does not reference this exact run, bundle hash and candidate")
   183	    worker_ids = {snapshot["session_id"] for snapshot in bundle["snapshots"].values()}
   184	    if item["reviewer_session"] != session_id or session_id in worker_ids:
   185	        raise ValueError("Review completion is not from the independent reviewer session launched for this run")
   186	    return item

   189	def wait_review(runtime, digest: str, bundle: dict, *, clock=time.time, sleep=time.sleep) -> dict:
   190	    """Idle without a completion file is not a verdict; the deadline counts from the reviewer's launch."""
   191	    validate_automatic(runtime.plan)
   192	    receipt = read_json(runtime.directory / f"{REVIEW_NODE}.interactive.json")
   193	    started = datetime.fromisoformat(receipt["launch_requested_at"]).timestamp()
   194	    path = runtime.directory / f"{REVIEW_NODE}.completion.json"
   195	    stop = runtime.directory / f"{REVIEW_NODE}.stop.json"
   196	    if stop.exists() and read_json(stop)["stopped"]:
   197	        # Recovered after acceptance and stop: the file must still bind to the stopped session.
   198	        return read_review_completion(runtime, digest, bundle, read_json(stop)["session_id"])
   199	    noted_blocked = False
   200	    while True:
   201	        if clock() >= started + runtime.plan["automatic"]["review_timeout_seconds"]:
   202	            raise RuntimeError("Reviewer deadline exhausted without a completion file; no second reviewer is launched")
   203	        row = runtime.sessions.locate(REVIEW_NODE, runtime.sessions.inventory())
   204	        if row is None:
   205	            raise RuntimeError("Native reviewer missing; reconciliation required")
   206	        # `blocked` is the reviewer waiting for a person: a question, or a refused action it reports.
   207	        # The operator can answer in the pane, so it is a waiting state until the deadline, not a failure.
   208	        if row["state"] == "blocked" and not noted_blocked:
   209	            noted_blocked = True
   210	            runtime.event(REVIEW_NODE, "interactive", "Reviewer is waiting for input in its pane; answer there or let the deadline stop it")
   211	        if row["state"] in {"idle", "done", "blocked"} and path.exists():
   212	            return read_review_completion(runtime, digest, bundle, row["sessionId"])
   213	        sleep(2)

   216	def review_resumable(runtime, state) -> bool:
   217	    """A review node interrupted mid-wait resumes against the same running reviewer; nothing relaunches."""
   218	    failed = [task.name for task in state.tasks if task.error and task.name in state.next]
   219	    if failed != [REVIEW_NODE]:
   220	        return False
   221	    receipt_path = runtime.directory / "automatic-review.json"
   222	    if not receipt_path.exists():
   223	        return False
   224	    receipt = read_json(receipt_path)
   225	    return receipt.get("transport") == "native" and receipt.get("status") == "running"

   228	def review_candidate(runtime) -> dict:
   229	    """Called only by the LangGraph review node. Ambiguous invocations never replay."""
   230	    from .pipeline import digest_file
   231	    validate_automatic(runtime.plan)
   232	    transport = reviewer_transport(runtime.plan)
   233	    bundle, digest = runtime.validate_bundle()
   234	    receipt_path = runtime.directory / "automatic-review.json"
   235	    cwd = runtime.directory / "review-worktree"
   236	    patch = runtime.directory / "review.diff"
   237	    if receipt_path.exists():
   238	        receipt = read_json(receipt_path)
   239	        if receipt.get("bundle_sha256") != digest or receipt.get("candidate_commit") != bundle["candidate_commit"]:
   240	            raise RuntimeError("Prior reviewer invocation reviewed different evidence; reconcile, no automatic relaunch")
   241	        if receipt.get("status") == "succeeded":
   242	            runtime.validate_review(receipt["review"])
   243	            return receipt["review"]
   244	        resumable = receipt.get("transport") == "native" and receipt.get("status") in {"launching", "running"}
   245	        if not resumable:
   246	            raise RuntimeError("Prior reviewer invocation needs reconciliation; no automatic relaunch")
   247	        if not cwd.is_dir() or git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or digest_file(patch) != receipt["patch_sha256"]:
   248	            raise RuntimeError("Review worktree or diff changed since the interrupted review; reconcile manually")
   249	    else:
   250	        if cwd.exists():
   251	            raise RuntimeError("Partial review worktree exists; reconcile rather than overwrite")
   252	        subprocess.run(["git", "-C", runtime.plan["repository"], "worktree", "add", "--detach", str(cwd), bundle["candidate_commit"]],
   253	                       check=True, capture_output=True)
   254	        with patch.open("w") as handle:
   255	            subprocess.run(["git", "-C", str(cwd), "diff", "--binary", runtime.plan["base_commit"], bundle["candidate_commit"]], stdout=handle, check=True)
   256	        receipt = {"transport": transport, "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   257	                   "status": "launching", "patch_sha256": digest_file(patch)}
   258	        save_json(receipt_path, receipt)
   259	    if transport == "print":
   260	        return review_print(runtime, receipt, receipt_path, cwd, patch, bundle, digest)
   261	    return review_native(runtime, receipt, receipt_path, cwd, patch, bundle, digest)

   264	def finish_review(runtime, receipt: dict, cwd: Path, patch: Path, bundle: dict, digest: str, reviewer: str, decision: dict) -> dict:
   265	    """Shared acceptance: evidence unchanged, record persisted for either verdict, approval validated."""
   266	    from .pipeline import digest_file
   267	    if git(cwd, "rev-parse", "HEAD") != bundle["candidate_commit"] or git(cwd, "status", "--porcelain"):
   268	        raise RuntimeError("Reviewer worktree changed")
   269	    if runtime.validate_bundle()[1] != digest or digest_file(patch) != receipt["patch_sha256"]:
   270	        raise RuntimeError("Evidence changed during review")
   271	    review = {"run_id": bundle["run_id"], "bundle_sha256": digest, "candidate_commit": bundle["candidate_commit"],
   272	              "reviewer": reviewer, "independent": True, "verdict": decision["verdict"], "findings": decision["findings"]}
   273	    # The record is kept for a blocked verdict too: the viewer shows why the run stopped.
   274	    save_json(runtime.directory / "review.json", review)
   275	    if review["verdict"] != "approved" or any(x["severity"] in {"P0", "P1"} and x["disposition"] != "resolved" for x in review["findings"]):
   276	        raise RuntimeError("Independent reviewer blocked the candidate")
   277	    runtime.validate_review(review)
   278	    receipt.update(status="succeeded", review=review)
   279	    return review

   282	def review_native(runtime, receipt: dict, receipt_path: Path, cwd: Path, patch: Path, bundle: dict, digest: str) -> dict:
   283	    completion_path = runtime.directory / f"{REVIEW_NODE}.completion.json"
   284	    try:
   285	        prompt = native_review_prompt(runtime, patch, bundle, digest, completion_path)
   286	        launch = runtime.sessions.run_reviewer(prompt, completion_path)
   287	        receipt.update(status="running", session_id=launch["session_id"], background_id=launch.get("background_id"))
   288	        save_json(receipt_path, receipt)
   289	        runtime.event(REVIEW_NODE, "interactive", f"Reviewer session {launch['session_id']} launched in review-worktree; awaiting completion file, idle is not a verdict")
   290	        try:
   291	            mapping = attach_reviewer_pane(runtime.sessions)
   292	        except Exception as error:  # The pane is a window onto the review, not part of the verdict.
   293	            runtime.event(REVIEW_NODE, "interactive", f"Reviewer pane not attached: {error}. Attach manually with workflow.interactive attach-one --node review")
   294	        else:
   295	            if mapping is not None:
   296	                runtime.event(REVIEW_NODE, "interactive", f"Reviewer pane {mapping[REVIEW_NODE]['pane_id']} attached in the workflow tab")
   297	        decision = wait_review(runtime, digest, bundle)
   298	        intent = runtime.stop_session(REVIEW_NODE)
   299	        if intent["session_id"] != decision["reviewer_session"]:
   300	            raise RuntimeError("Reviewer identity changed between acceptance and stop; reconcile manually")
   301	        runtime.event(REVIEW_NODE, "stopped", f"Reviewer session {intent['session_id']} stopped after its completion file was accepted; transcript remains resumable")
   302	        return finish_review(runtime, receipt, cwd, patch, bundle, digest, decision["reviewer_session"], decision)
   303	    except KeyboardInterrupt:
   304	        # Operator interruption is not a review failure: the reviewer keeps running and `automatic --live` resumes the wait.
   305	        runtime.event(REVIEW_NODE, "interrupted", RESUME_NOTE.format(directory=runtime.directory))
   306	        raise
   307	    except BaseException as error:
   308	        receipt.update(status="blocked", error=str(error))
   309	        if (runtime.directory / f"{REVIEW_NODE}.interactive.json").exists():
   310	            try:
   311	                runtime.stop_session(REVIEW_NODE)
   312	            except Exception as cleanup_error:
   313	                runtime.event(REVIEW_NODE, "blocked", f"Could not confirm reviewer stop: {cleanup_error}")
   314	        raise
   315	    finally:
   316	        save_json(receipt_path, receipt)

   319	def review_print(runtime, receipt: dict, receipt_path: Path, cwd: Path, patch: Path, bundle: dict, digest: str) -> dict:
   320	    """Headless fallback for environments without an attachable terminal; identical acceptance rules."""
   321	    from jsonschema import validate
   322	    if receipt["status"] != "launching":
   323	        raise RuntimeError("Prior reviewer invocation needs reconciliation; no automatic relaunch")
   324	    session_id = str(uuid.uuid4())
   325	    receipt.update(session_id=session_id)
   326	    save_json(receipt_path, receipt)
   327	    schema = structured_review_schema()
   328	    command = [runtime.sessions.executable, "--print", "--output-format", "json", "--session-id", session_id,
   329	               "--safe-mode", "--strict-mcp-config", "--mcp-config", '{"mcpServers":{}}',
   330	               "--tools", "Read,Glob,Grep", "--permission-mode", "dontAsk", "--permission-prompts", "none",
   331	               "--add-dir", str(runtime.directory), "--json-schema", json.dumps(schema)]
   332	    env = {key: value for key, value in os.environ.items() if not key.startswith("HERDR_")}
   333	    process = None
   334	    try:
   335	        with (runtime.directory / "review.stdout.json").open("w") as output, (runtime.directory / "review.stderr.log").open("w") as errors:
   336	            process = subprocess.Popen(command, cwd=cwd, env=env, stdin=subprocess.PIPE, stdout=output, stderr=errors,
   337	                                       text=True, start_new_session=True)
   338	            receipt.update(status="running", pid=process.pid)
   339	            save_json(receipt_path, receipt)
   340	            process.communicate(review_prompt(runtime, patch), timeout=runtime.plan["automatic"]["review_timeout_seconds"])
   341	        result = read_json(runtime.directory / "review.stdout.json")
   342	        if process.returncode != 0 or result.get("session_id") != session_id or result.get("is_error") is not False or result.get("subtype") != "success":
   343	            raise RuntimeError("Reviewer did not succeed; inspect retained output. No automatic retry/provider switch.")
   344	        decision = result.get("structured_output")
   345	        validate(decision, schema)
   346	        return finish_review(runtime, receipt, cwd, patch, bundle, digest, session_id, decision)
   347	    except BaseException as error:
   348	        if process is not None and process.poll() is None:
   349	            terminate(process)
   350	        receipt.update(status="blocked", error=str(error))
   351	        raise
   352	    finally:
   353	        save_json(receipt_path, receipt)

   356	def advance_failed_checks(runtime, state) -> bool:
   357	    """Retry only recorded failing verification packets, never launches or review."""
   358	    # A checkpoint can carry an error from an earlier attempt of a task that has since
   359	    # succeeded (its writes are applied and it is no longer pending). Only pending
   360	    # tasks with errors are failures to classify.
   361	    failures = [task.name for task in state.tasks if task.error and task.name in state.next]
   362	    if not failures or any(name not in {"verify_ui", "verify_adapter", "candidate"} for name in failures):
   363	        return False
   364	    targets = []
   365	    for name in failures:
   366	        phase = "candidate" if name == "candidate" else "worker"
   367	        stage_targets = []
   368	        for node in NODES if phase == "candidate" else (name.removeprefix("verify_"),):
   369	            attempt = runtime.attempt(phase, node)
   370	            path = runtime.directory / "verification" / phase / node / str(attempt) / "packet.json"
   371	            if path.exists() and read_json(path)["gate"]["status"] != "passed":
   372	                previous = runtime.directory / "verification" / phase / node / str(attempt - 1) / "packet.json"
   373	                if attempt > 1 and previous.exists() and read_json(previous)["gate"]["reasons"] == read_json(path)["gate"]["reasons"]:
   374	                    # Retries rerun immutable code; two identical failures mean the cause is
   375	                    # deterministic (code or environment), and more attempts only burn time.
   376	                    raise RuntimeError(f"{phase}/{node} failed identically on attempts {attempt - 1} and {attempt}; "
   377	                                       f"not transient, inspect {path}")
   378	                stage_targets.append((phase, node))
   379	        if not stage_targets:
   380	            return False
   381	        targets.extend(stage_targets)
   382	    if not targets:
   383	        return False
   384	    # Check all bounds before changing any counters.
   385	    if any(runtime.attempt(p, n) >= runtime.policy.get("max_verification_attempts", 3) for p, n in targets):
   386	        raise RuntimeError("Verification retry limit exhausted; work and evidence retained")
   387	    for phase, node in targets:
   388	        runtime.retry_check(phase, node)
   389	    return True

   392	def supervise(directory: Path) -> None:
   393	    """Each recovery uses a new controller process, not just an in-memory replay."""
   394	    validate_automatic(read_json(directory / "plan.json"))
   395	    with run_lock(directory, "automatic-supervisor.lock"):
   396	        for _ in range(45):
   397	            try:
   398	                result = subprocess.run([sys.executable, "-m", "workflow", "automatic-step", str(directory), "--live"],
   399	                                        cwd=Path(__file__).resolve().parents[1])
   400	            except KeyboardInterrupt:
   401	                raise RuntimeError(RESUME_NOTE.format(directory=directory)) from None
   402	            if result.returncode == 0:
   403	                return
   404	            if result.returncode != 75:
   405	                raise RuntimeError(f"Automatic controller blocked (exit {result.returncode}); inspect retained run")
   406	        raise RuntimeError("Automatic controller restart limit exhausted")
   407	
   408	
   409	RESUME_NOTE = ("Supervisor interrupted. Native workers were NOT stopped and keep running; "
   410	               "resume with: python -m workflow automatic {directory} --live")
   411	
   412	
   413	def drive(runtime, *, single_step=False) -> str | None:
   414	    """Advance persisted graph state; CLI supervision restarts this process at joins."""
   415	    from .pipeline import build_pipeline, report
   416	    validate_automatic(runtime.plan)
   417	    if git(Path(runtime.plan["repository"]), "symbolic-ref", "--short", "HEAD") != runtime.plan["source_branch"]:
   418	        raise RuntimeError("Source feature branch changed; no automatic continuation")
   419	    runtime.event("controller", "running", f"Automatic checkpoint controller PID {os.getpid()}")
   420	    config = {"configurable": {"thread_id": runtime.plan["run_id"]}, "max_concurrency": 2}
   421	    while True:
   422	        with SqliteSaver.from_conn_string(str(runtime.directory / "pipeline.sqlite")) as saver:
   423	            graph = build_pipeline(saver, runtime)
   424	            state = graph.get_state(config)
   425	            if not state.values or any(name.startswith("launch_") for name in state.next):
   426	                raise RuntimeError("Automatic supervision requires a completed start; reconcile uncertain launches explicitly")
   427	            if not state.next:
   428	                commit = state.values.get("integrated_commit")
   429	                if (not commit or git(Path(runtime.plan["repository"]), "rev-parse", "HEAD") != commit
   430	                        or git(Path(runtime.plan["repository"]), "status", "--porcelain")):
   431	                    raise RuntimeError("No verified feature-branch completion")
   432	                runtime.validate_review(read_json(runtime.directory / "review.json"))
   433	                report(runtime, state)
   434	                return commit
   435	            pending = [item.value.get("kind") for task in state.tasks for item in task.interrupts]
   436	            value = None
   437	            if pending == ["worker_handoff"]:
   438	                try:
   439	                    wait_handoffs(runtime)
   440	                except KeyboardInterrupt:
   441	                    # Operator/terminal interruption is not a worker failure: leave the
   442	                    # native sessions running so `automatic --live` can resume polling.
   443	                    runtime.event("controller", "interrupted", RESUME_NOTE.format(directory=runtime.directory))
   444	                    report(runtime, state)
   445	                    raise
   446	                except BaseException as error:
   447	                    # Deadline, quota block, missing/blocked completion: stop the workers so
   448	                    # no session keeps consuming usage for a run that cannot continue.
   449	                    runtime.event("controller", "blocked", str(error))
   450	                    try:
   451	                        runtime.stop_workers()
   452	                    except Exception as cleanup_error:
   453	                        runtime.event("freeze", "blocked", f"Could not confirm worker stop: {cleanup_error}")
   454	                    report(runtime, state)
   455	                    raise
   456	                value = Command(resume={"freeze": True})
   457	            elif pending:
   458	                raise RuntimeError("Unexpected manual gate in automatic run; inspect state")
   459	            elif any(task.error for task in state.tasks) and not advance_failed_checks(runtime, state) and not review_resumable(runtime, state):
   460	                raise RuntimeError("Non-retryable graph failure; inspect retained evidence")
   461	            try:
   462	                graph.invoke(value, config)
   463	            except Exception:
   464	                failed = graph.get_state(config)
   465	                if not any(task.error for task in failed.tasks):
   466	                    raise
   467	                # Next loop reopens the checkpointer and classifies the exact failure.
   468	            finally:
   469	                report(runtime, graph.get_state(config))
   470	            if single_step:
   471	                return None  # Exit 75: supervisor reopens this checkpoint in a fresh process.
