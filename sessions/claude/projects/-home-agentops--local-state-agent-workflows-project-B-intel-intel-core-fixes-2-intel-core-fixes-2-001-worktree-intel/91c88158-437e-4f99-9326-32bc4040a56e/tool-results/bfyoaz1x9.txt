    67	/** Recursively frozen, so a step that mutated its input would throw. */
    68	function deepFreeze<T>(value: T): T {
    69	  if (typeof value === "object" && value !== null && !Object.isFrozen(value)) {
    70	    Object.freeze(value);
    71	    for (const child of Object.values(value)) deepFreeze(child);
    72	  }
    73	  return value;
    74	}

    90	  step(tick: number, world: World, requests: TargetedRumorRequest[] = []): IntelStepResult {
    91	    const view = deepFreeze(viewOf(tick, world));
    92	    const before = hashState(deepFreeze(this.state));
    93	    const stepped = stepIntel(this.config, this.state, view, deepFreeze(requests));
    94	    expect(hashState(this.state)).toBe(before);
    95	    this.state = stepped.state;
    96	    return stepped;
    97	  }
