    59	/** Recursively frozen, so a step that mutated its input would throw. */
    60	function deepFreeze<T>(value: T): T {
    61	  if (typeof value === "object" && value !== null && !Object.isFrozen(value)) {
    62	    Object.freeze(value);
    63	    for (const child of Object.values(value)) deepFreeze(child);
    64	  }
    65	  return value;
    66	}

    68	function viewOf(tick: number, world: World): IntelWorldView {
    69	  return { tick, players: Object.entries(world.players).map(([id, at]) => ({ id, ...at })), rubbings: world.rubbings };
    70	}

    72	/** Steps with frozen input and state, checking neither changes. */
    73	class Run {
    74	  state: IntelState;
    75	  constructor(
    76	    readonly config: IntelConfig,
    77	    seed = 1,
    78	  ) {
    79	    this.state = createIntelState(config, seed);
    80	  }
    81	
    82	  step(tick: number, world: World, requests: TargetedRumorRequest[] = []): IntelStepResult {
    83	    const view = deepFreeze(viewOf(tick, world));
    84	    const before = hashState(deepFreeze(this.state));
    85	    const stepped = stepIntel(this.config, this.state, view, deepFreeze(requests));
    86	    expect(hashState(this.state)).toBe(before);
    87	    this.state = stepped.state;
    88	    return stepped;
    89	  }
    90	
    91	  available(tick: number, world: World, playerId: string, type: RubbingType) {
    92	    return targetedRumorAvailability(this.config, this.state, viewOf(tick, world), playerId, type);
    93	  }
    94	}

    96	function to(result: IntelStepResult, playerId: string): IntelMessageBody[] {
    97	  return result.messages.filter((message) => message.to.kind === "player" && message.to.playerId === playerId).map((message) => message.body);
    98	}

   100	function everyone(result: IntelStepResult): IntelMessageBody[] {
   101	  return result.messages.filter((message) => message.to.kind === "everyone").map((message) => message.body);
   102	}

   104	function only<K extends IntelMessageBody["kind"]>(bodies: IntelMessageBody[], kind: K): Extract<IntelMessageBody, { kind: K }> {
   105	  const found = bodies.filter((body) => body.kind === kind);
   106	  expect(found).toHaveLength(1);
   107	  return found[0] as Extract<IntelMessageBody, { kind: K }>;
   108	}
