[theme]
# GitHub Dark Default (Primer). No built-in GitHub theme exists in Herdr, so
# this layers the GitHub Dark palette over a complete built-in dark base.
#
# THIS FILE IS WHAT RENDERS THE UI, INCLUDING FOR `herdr --remote vea-vps`.
# Per the Herdr docs: "Themes, sidebar layouts, copy behavior, and other
# presentation settings come from the client's local config, including when
# viewing an SSH machine." The remote host's [theme] block is NOT used for
# rendering, and --remote-keybindings only moves keybindings, never the theme.
# Keep this in sync with vea-vps:~/.config/herdr/config.toml.
#
# NOTE: base must NOT be "terminal". The "terminal" theme derives every
# un-overridden token from the HOST terminal's ANSI palette; when that palette
# is unavailable or reports a dark foreground, Herdr UI text renders black on
# its dark surfaces (unreadable). "one-dark" supplies every token.
name = "one-dark"
auto_switch = false
